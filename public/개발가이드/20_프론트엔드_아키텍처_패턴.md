## 1. AI 기반 아키텍처 설계

> 프론트엔드 아키텍처 설계에서 AI를 활용하면 **기존 코드베이스의 구조적 문제를 자동 탐지**하고, **최적의 모듈 경계를 데이터 기반으로 제안**받을 수 있다. **전체 아키텍처 설계 시간의 30% 이상을 AI 기반 분석에 투자**한다.

### 1.1 AI 프롬프트 7개 체계

## 2. Feature-Sliced Design (FSD) 상세 가이드

### 2.1 FSD 7계층 구조

```
src/
  app/           # 앱 초기화, 프로바이더, 라우팅, 글로벌 스타일
  processes/     # (선택) 여러 페이지에 걸친 비즈니스 프로세스
  pages/         # 라우트에 대응하는 페이지 컴포넌트
  widgets/       # 페이지를 구성하는 독립적 UI 블록
  features/      # 사용자 인터랙션 (버튼 클릭, 폼 제출 등)
  entities/      # 비즈니스 엔티티 (User, Product, Order 등)
  shared/        # 재사용 가능한 유틸, UI 키트, API 클라이언트
```

```typescript
// domain/entities/Product.ts - 순수 비즈니스 로직 (프레임워크 의존 없음)
export interface Product {
  id: string;
  name: string;
  price: number;
  stock: number;
  category: string;
}

export function canPurchase(product: Product, quantity: number): boolean {
  return product.stock >= quantity && quantity > 0;
}

export function calculateDiscount(product: Product, quantity: number): number {
  if (quantity >= 10) return product.price * quantity * 0.1;
  if (quantity >= 5) return product.price * quantity * 0.05;
  return 0;
}

export function calculateTotal(product: Product, quantity: number): number {
  return product.price * quantity - calculateDiscount(product, quantity);
}
```

```typescript
// application/usecases/PurchaseProduct.ts - 유즈케이스
import type { Product } from '@/domain/entities/Product';
import { canPurchase, calculateTotal } from '@/domain/entities/Product';
import type { ProductRepository } from '@/application/ports/ProductRepository';

export interface PurchaseProductInput {
  productId: string;
  quantity: number;
  userId: string;
}

export interface PurchaseProductOutput {
  success: boolean;
  orderId?: string;
  totalAmount?: number;
  error?: string;
}

export class PurchaseProductUseCase {
  constructor(
    private readonly productRepo: ProductRepository,
    private readonly orderService: OrderService,
  ) {}

  async execute(input: PurchaseProductInput): Promise<PurchaseProductOutput> {
    const product = await this.productRepo.findById(input.productId);
    if (!product) {
      return { success: false, error: '상품을 찾을 수 없습니다.' };
    }

    if (!canPurchase(product, input.quantity)) {
      return { success: false, error: '재고가 부족합니다.' };
    }

    const totalAmount = calculateTotal(product, input.quantity);
    const orderId = await this.orderService.createOrder({
      userId: input.userId,
      productId: input.productId,
      quantity: input.quantity,
      totalAmount,
    });

    return { success: true, orderId, totalAmount };
  }
}

interface OrderService {
  createOrder(params: {
    userId: string;
    productId: string;
    quantity: number;
    totalAmount: number;
  }): Promise<string>;
```

```typescript
import type { Product } from '@/domain/entities/Product';

export function useProductFilter(products: Product[]) {
  const [category, setCategory] = useState<string>('all');
  const [searchQuery, setSearchQuery] = useState('');

  const filtered = useMemo(() => {
    return products.filter((p) => {
      const matchCategory = category === 'all' || p.category === category;
      const matchSearch = p.name.toLowerCase().includes(searchQuery.toLowerCase());
      return matchCategory && matchSearch;

import type { Product } from '@/domain/entities/Product';

type SortKey = 'name' | 'price' | 'stock';
type SortOrder = 'asc' | 'desc';

export function useProductSort(products: Product[]) {
  const [sortKey, setSortKey] = useState<SortKey>('name');
  const [sortOrder, setSortOrder] = useState<SortOrder>('asc');

  const sorted = useMemo(() => {
    return [...products].sort((a, b) => {
      const aVal = a[sortKey];
      const bVal = b[sortKey];
      const cmp = typeof aVal === 'string' ? aVal.localeCompare(bVal as string) : (aVal as number) - (bVal as number);
      return sortOrder === 'asc' ? cmp : -cmp;
    });
  }, [products, sortKey, sortOrder]);

  return { sorted, sortKey, setSortKey, sortOrder, setSortOrder };
}
```

### 6.2 모듈 경계 자동 검증 스크립트

```typescript
// scripts/check-module-boundaries.ts
import * as fs from 'fs';
import * as path from 'path';

interface BoundaryCheckResult {
  totalFiles: number;
  violations: BoundaryViolation[];
  score: number;
}

interface BoundaryViolation {
  file: string;
  importPath: string;
  violationType: 'layer-skip' | 'cross-slice' | 'deep-import' | 'circular';
  message: string;
}

const LAYER_ORDER = ['app', 'processes', 'pages', 'widgets', 'features', 'entities', 'shared'];

function getLayer(filePath: string): string | null {
  const match = filePath.match(/src\/(\w+)\//);
  return match ? match[1] : null;
}

function getSlice(filePath: string): string | null {
  const match = filePath.match(/src\/\w+\/(\w+)/);
  return match ? match[1] : null;
}

function checkImport(fromFile: string, importPath: string): BoundaryViolation | null {
  const fromLayer = getLayer(fromFile);
  const toLayer = getLayer(importPath);

  if (!fromLayer || !toLayer) return null;

  const fromIndex = LAYER_ORDER.indexOf(fromLayer);
  const toIndex = LAYER_ORDER.indexOf(toLayer);

  // 상위 계층에서 하위 계층으로만 의존 가능
  if (fromIndex > toIndex) {
    return {
      file: fromFile,
      importPath,
      violationType: 'layer-skip',
      message: `${fromLayer} -> ${toLayer}: 하위 계층이 상위 계층을 참조`,
    };
  }

  // 같은 계층의 다른 슬라이스 간 직접 import 불가
  if (fromIndex === toIndex && fromLayer !== 'shared') {
    const fromSlice = getSlice(fromFile);
    const toSlice = getSlice(importPath);
    if (fromSlice && toSlice && fromSlice !== toSlice) {
      return {
        file: fromFile,
        importPath,
        violationType: 'cross-slice',
        message: `${fromLayer}/${fromSlice} -> ${toLayer}/${toSlice}: 동일 계층 간 직접 참조`,
      };
    }
  }

  return null;
}

// CI에서 실행: npx tsx scripts/check-module-boundaries.ts
console.log('Module boundary check completed');
export { checkImport, type BoundaryCheckResult };
```

```typescript
interface ArchitectureFlags {
  // 새로운 아키텍처 패턴으로의 점진적 전환을 제어
  useNewDataLayer: boolean;     // 새 데이터 계층 사용

  useNewDIContainer: boolean;   // 새 DI 컨테이너 사용
}

// Feature Flag 기반 서비스 선택
function createDataLayer(flags: ArchitectureFlags) {
  if (flags.useNewDataLayer) {
    // 새로운 Clean Architecture 기반 데이터 계층
    return {
      productRepo: new CleanProductRepository(),
      orderRepo: new CleanOrderRepository(),
    };
  }

  // 레거시 데이터 계층
  return {
    productRepo: new LegacyProductRepository(),
    orderRepo: new LegacyOrderRepository(),
  };
}

// Preview 환경에서 새 아키텍처 사전 검증
function resolveArchitectureFlags(): ArchitectureFlags {
  const params = new URLSearchParams(window.location.search);

  return {
    useNewDataLayer: params.get('arch_new_data') === 'true',
    useEventDrivenComm: params.get('arch_event_driven') === 'true',
    useNewDIContainer: params.get('arch_new_di') === 'true',
  };
}

export { createDataLayer, resolveArchitectureFlags };
export type { ArchitectureFlags };

// 타입 선언
declare class CleanProductRepository {}
declare class CleanOrderRepository {}
declare class LegacyProductRepository {}
declare class LegacyOrderRepository {}
```

## 8. 아키텍처 의사결정 프레임워크

### 8.1 가중치 매트릭스 (Weighted Decision Matrix)

```typescript
// tools/architecture-decision-matrix.ts
interface DecisionCriterion {
  name: string;
  weight: number; // 1-10
  description: string;
}

interface ArchitectureOption {
  name: string;
  description: string;
  scores: Record<string, number>; // criterion name -> score (1-10)
}

interface DecisionResult {
  winner: string;
  rankings: { option: string; totalScore: number; normalizedScore: number }[];
  analysis: string;
}

const DEFAULT_CRITERIA: DecisionCriterion[] = [
  { name: 'maintainability', weight: 9, description: '코드 유지보수 용이성' },
  { name: 'scalability', weight: 8, description: '팀/기능 확장 시 대응력' },
  { name: 'testability', weight: 8, description: '테스트 작성 및 실행 용이성' },
  { name: 'onboarding', weight: 7, description: '새 팀원 학습 곡선' },
  { name: 'performance', weight: 6, description: '런타임 성능 영향' },
  { name: 'ecosystem', weight: 5, description: '라이브러리/도구 호환성' },
  { name: 'migration-cost', weight: 7, description: '기존 코드 마이그레이션 비용' },
];

function evaluateArchitectures(
  criteria: DecisionCriterion[],
  options: ArchitectureOption[],
): DecisionResult {
  const totalWeight = criteria.reduce((sum, c) => sum + c.weight, 0);

  const rankings = options.map((option) => {
    const totalScore = criteria.reduce((sum, criterion) => {
      const score = option.scores[criterion.name] || 0;
      return sum + score * criterion.weight;
    }, 0);

    return {
      option: option.name,
      totalScore,
      normalizedScore: Math.round((totalScore / (totalWeight * 10)) * 100),
    };
  });

  rankings.sort((a, b) => b.totalScore - a.totalScore);

  return {
    winner: rankings[0].option,
    rankings,
    analysis: `${rankings[0].option}이 가중 점수 ${rankings[0].normalizedScore}점으로 1위`,
  };
}

// 사용 예: FSD vs Clean Architecture vs Hexagonal
const options: ArchitectureOption[] = [
  {
    name: 'Feature-Sliced Design',
    description: '기능 단위 슬라이싱',
    scores: {
      maintainability: 9,
      scalability: 9,
      testability: 7,
      onboarding: 6,
      performance: 8,
      ecosystem: 7,
      'migration-cost': 5,
    },
  },
  {
    name: 'Clean Architecture',
    description: '계층형 클린 아키텍처',
    scores: {
      maintainability: 8,
      scalability: 8,
      testability: 9,
      onboarding: 5,
      performance: 7,
      ecosystem: 6,
      'migration-cost': 4,
    },
  },
  {
    name: 'Hexagonal Architecture',
    description: '포트와 어댑터',
    scores: {
      maintainability: 8,
      scalability: 7,
      testability: 10,
      onboarding: 4,
      performance: 7,
      ecosystem: 5,
      'migration-cost': 3,
    },
  },
];

const result = evaluateArchitectures(DEFAULT_CRITERIA, options);
console.log('Decision Result:', result);
```

### 8.2 아키텍처 패턴별 비교표

| 기준 | FSD | Clean Architecture | Hexagonal | 단순 디렉토리 |
|---|---|---|---|---|
| 학습 곡선 | 중간 | 높음 | 높음 | 낮음 |
| 확장성 | 높음 | 높음 | 중간 | 낮음 |
| 테스트 용이성 | 중간 | 높음 | 매우 높음 | 낮음 |
| 팀 규모 적합 | 3-15명 | 5-20명 | 5-15명 | 1-5명 |
| 프로젝트 규모 | 중-대 | 대 | 중-대 | 소-중 |
| 프레임워크 독립성 | 낮음 | 높음 | 매우 높음 | 낮음 |
| 보일러플레이트 | 중간 | 많음 | 많음 | 적음 |
| 마이그레이션 비용 | 중간 | 높음 | 높음 | - |

### 8.3 의사결정 플로우차트

```
팀 규모 <= 3명?
  ├─ Yes -> 단순 디렉토리 구조
  └─ No
      ├─ 프레임워크 교체 가능성 있음?
      │   ├─ Yes -> Hexagonal Architecture
      │   └─ No
      │       ├─ 복잡한 비즈니스 로직 중심?
      │       │   ├─ Yes -> Clean Architecture
      │       │   └─ No -> Feature-Sliced Design
      └─ 마이크로 프론트엔드?
          └─ Yes -> FSD + Module Federation (19_가이드 참조)
```
