// playwright.config.ts
import { defineConfig, devices } from "@playwright/test";

type EnvironmentType = "production" | "beta" | "preview";

interface TestEnvironment {
  name: string;
  type: EnvironmentType;
  baseURL: string;
}

function resolveEnvironments(): TestEnvironment[] {
  const cdnDomain = process.env.CDN_DOMAIN ?? "app.example.com";
  const envList = process.env.TEST_ENVIRONMENTS ?? "production";

  return envList.split(",").map((env) => {
    const trimmed = env.trim();
    if (trimmed === "production") {
      return { name: "production", type: "production", baseURL: `https://${cdnDomain}` };
    }
    if (trimmed.startsWith("beta-")) {
      return { name: trimmed, type: "beta", baseURL: `https://${cdnDomain}/${trimmed}` };
    }
    if (trimmed.startsWith("pr-")) {
      return { name: trimmed, type: "preview", baseURL: `https://${cdnDomain}/${trimmed}` };
    }
    return { name: trimmed, type: "preview", baseURL: `https://${cdnDomain}/${trimmed}` };
  });
}

function buildBrowserStackProjects() {
  return [

function buildBrowserStackWsUrl(
  browser: string,
  os: string,
  browserVersion: string,
  device?: string,
): string {
  const caps = {
    browser,
    os,
    browser_version: browserVersion,
    ...(device ? { device } : {}),
    "browserstack.username": process.env.BROWSERSTACK_USERNAME,
    "browserstack.accessKey": process.env.BROWSERSTACK_ACCESS_KEY,
    "browserstack.playwrightVersion": "1.48.0",
    project: "Multi-Beta Compat Test",
    build: `${process.env.TEST_ENVIRONMENT ?? "local"}-${Date.now()}`,
  };
  const encoded = encodeURIComponent(JSON.stringify(caps));
  return `wss://cdp.browserstack.com/playwright?caps=${encoded}`;
}

### 1.2 환경별 테스트 실행 매트릭스 스크립트

```typescript
// scripts/run-compat-matrix.ts
import { execSync } from "node:child_process";

interface MatrixConfig {
  environments: string[];
  browsers: string[];
  testPattern: string;
  parallel: boolean;
}

interface MatrixResult {
  environment: string;
  browser: string;
  passed: number;
  failed: number;
  skipped: number;
  duration: number;
}

function runTestMatrix(config: MatrixConfig): MatrixResult[] {
  const results: MatrixResult[] = [];

  for (const env of config.environments) {
    for (const browser of config.browsers) {
      console.log(`[Matrix] ${env} x ${browser} 테스트 시작...`);

      const startTime = Date.now();
      try {
        const output = execSync(
          [
            "npx playwright test",
            `--project=${browser}`,
            `--grep="${config.testPattern}"`,
            "--reporter=json",
          ].join(" "),
          {
            encoding: "utf-8",
            env: {
              ...process.env,
              TEST_ENVIRONMENTS: env,
              PLAYWRIGHT_JSON_OUTPUT_FILE: `test-results/${env}-${browser}.json`,
            },
            maxBuffer: 50 * 1024 * 1024,
          },
        );

        const report = JSON.parse(output);
        results.push({
          environment: env,
          browser,
          passed: report.stats?.expected ?? 0,
          failed: report.stats?.unexpected ?? 0,
          skipped: report.stats?.skipped ?? 0,
          duration: Date.now() - startTime,
        });
      } catch (error) {
        results.push({
          environment: env,
          browser,
          passed: 0,
          failed: 1,
          skipped: 0,
          duration: Date.now() - startTime,
        });
      }
    }
  }

  return results;
}

function generateMatrixReport(results: MatrixResult[]): string {
  const header = "| 환경 | 브라우저 | 성공 | 실패 | 스킵 | 시간 |";
  const separator = "|------|---------|------|------|------|------|";
  const rows = results.map(
    (r) =>
      `| ${r.environment} | ${r.browser} | ${r.passed} | ${r.failed} | ${r.skipped} | ${(r.duration / 1000).toFixed(1)}s |`,
  );

  return [header, separator, ...rows].join("\n");
}

// 실행
const config: MatrixConfig = {
  environments: (process.env.TEST_ENVIRONMENTS ?? "production").split(","),
  browsers: ["chromium", "firefox", "webkit", "mobile-chrome", "mobile-safari"],
  testPattern: "compatibility|visual",
  parallel: true,
};

const results = runTestMatrix(config);
console.log("\n## 크로스 브라우저 테스트 매트릭스 결과\n");
console.log(generateMatrixReport(results));

export { runTestMatrix, generateMatrixReport };
export type { MatrixConfig, MatrixResult };
```

### 1.3 환경별 호환성 테스트

```typescript
// e2e/multi-beta-compatibility.spec.ts
import { test, expect } from "@playwright/test";

const TARGET_ENV = process.env.TEST_ENVIRONMENTS ?? "production";

test.describe(`[${TARGET_ENV}] 크로스 브라우저 호환성`, () => {
  test("CSS Container Queries 동작 확인", async ({ page }) => {
    await page.goto("/components/card-grid");
    await page.waitForLoadState("networkidle");

    // 넓은 뷰포트
    await page.setViewportSize({ width: 1200, height: 800 });
    const cardBody = page.locator(".card-body").first();
    await expect(cardBody).toHaveCSS("display", "grid");

    // 좁은 뷰포트
    await page.setViewportSize({ width: 400, height: 800 });
    await expect(cardBody).toHaveCSS("display", "flex");
  });

  test("View Transitions API 동작 확인", async ({ page }) => {
    await page.goto("/");

    const hasViewTransitions = await page.evaluate(
      () => "startViewTransition" in document,
    );

    if (hasViewTransitions) {
      await page.click('a[href$="/about"]');
      await page.waitForFunction(
        () => !document.querySelector("::view-transition"),
        null,
        { timeout: 3000 },
      ).catch(() => { /* 전환 완료 또는 미지원 */ });
    } else {
      await page.click('a[href$="/about"]');
    }

    await expect(page).toHaveURL(/\/about$/);
  });

  test("CSS :has() 선택자 동작 확인", async ({ page }) => {
    await page.goto("/components/form");

    const emailInput = page.locator('input[type="email"]');
    await emailInput.fill("invalid-email");
    await emailInput.blur();

    const formGroup = page.locator(".form-group").filter({ has: emailInput });
    await expect(formGroup).toHaveCSS("border-color", "rgb(239, 68, 68)");
  });

  test("CSS Nesting 동작 확인", async ({ page }) => {
    await page.goto("/components/nav");

    const link = page.locator(".nav a").first();
    const defaultColor = await link.evaluate((el) =>
      getComputedStyle(el).color,
    );
    expect(defaultColor).toBeTruthy();

    await link.hover();
    const hoverColor = await link.evaluate((el) =>
      getComputedStyle(el).color,
    );
    expect(hoverColor).not.toBe(defaultColor);
  });

  test("시각적 스냅샷 비교", async ({ page, browserName }) => {
    const pages = [
      { name: "home", path: "/" },
      { name: "about", path: "/about" },
    ];

    for (const { name, path } of pages) {
      await page.goto(path);
      await page.waitForLoadState("networkidle");

      await expect(page).toHaveScreenshot(
        `${TARGET_ENV}-${browserName}-${name}.png`,
        {
          maxDiffPixelRatio: 0.02,
          fullPage: true,
        },
      );
    }
  });
});

## 2. Preview 환경에서 호환성 자동 검증 CI

### 2.1 PR별 크로스 브라우저 검증 워크플로우

```yaml
# .github/workflows/preview-compat-test.yml
name: Preview Compatibility Test

on:
  pull_request:
    types: [opened, synchronize]

permissions:
  contents: read
  pull-requests: write
  checks: write

concurrency:
  group: compat-${{ github.event.pull_request.number }}
  cancel-in-progress: true

jobs:
  deploy-preview:
    runs-on: ubuntu-latest
    outputs:
      preview_url: ${{ steps.deploy.outputs.preview_url }}
    steps:
      - uses: actions/checkout@v4
      - uses: pnpm/action-setup@v4
      - uses: actions/setup-node@v4
        with:
          node-version: 22
          cache: pnpm

      - run: pnpm install --frozen-lockfile

      - name: Build
        run: pnpm build
        env:
          VITE_BASE_PATH: /pr-${{ github.event.pull_request.number }}/

      - name: Configure AWS credentials
        uses: aws-actions/configure-aws-credentials@v4
        with:
          role-to-assume: ${{ secrets.AWS_DEPLOY_ROLE_ARN }}
          aws-region: ap-northeast-2

      - name: Deploy to S3
        id: deploy
        run: |
          aws s3 sync dist/ \
            s3://${{ vars.PREVIEW_BUCKET }}/pr-${{ github.event.pull_request.number }}/ \
            --delete
          echo "preview_url=https://${{ vars.CDN_DOMAIN }}/pr-${{ github.event.pull_request.number }}/" >> "$GITHUB_OUTPUT"

  cross-browser-test:
    needs: deploy-preview
    runs-on: ubuntu-latest
    strategy:
      fail-fast: false
      matrix:
        browser: [chromium, firefox, webkit]
        include:
          - browser: chromium
            label: Chrome
          - browser: firefox
            label: Firefox
          - browser: webkit
            label: Safari

    steps:
      - uses: actions/checkout@v4
      - uses: pnpm/action-setup@v4
      - uses: actions/setup-node@v4
        with:
          node-version: 22
          cache: pnpm

      - run: pnpm install --frozen-lockfile
      - run: npx playwright install --with-deps ${{ matrix.browser }}

      - name: Run compatibility tests
        run: |
          npx playwright test \
            --project=${{ matrix.browser }} \
            e2e/multi-beta-compatibility.spec.ts
        env:
          TEST_ENVIRONMENTS: pr-${{ github.event.pull_request.number }}
          PLAYWRIGHT_BASE_URL: ${{ needs.deploy-preview.outputs.preview_url }}

      - uses: actions/upload-artifact@v4
        if: always()
        with:
          name: test-results-${{ matrix.browser }}
          path: test-results/
          retention-days: 7

  browserstack-test:
    needs: deploy-preview
    runs-on: ubuntu-latest
    if: contains(github.event.pull_request.labels.*.name, 'compat-full')
    steps:
      - uses: actions/checkout@v4
      - uses: pnpm/action-setup@v4
      - uses: actions/setup-node@v4
        with:
          node-version: 22
          cache: pnpm

      - run: pnpm install --frozen-lockfile

      - name: Run BrowserStack tests
        run: |
          npx playwright test \
            --project=bs-chrome-win \
            --project=bs-firefox-win \
            --project=bs-safari-mac \
            --project=bs-edge-win \
            --project=bs-chrome-android \
            --project=bs-safari-ios \
            e2e/multi-beta-compatibility.spec.ts
        env:
          TEST_ENVIRONMENTS: pr-${{ github.event.pull_request.number }}
          PLAYWRIGHT_BASE_URL: ${{ needs.deploy-preview.outputs.preview_url }}
          USE_BROWSERSTACK: "true"
          BROWSERSTACK_USERNAME: ${{ secrets.BROWSERSTACK_USERNAME }}
          BROWSERSTACK_ACCESS_KEY: ${{ secrets.BROWSERSTACK_ACCESS_KEY }}

  report:
    needs: [cross-browser-test, browserstack-test]
    if: always()
    runs-on: ubuntu-latest
    steps:
      - uses: actions/checkout@v4

      - uses: actions/download-artifact@v4
        with:
          path: all-results/

      - name: Generate compatibility report
        uses: actions/github-script@v7
        with:
          script: |
            const fs = require('fs');
            const path = require('path');
            const prNumber = context.payload.pull_request.number;

            // 결과 파일 수집
            let reportBody = '## 크로스 브라우저 호환성 테스트 결과\n\n';
            reportBody += `| 브라우저 | 결과 | 세부사항 |\n`;
            reportBody += `|---------|------|----------|\n`;

            const resultDirs = fs.readdirSync('all-results/').filter(d => d.startsWith('test-results-'));
            for (const dir of resultDirs) {
              const browser = dir.replace('test-results-', '');
              const reportPath = path.join('all-results', dir, 'report.json');
              if (fs.existsSync(reportPath)) {
                const report = JSON.parse(fs.readFileSync(reportPath, 'utf-8'));
                const passed = report.stats?.expected ?? 0;
                const failed = report.stats?.unexpected ?? 0;
                const status = failed === 0 ? 'PASS' : 'FAIL';
                reportBody += `| ${browser} | ${status} | ${passed} passed, ${failed} failed |\n`;
              } else {
                reportBody += `| ${browser} | SKIP | 결과 파일 없음 |\n`;
              }
            }

            reportBody += '\n> `compat-full` 라벨을 추가하면 BrowserStack 실기기 테스트도 실행됩니다.';

            // PR 코멘트
            const { data: comments } = await github.rest.issues.listComments({
              owner: context.repo.owner,
              repo: context.repo.repo,
              issue_number: prNumber,
            });
            const existing = comments.find(c => c.body?.includes('크로스 브라우저 호환성 테스트'));
            if (existing) {
              await github.rest.issues.updateComment({
                owner: context.repo.owner,
                repo: context.repo.repo,
                comment_id: existing.id,
                body: reportBody,
              });
            } else {
              await github.rest.issues.createComment({
                owner: context.repo.owner,
                repo: context.repo.repo,
                issue_number: prNumber,
                body: reportBody,
              });
            }
```

// utils/baseline-check.ts
const BASELINE_CHECKS = {
  containerQueries: () => CSS.supports("container-type", "inline-size"),
  viewTransitions: () => "startViewTransition" in document,
  importMaps: () => HTMLScriptElement.supports?.("importmap") ?? false,
  cssNesting: () => CSS.supports("selector(&)"),
  cascadeLayers: () => CSS.supports("@layer base"),
  subgrid: () => CSS.supports("grid-template-rows", "subgrid"),
  hasSelector: () => CSS.supports("selector(:has(*))"),
  popover: () => "popover" in HTMLElement.prototype,
} as const;

type BaselineFeature = keyof typeof BASELINE_CHECKS;

function checkBaseline(features: BaselineFeature[]): Record<BaselineFeature, boolean> {
  const result = {} as Record<BaselineFeature, boolean>;
  for (const feature of features) {
    try {
      result[feature] = BASELINE_CHECKS[feature]();
    } catch {
      result[feature] = false;
    }
  }
  return result;
}

export { checkBaseline, BASELINE_CHECKS };
export type { BaselineFeature };
```

## 7. Polyfill 전략

### 7.1 조건부 Polyfill 로딩

```typescript
// polyfills/index.ts
async function loadPolyfills(): Promise<void> {
  const polyfills: Promise<unknown>[] = [];

  if (typeof Object.groupBy !== "function") {
    polyfills.push(import("core-js/actual/object/group-by"));
  }

  if (typeof Promise.withResolvers !== "function") {
    polyfills.push(import("core-js/actual/promise/with-resolvers"));
  }

  if (typeof Set.prototype.union !== "function") {
    polyfills.push(import("core-js/actual/set"));
  }

  if (polyfills.length > 0) {
    await Promise.all(polyfills);
    console.info(`[Polyfill] ${polyfills.length}개 로드 완료`);
  }
}

export { loadPolyfills };
```

### 7.2 Feature Detection + 조건부 로딩 유틸

```typescript
// utils/feature-detect.ts
interface BrowserFeatures {
  containerQueries: boolean;
  viewTransitions: boolean;
  hasSelector: boolean;
  cascadeLayers: boolean;
  subgrid: boolean;
}

function detectFeatures(): BrowserFeatures {
  return {
    containerQueries: CSS.supports("container-type", "inline-size"),
    viewTransitions: "startViewTransition" in document,
    hasSelector: CSS.supports("selector(:has(*))"),
    cascadeLayers: CSS.supports("@layer base"),
    subgrid: CSS.supports("grid-template-rows", "subgrid"),
  };
}

function loadPolyfillsConditionally(features: BrowserFeatures): void {
  const polyfillMap: Partial<Record<keyof BrowserFeatures, string>> = {
    containerQueries: "/polyfills/container-queries.js",
    viewTransitions: "/polyfills/view-transitions.js",
  };

  for (const [feature, url] of Object.entries(polyfillMap)) {
    if (!features[feature as keyof BrowserFeatures]) {
      const script = document.createElement("script");
      script.src = url;
      script.async = true;
      document.head.appendChild(script);
    }
  }
}

export { detectFeatures, loadPolyfillsConditionally };
export type { BrowserFeatures };
```

## 8. AI 기반 호환성 이슈 자동 탐지

### 8.1 eslint-plugin-compat + AI 분석 파이프라인

```typescript
// scripts/ai-compat-check.ts
import { execSync } from "node:child_process";
import { writeFileSync } from "node:fs";

interface CompatIssue {
  file: string;
  line: number;
  rule: string;
  message: string;
  api: string;
}

function runEslintCompat(): CompatIssue[] {
  const result = execSync(
    'npx eslint --format json --rule \'{"compat/compat": "error"}\' src/',
    { encoding: "utf-8", maxBuffer: 10 * 1024 * 1024 },
  );

  const parsed = JSON.parse(result) as Array<{
    filePath: string;
    messages: Array<{ line: number; ruleId: string; message: string }>;
  }>;

  return parsed.flatMap((file) =>
    file.messages
      .filter((msg) => msg.ruleId === "compat/compat")
      .map((msg) => ({
        file: file.filePath,
        line: msg.line,
        rule: msg.ruleId,
        message: msg.message,
        api: msg.message.match(/'([^']+)'/)?.[1] ?? "unknown",
      })),
  );
}

function generateAiPrompt(issues: CompatIssue[]): string {
  const grouped = Object.groupBy(issues, (i) => i.api);
  const summary = Object.entries(grouped)
    .map(
      ([api, items]) =>
        `- ${api}: ${items!.length}건 (${items!.map((i) => `${i.file}:${i.line}`).join(", ")})`,
    )
    .join("\n");

  return `다음 호환성 이슈에 대해 각각 Polyfill 또는 대체 방안을 TypeScript 코드로 제시해줘.\n\n${summary}`;
}

const issues = runEslintCompat();
const prompt = generateAiPrompt(issues);
writeFileSync("compat-ai-prompt.txt", prompt, "utf-8");
console.log(`[AI Compat] ${issues.length}건 발견. compat-ai-prompt.txt 저장 완료.`);
```

### 8.2 테스트 결과 AI 분석

```typescript
// scripts/analyze-test-results.ts
import { readFileSync } from "node:fs";

interface TestResult {
  title: string;
  status: "passed" | "failed" | "skipped";
  browser: string;
  duration: number;
  error?: string;
  environment?: string;
}

function parsePlaywrightReport(reportPath: string): TestResult[] {
  const raw = readFileSync(reportPath, "utf-8");
  const report = JSON.parse(raw) as {
    suites: Array<{
      specs: Array<{
        title: string;
        tests: Array<{
          projectName: string;
          status: string;
          duration: number;
          results: Array<{ error?: { message: string } }>;
        }>;
      }>;
    }>;
  };

  return report.suites.flatMap((suite) =>
    suite.specs.flatMap((spec) =>
      spec.tests.map((test) => ({
        title: spec.title,
        status: test.status as TestResult["status"],
        browser: test.projectName,
        duration: test.duration,
        error: test.results[0]?.error?.message,
      })),
    ),
  );
}

function buildAnalysisPrompt(results: TestResult[]): string {
  const failures = results.filter((r) => r.status === "failed");
  const byBrowser = Object.groupBy(failures, (f) => f.browser);

  return `다음은 Playwright 크로스 브라우저 테스트 실패 결과이다.
브라우저별 실패 원인을 분석하고, 호환성 문제인지 로직 버그인지 분류해줘.
각 문제에 대해 수정 코드를 TypeScript로 제시해줘.

--- 실패 목록 (${failures.length}건) ---
${failures.map((f) => `[${f.browser}] ${f.title}: ${f.error ?? "unknown"}`).join("\n")}

--- 브라우저별 실패 수 ---
${Object.entries(byBrowser).map(([b, fs]) => `${b}: ${fs!.length}건`).join("\n")}`;
}

const results = parsePlaywrightReport("test-results/report.json");
console.log(buildAnalysisPrompt(results));

export { parsePlaywrightReport, buildAnalysisPrompt };
export type { TestResult };
```
