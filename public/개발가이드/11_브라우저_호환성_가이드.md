### 1.3 자동 탐지 파이프라인 통합

```typescript
// scripts/ai-compat-check.ts
import { execSync } from "node:child_process";
import { readFileSync, writeFileSync } from "node:fs";

interface CompatIssue {
  file: string;
  line: number;
  rule: string;
  message: string;
  api: string;
}

function runEslintCompat(): CompatIssue[] {
  const result = execSync(
    'npx eslint --format json --rule \'{"compat/compat": "error"}\'  src/',
    { encoding: "utf-8", maxBuffer: 10 * 1024 * 1024 },
  );

  const parsed = JSON.parse(result) as Array<{
    filePath: string;
    messages: Array<{ line: number; ruleId: string; message: string }>;
  }>;

  return parsed.flatMap((file) =>
    file.messages
      .filter((msg) => msg.ruleId === "compat/compat")
      .map((msg) => ({
        file: file.filePath,
        line: msg.line,
        rule: msg.ruleId,
        message: msg.message,
        api: extractApiName(msg.message),
      })),
  );
}

function extractApiName(message: string): string {
  const match = message.match(/'([^']+)'/);
  return match?.[1] ?? "unknown";
}

function generateAiPrompt(issues: CompatIssue[]): string {
  const grouped = Object.groupBy(issues, (i) => i.api);
  const summary = Object.entries(grouped)
    .map(([api, items]) => `- ${api}: ${items!.length}건 (${items!.map((i) => `${i.file}:${i.line}`).join(", ")})`)
    .join("\n");

  return `다음 호환성 이슈에 대해 각각 Polyfill 또는 대체 방안을 TypeScript 코드로 제시해줘.\n\n${summary}`;
}

// 실행
const issues = runEslintCompat();
const prompt = generateAiPrompt(issues);
writeFileSync("compat-ai-prompt.txt", prompt, "utf-8");

console.log(`[AI Compat] ${issues.length}건 발견. compat-ai-prompt.txt에 프롬프트 저장 완료.`);
```

## 2. AI 기반 Polyfill 코드 자동 생성

Copilot, Cursor 등 AI 코딩 도구를 활용하여 Feature Detection + Polyfill 코드를 자동 생성한다.

### 2.1 Copilot / Cursor 활용 패턴

```typescript
// AI에게 다음 주석과 함께 코드 생성 요청:
// Copilot: "structuredClone이 없는 브라우저를 위한 polyfill을 작성해줘.
//           Feature detection을 포함하고, 순환 참조를 처리해야 해."

// AI가 생성하는 결과물 예시:
function ensureStructuredClone(): void {
  if (typeof globalThis.structuredClone === "function") {
    return;
  }

  globalThis.structuredClone = <T>(value: T): T => {
    // MessageChannel 기반 deep clone (순환 참조 지원)
    if (typeof MessageChannel !== "undefined") {
      return new Promise<T>((resolve) => {
        const channel = new MessageChannel();
        channel.port1.onmessage = (event) => resolve(event.data as T);
        channel.port2.postMessage(value);
      }) as unknown as T;
    }
    // 최소 fallback (순환 참조 미지원)
    return JSON.parse(JSON.stringify(value)) as T;
  };
}
```

### 2.2 Feature Detection 유틸 자동 생성

```typescript
// utils/feature-detect.ts
// AI 생성 대상: "주요 모던 API의 feature detection 유틸을 만들어줘"

interface BrowserFeatures {
  containerQueries: boolean;
  viewTransitions: boolean;
  importMaps: boolean;
  hasSelector: boolean;
  cascadeLayers: boolean;
  subgrid: boolean;
}

function detectFeatures(): BrowserFeatures {
  return {
    containerQueries: CSS.supports("container-type", "inline-size"),
    viewTransitions: "startViewTransition" in document,
    importMaps: HTMLScriptElement.supports?.("importmap") ?? false,
    hasSelector: CSS.supports("selector(:has(*))"),
    cascadeLayers: CSS.supports("@layer base"),
    subgrid: CSS.supports("grid-template-rows", "subgrid"),
  };
}

function loadPolyfillsConditionally(features: BrowserFeatures): void {
  const polyfillMap: Record<string, string> = {
    containerQueries: "/polyfills/container-queries.js",
    viewTransitions: "/polyfills/view-transitions.js",
  };

  for (const [feature, url] of Object.entries(polyfillMap)) {
    if (!features[feature as keyof BrowserFeatures]) {
      const script = document.createElement("script");
      script.src = url;
      script.async = true;
      document.head.appendChild(script);
    }
  }
}

export { detectFeatures, loadPolyfillsConditionally };
export type { BrowserFeatures };
```

## 3. AI 기반 크로스 브라우저 테스트 결과 분석

Playwright 테스트 결과를 AI에 전달하여, 브라우저별 실패 패턴 분석과 수정 방안을 자동으로 도출한다.

### 3.1 테스트 결과 수집 및 AI 분석

```typescript
// scripts/analyze-test-results.ts
import { readFileSync } from "node:fs";

interface TestResult {
  title: string;
  status: "passed" | "failed" | "skipped";
  browser: string;
  duration: number;
  error?: string;
  screenshot?: string;
}

interface AnalysisReport {
  browserFailureMap: Record<string, TestResult[]>;
  commonPatterns: string[];
  aiPrompt: string;
}

function parsePlaywrightReport(reportPath: string): TestResult[] {
  const raw = readFileSync(reportPath, "utf-8");
  const report = JSON.parse(raw) as {
    suites: Array<{
      specs: Array<{
        title: string;
        tests: Array<{
          projectName: string;
          status: string;
          duration: number;
          results: Array<{ error?: { message: string } }>;
        }>;
      }>;
    }>;
  };

  return report.suites.flatMap((suite) =>
    suite.specs.flatMap((spec) =>
      spec.tests.map((test) => ({
        title: spec.title,
        status: test.status as TestResult["status"],
        browser: test.projectName,
        duration: test.duration,
        error: test.results[0]?.error?.message,
      })),
    ),
  );
}

function buildAnalysisReport(results: TestResult[]): AnalysisReport {
  const failures = results.filter((r) => r.status === "failed");
  const browserFailureMap = Object.groupBy(failures, (f) => f.browser) as Record<string, TestResult[]>;

  const commonPatterns = findCommonPatterns(failures);

  const aiPrompt = `다음은 Playwright 크로스 브라우저 테스트 실패 결과이다.
브라우저별 실패 원인을 분석하고, 호환성 문제인지 로직 버그인지 분류해줘.
각 문제에 대해 수정 코드를 TypeScript로 제시해줘.

--- 실패 목록 (${failures.length}건) ---
${failures.map((f) => `[${f.browser}] ${f.title}: ${f.error ?? "unknown"}`).join("\n")}

--- 브라우저별 실패 수 ---
${Object.entries(browserFailureMap).map(([b, fs]) => `${b}: ${fs.length}건`).join("\n")}`;

  return { browserFailureMap, commonPatterns, aiPrompt };
}

function findCommonPatterns(failures: TestResult[]): string[] {
  const errorMessages = failures.map((f) => f.error).filter(Boolean) as string[];
  const patterns: string[] = [];

  const keywords = ["timeout", "not found", "not supported", "undefined", "TypeError"];
  for (const keyword of keywords) {
    const count = errorMessages.filter((msg) => msg.toLowerCase().includes(keyword.toLowerCase())).length;
    if (count >= 2) {
      patterns.push(`"${keyword}" 패턴: ${count}건`);
    }
  }
  return patterns;
}

// 실행
const results = parsePlaywrightReport("test-results/report.json");
const report = buildAnalysisReport(results);
console.log(report.aiPrompt);
```

### 3.2 시각적 회귀 테스트 AI 분석

```typescript
// scripts/visual-diff-ai.ts
import { readFileSync, readdirSync } from "node:fs";
import { join } from "node:path";

interface VisualDiff {
  testName: string;
  browser: string;
  baselinePath: string;
  actualPath: string;
  diffPath: string;
  diffPercentage: number;
}

function collectVisualDiffs(diffDir: string): VisualDiff[] {
  const files = readdirSync(diffDir).filter((f) => f.endsWith("-diff.png"));

  return files.map((file) => {
    const parts = file.replace("-diff.png", "").split("--");
    return {
      testName: parts[0],
      browser: parts[1] ?? "unknown",
      baselinePath: join(diffDir, file.replace("-diff.png", "-expected.png")),
      actualPath: join(diffDir, file.replace("-diff.png", "-actual.png")),
      diffPath: join(diffDir, file),
      diffPercentage: parseDiffPercentage(join(diffDir, file)),
    };
  });
}

function parseDiffPercentage(diffPath: string): number {
  // diff 메타데이터에서 차이 비율 추출
  try {
    const meta = readFileSync(diffPath.replace(".png", ".json"), "utf-8");
    return (JSON.parse(meta) as { percentage: number }).percentage;
  } catch {
    return -1;
  }
}

function generateVisualDiffPrompt(diffs: VisualDiff[]): string {
  const significant = diffs.filter((d) => d.diffPercentage > 0.1);

  return `다음 시각적 회귀 테스트에서 유의미한 차이가 발견되었다.
각 항목에 대해 CSS 호환성 문제인지, 레이아웃 변경인지 분류하고 수정 방안을 제시해줘.

${significant.map((d) => `- [${d.browser}] ${d.testName}: 차이 ${d.diffPercentage.toFixed(2)}%`).join("\n")}`;
}

export { collectVisualDiffs, generateVisualDiffPrompt };
export type { VisualDiff };
```

## 5. Baseline 2026 기준 기능 분류

### 5.1 Baseline Widely Available (안전 사용)

모든 주요 브라우저에서 30개월 이상 지원된 기능이다. Polyfill 없이 사용 가능하다.

| 기능 | Baseline 연도 | 비고 |
|------|-------------|------|
| CSS Grid | 2020 | `display: grid`, `grid-template` |
| CSS Custom Properties | 2020 | `--var`, `var()` |
| ES Modules | 2020 | `import` / `export` |
| IntersectionObserver | 2020 | 무한 스크롤, Lazy Load |
| CSS `aspect-ratio` | 2022 | 비율 유지 레이아웃 |
| `structuredClone()` | 2022 | 깊은 복사 |
| CSS Container Queries | 2023 | `@container`, `container-type` |
| CSS `:has()` Selector | 2024 | 부모/형제 선택자 |
| View Transitions API | 2024 | SPA 페이지 전환 애니메이션 |

### 5.2 Baseline Newly Available (주의 사용)

주요 브라우저 모두 지원하지만 30개월 미만이다. Feature Detection 필수.

| 기능 | Baseline 연도 | 비고 |
|------|-------------|------|
| Import Maps | 2023 | 번들러 없는 모듈 매핑 |
| CSS Subgrid | 2023 | 중첩 그리드 정렬 |
| CSS Nesting | 2024 | SCSS 없이 네이티브 중첩 |
| Cascade Layers | 2024 | `@layer` 기반 스타일 우선순위 |
| Popover API | 2025 | 네이티브 팝오버 |

### 5.3 Feature Detection 패턴

```typescript
// utils/baseline-check.ts
const BASELINE_CHECKS = {
  containerQueries: () => CSS.supports("container-type", "inline-size"),
  viewTransitions: () => "startViewTransition" in document,
  importMaps: () => HTMLScriptElement.supports?.("importmap") ?? false,
  cssNesting: () => CSS.supports("selector(&)"),
  cascadeLayers: () => CSS.supports("@layer base"),
  subgrid: () => CSS.supports("grid-template-rows", "subgrid"),
  hasSelector: () => CSS.supports("selector(:has(*))"),
  popover: () => "popover" in HTMLElement.prototype,
} as const;

type BaselineFeature = keyof typeof BASELINE_CHECKS;

function checkBaseline(features: BaselineFeature[]): Record<BaselineFeature, boolean> {
  const result = {} as Record<BaselineFeature, boolean>;
  for (const feature of features) {
    try {
      result[feature] = BASELINE_CHECKS[feature]();
    } catch {
      result[feature] = false;
    }
  }
  return result;
}

export { checkBaseline, BASELINE_CHECKS };
export type { BaselineFeature };
```

## 7. Polyfill 전략 2026

### 7.1 core-js v4 + 조건부 로딩

```typescript
// polyfills/index.ts
// core-js v4는 모듈별 import를 지원하여 번들 크기를 최소화한다

async function loadPolyfills(): Promise<void> {
  const polyfills: Promise<unknown>[] = [];

  // Object.groupBy (Baseline 2024)
  if (typeof Object.groupBy !== "function") {
    polyfills.push(import("core-js/actual/object/group-by"));
  }

  // Promise.withResolvers (Baseline 2024)
  if (typeof Promise.withResolvers !== "function") {
    polyfills.push(import("core-js/actual/promise/with-resolvers"));
  }

  // Set methods (Baseline 2025)
  if (typeof Set.prototype.union !== "function") {
    polyfills.push(import("core-js/actual/set"));
  }

  // Iterator helpers (Baseline 2025)
  if (typeof Iterator.prototype?.map !== "function") {
    polyfills.push(import("core-js/actual/iterator"));
  }

  if (polyfills.length > 0) {
    await Promise.all(polyfills);
    console.info(`[Polyfill] ${polyfills.length}개 polyfill 로드 완료`);
  }
}

export { loadPolyfills };
```

### 7.2 서비스 기반 Polyfill (polyfill.io 대안)

```typescript
// polyfills/service.ts
// polyfill.io 보안 이슈(2024) 이후, 자체 호스팅 또는 cdnjs 사용 권장

interface PolyfillConfig {
  features: string[];
  flags?: ("gated" | "always")[];
  callback?: string;
}

function buildPolyfillUrl(config: PolyfillConfig): string {
  const baseUrl = "https://cdnjs.cloudflare.com/polyfill/v3/polyfill.min.js";
  const features = config.features.join(",");
  const flags = config.flags?.join(",") ?? "gated";

  const params = new URLSearchParams({
    features,
    flags,
  });

  if (config.callback) {
    params.set("callback", config.callback);
  }

  return `${baseUrl}?${params.toString()}`;
}

function injectPolyfillScript(config: PolyfillConfig): void {
  const script = document.createElement("script");
  script.src = buildPolyfillUrl(config);
  script.crossOrigin = "anonymous";
  script.async = true;
  document.head.appendChild(script);
}

export { buildPolyfillUrl, injectPolyfillScript };
export type { PolyfillConfig };
```

```typescript
// e2e/compatibility.spec.ts
import { test, expect } from "@playwright/test";

test.describe("브라우저 호환성 검증", () => {
  test("CSS Container Queries 동작 확인", async ({ page }) => {
    await page.goto("/components/card-grid");

    // 넓은 뷰포트: 그리드 레이아웃
    await page.setViewportSize({ width: 1200, height: 800 });
    const cardBody = page.locator(".card-body").first();
    await expect(cardBody).toHaveCSS("display", "grid");

    // 좁은 컨테이너: 플렉스 레이아웃
    await page.setViewportSize({ width: 400, height: 800 });
    await expect(cardBody).toHaveCSS("display", "flex");
  });

  test("View Transitions API 동작 확인", async ({ page }) => {
    await page.goto("/");

    const hasViewTransitions = await page.evaluate(
      () => "startViewTransition" in document,
    );

    if (hasViewTransitions) {
      await page.click('a[href="/about"]');
      // 전환 애니메이션 완료 대기
      await page.waitForFunction(
        () => !document.querySelector("::view-transition"),
        null,
        { timeout: 3000 },
      );
    } else {
      // Fallback: 일반 네비게이션
      await page.click('a[href="/about"]');
    }

    await expect(page).toHaveURL("/about");
  });

  test("CSS :has() 선택자 동작 확인", async ({ page }) => {
    await page.goto("/components/form");

    // invalid 입력 시 부모 스타일 변경 확인
    const emailInput = page.locator('input[type="email"]');
    await emailInput.fill("invalid-email");
    await emailInput.blur();

    const formGroup = page.locator(".form-group").filter({ has: emailInput });
    await expect(formGroup).toHaveCSS("border-color", "rgb(239, 68, 68)");
  });
});
```
