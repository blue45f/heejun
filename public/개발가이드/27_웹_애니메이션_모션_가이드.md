```typescript
function convertToGPUAccelerated(
  keyframes: Record<string, string | number>
): Record<string, string | number> {
  const layoutProperties = ['width', 'height', 'top', 'left', 'right', 'bottom', 'margin', 'padding'];
  const converted: Record<string, string | number> = {};

    } else {
      converted[prop] = value;
    }
  }

  return converted;
```

```typescript
// AI 기반 애니메이션 성능 프로파일러
interface AnimationPerformanceReport {
  property: string;
  layer: 'composite' | 'paint' | 'layout';
  gpuAccelerated: boolean;
  recommendation: string;
}

function analyzeAnimationPerformance(
  element: HTMLElement
): AnimationPerformanceReport[] {
  const animations = element.getAnimations();
  const reports: AnimationPerformanceReport[] = [];

  const compositeOnly = new Set(['transform', 'opacity', 'filter']);
  const paintOnly = new Set(['background-color', 'color', 'box-shadow', 'border-color']);

  for (const animation of animations) {
    if (animation instanceof CSSAnimation || animation instanceof CSSTransition) {
      const effect = animation.effect as KeyframeEffect;
      const keyframes = effect.getKeyframes();

      for (const keyframe of keyframes) {
        for (const prop of Object.keys(keyframe)) {
          if (['offset', 'easing', 'composite'].includes(prop)) continue;

          const layer = compositeOnly.has(prop)
            ? 'composite'
            : paintOnly.has(prop)
              ? 'paint'
              : 'layout';

          reports.push({
            property: prop,
            layer,
            gpuAccelerated: layer === 'composite',
            recommendation: layer === 'layout'
              ? `"${prop}"은 레이아웃 재계산을 유발합니다. transform으로 대체하세요.`
              : layer === 'paint'
                ? `"${prop}"은 페인트를 유발합니다. 가능하면 opacity/filter로 대체하세요.`
                : '최적화된 속성입니다.',
          });
        }
      }
    }
  }

  return reports;
}
```

```typescript
  },
```

```typescript
    fast: 0.03,
    normal: 0.05,
    slow: 0.1,
  },
} as const;

// 모션 유틸리티 훅
function useMotion(
  type: keyof typeof motionSystem.enter,
  options?: { delay?: number; duration?: number }
) {
  const shouldReduceMotion = useReducedMotion();
  const pattern = motionSystem.enter[type];
  const transition = motionSystem.transitions.element;

    return {
      initial: {},
      animate: {},
      transition: { duration: 0 },
    };

  return {
    ...pattern,
    transition: {
      ...transition,
      ...options,
    },
  };
```

### 1.5 AI 접근성 모션 감사

```typescript
// AI 프롬프트: "이 페이지의 모든 애니메이션이 접근성 기준을 충족하는지 검사해줘"

interface MotionAccessibilityIssue {
  element: string;
  issue: 'no-reduced-motion' | 'excessive-duration' | 'flash-risk' | 'auto-play';
  severity: 'error' | 'warning';
  fix: string;
}

function auditMotionAccessibility(
  root: HTMLElement
): MotionAccessibilityIssue[] {
  const issues: MotionAccessibilityIssue[] = [];
  const allAnimated = root.querySelectorAll('[style*="animation"], [style*="transition"]');

  allAnimated.forEach((el) => {
    const styles = getComputedStyle(el);

    // prefers-reduced-motion 미디어 쿼리 존재 여부 확인
    const animations = el.getAnimations();
    if (animations.length > 0) {
      // 3초 이상 지속되는 애니메이션 경고
      for (const anim of animations) {
        const effect = anim.effect as KeyframeEffect;
        const timing = effect.getComputedTiming();
        if (timing.duration && Number(timing.duration) > 3000) {
          issues.push({
            element: el.tagName + (el.id ? `#${el.id}` : ''),
            issue: 'excessive-duration',
            severity: 'warning',
            fix: '애니메이션이 3초를 초과합니다. 사용자 주의를 과도하게 끌 수 있습니다.',
          });
        }
      }
    }

    // 자동 재생 애니메이션 확인
    if (styles.animationIterationCount === 'infinite') {
      issues.push({
        element: el.tagName + (el.id ? `#${el.id}` : ''),
        issue: 'auto-play',
        severity: 'error',
        fix: '무한 반복 애니메이션은 pause 버튼을 제공하거나 prefers-reduced-motion 시 정지해야 합니다.',
      });
    }
  });

  return issues;
}
```

## 2. 애니메이션 기술 비교

### CSS Animations vs Web Animations API vs Framer Motion vs GSAP

| 기준 | CSS Animations | Web Animations API | Framer Motion | GSAP |
|------|---------------|-------------------|--------------|------|
| **번들 크기** | 0 KB | 0 KB | ~30 KB | ~25 KB |
| **학습 곡선** | 낮음 | 중간 | 중간 | 높음 |
| **React 통합** | 제한적 | 수동 | 네이티브 | 플러그인 |
| **제어 수준** | 낮음 | 높음 | 높음 | 매우 높음 |
| **타임라인** | 제한적 | 지원 | 제한적 | 강력 |
| **스크롤 연동** | CSS scroll-driven | 직접 구현 | useScroll | ScrollTrigger |
| **Exit 애니메이션** | 불가 | 수동 | AnimatePresence | 수동 |
| **레이아웃 애니메이션** | 불가 | 불가 | layout prop | Flip 플러그인 |
| **성능** | 우수 | 우수 | 좋음 | 좋음 |
| **적합 사례** | 단순 전환 | 세밀한 제어 | React UI | 복잡한 시퀀스 |


### 선택 기준

```typescript
// 프로젝트별 기술 선택 가이드
type AnimationNeed =
  | 'simple-transition'     // → CSS Animations
  | 'framework-agnostic'    // → Web Animations API
  | 'react-ui'              // → Framer Motion
  | 'complex-sequence'      // → GSAP
  | 'scroll-driven'         // → CSS scroll-driven + Intersection Observer
  | 'layout-animation'      // → Framer Motion (layout prop)
  | 'svg-morphing'          // → GSAP MorphSVG
  | 'lottie-illustration';  // → Lottie

function selectAnimationTech(needs: AnimationNeed[]): string {
  if (needs.includes('complex-sequence') || needs.includes('svg-morphing')) {
    return 'GSAP - 복잡한 타임라인/SVG에 최적';
  }
  if (needs.includes('react-ui') || needs.includes('layout-animation')) {
    return 'Framer Motion - React UI 애니메이션에 최적';
  }
  if (needs.includes('framework-agnostic')) {
    return 'Web Animations API - 프레임워크 무관 제어';
  }
  return 'CSS Animations - 단순하고 성능 우수';
}
```

```typescript
// View Transitions API: 브라우저 네이티브 페이지 전환

// 기본 사용법
async function navigateWithTransition(
  updateCallback: () => void | Promise<void>
): Promise<void> {
  // View Transitions API 지원 확인
  if (!document.startViewTransition) {
    await updateCallback();
    return;
  }

  const transition = document.startViewTransition(async () => {

  });

}

        navigate(to);
      });
    } else {
      navigate(to);
    }
  };
}
```

### 4.2 Intersection Observer 패턴

```typescript
// 범용 스크롤 애니메이션 훅
import { useEffect, useRef, useState, useCallback } from 'react';

interface UseScrollAnimationOptions {
  threshold?: number;
  rootMargin?: string;
  triggerOnce?: boolean;
}

function useScrollAnimation<T extends HTMLElement>(
  options: UseScrollAnimationOptions = {}
) {
  const { threshold = 0.2, rootMargin = '0px', triggerOnce = true } = options;
  const ref = useRef<T>(null);
  const [isInView, setIsInView] = useState(false);

  useEffect(() => {
    const element = ref.current;
    if (!element) return;

    const observer = new IntersectionObserver(
      ([entry]) => {
        if (entry.isIntersecting) {
          setIsInView(true);
          if (triggerOnce) {
            observer.unobserve(element);
          }
        } else if (!triggerOnce) {
          setIsInView(false);
        }
      },
      { threshold, rootMargin }
    );

    observer.observe(element);
    return () => observer.disconnect();
  }, [threshold, rootMargin, triggerOnce]);

  return { ref, isInView };
}

// 사용 예시
function RevealSection({ children }: { children: React.ReactNode }) {
  const { ref, isInView } = useScrollAnimation<HTMLDivElement>({
    threshold: 0.2,
    triggerOnce: true,
  });

  return (
    <div
      ref={ref}
      style={{
        opacity: isInView ? 1 : 0,
        transform: isInView ? 'translateY(0)' : 'translateY(30px)',
        transition: 'opacity 0.6s ease-out, transform 0.6s ease-out',
      }}
    >
      {children}
    </div>
  );
}
```

```typescript
function useWillChange<T extends HTMLElement>() {
  const ref = useRef<T>(null);

  const activate = useCallback((properties: string) => {
    if (ref.current) {
```

```typescript
// 커스텀 애니메이션 루프 (60fps 보장)
class AnimationLoop {
  private animationId: number | null = null;
  private lastTime = 0;
  private readonly targetFPS = 60;
  private readonly frameInterval = 1000 / 60;

  start(callback: (deltaTime: number) => void): void {
    const tick = (currentTime: number) => {
      this.animationId = requestAnimationFrame(tick);

      const deltaTime = currentTime - this.lastTime;

      // 프레임 드랍 시 스킵하지 않고 보간
      if (deltaTime >= this.frameInterval) {
        this.lastTime = currentTime - (deltaTime % this.frameInterval);
        callback(deltaTime / 1000); // 초 단위로 변환
      }
    };

    this.animationId = requestAnimationFrame(tick);
  }

  stop(): void {
    if (this.animationId !== null) {
      cancelAnimationFrame(this.animationId);
      this.animationId = null;
    }
  }
}

// React 훅으로 래핑
function useAnimationFrame(callback: (deltaTime: number) => void, active = true) {
  const callbackRef = useRef(callback);
  callbackRef.current = callback;

  useEffect(() => {
    if (!active) return;

    let animationId: number;
    let lastTime = performance.now();

    const tick = (currentTime: number) => {
      const deltaTime = (currentTime - lastTime) / 1000;
      lastTime = currentTime;
      callbackRef.current(deltaTime);
      animationId = requestAnimationFrame(tick);
    };

    animationId = requestAnimationFrame(tick);
    return () => cancelAnimationFrame(animationId);
  }, [active]);
}
```

```typescript
// 프레임 드랍 감지 및 자동 품질 조정
class AdaptiveAnimationQuality {
  private frameCount = 0;
  private lastCheck = performance.now();
  private currentQuality: 'high' | 'medium' | 'low' = 'high';

  checkPerformance(): 'high' | 'medium' | 'low' {
    this.frameCount++;
    const now = performance.now();
    const elapsed = now - this.lastCheck;

    if (elapsed >= 1000) {
      const fps = (this.frameCount / elapsed) * 1000;
      this.frameCount = 0;
      this.lastCheck = now;

      if (fps < 30) {
        this.currentQuality = 'low';
      } else if (fps < 50) {
        this.currentQuality = 'medium';
      } else {
        this.currentQuality = 'high';
      }
    }

    return this.currentQuality;
  }
}
```

```typescript
function useReducedMotionPreference(): boolean {
  const [prefersReduced, setPrefersReduced] = useState(() => {
    if (typeof window === 'undefined') return false;
    return window.matchMedia('(prefers-reduced-motion: reduce)').matches;
  });

  useEffect(() => {
    const mediaQuery = window.matchMedia('(prefers-reduced-motion: reduce)');

    const handler = (event: MediaQueryListEvent) => {
      setPrefersReduced(event.matches);
    };

    mediaQuery.addEventListener('change', handler);
    return () => mediaQuery.removeEventListener('change', handler);
  }, []);

  return prefersReduced;
}
```

```typescript
function InteractiveLottie({
  animationData,
  trigger = 'hover',
}: {
  animationData: Record<string, unknown>;
  trigger?: 'hover' | 'click' | 'scroll';
}) {
  const lottieRef = useRef<LottieRefCurrentProps>(null);
  const containerRef = useRef<HTMLDivElement>(null);

  const play = () => lottieRef.current?.play();
  const stop = () => {
    lottieRef.current?.stop();

      ([entry]) => {
        if (entry.isIntersecting) {
          play();
        } else {
          stop();
        }

      { threshold: 0.5 }
    );

    observer.observe(containerRef.current);
    return () => observer.disconnect();
  }, [trigger]);

  return (
    <div
      ref={containerRef}
      onMouseEnter={trigger === 'hover' ? play : undefined}
      onMouseLeave={trigger === 'hover' ? stop : undefined}
      onClick={trigger === 'click' ? play : undefined}
    >
      <Lottie
        lottieRef={lottieRef}
        animationData={animationData}
        loop={false}
        autoplay={trigger === 'scroll'}
      />
    </div>
  );
}

// Lottie 최적화 체크리스트
// 1. JSON 파일 크기: 50KB 이하 권장 (gzip 후)
// 2. 복잡한 표현식/이펙트 제거 (After Effects → Bodymovin 내보내기 시)
// 3. 사용하지 않는 에셋 제거
// 4. lottie-light (SVG 렌더러만) 사용 고려
// 5. 뷰포트 밖 애니메이션은 pause 처리
// 6. 모바일에서는 Canvas 렌더러 고려 (많은 애니메이션 동시 재생 시)
```
