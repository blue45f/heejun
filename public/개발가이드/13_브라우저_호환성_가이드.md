## 2. Web Platform Baseline 2026

### 왜 중요한가

주니어 개발자가 가장 많이 겪는 실수는 **"내 Chrome에서 되니까 괜찮겠지"**라는 착각이다. Web Platform Baseline은 **모든 주요 브라우저에서 안정적으로 사용 가능한 기능의 기준선**을 정의한다. Baseline을 기준으로 코드를 작성하면 "이 기능을 써도 될까?"에 대한 객관적 판단이 가능하다. Widely Available 기능은 안심하고 사용하고, Newly Available 기능은 Feature Detection과 함께 사용하며, Limited 기능은 반드시 Fallback을 준비하는 것이 원칙이다.

### 2.5 Feature Detection 유틸

```typescript
// utils/baseline-detect.ts

const BASELINE_CHECKS = {
  containerQueries: () => CSS.supports("container-type", "inline-size"),
  viewTransitions: () => "startViewTransition" in document,
  importMaps: () => HTMLScriptElement.supports?.("importmap") ?? false,
  cssNesting: () => CSS.supports("selector(&)"),
  cascadeLayers: () => CSS.supports("@layer base"),
  subgrid: () => CSS.supports("grid-template-rows", "subgrid"),
  hasSelector: () => CSS.supports("selector(:has(*))"),
  popover: () => "popover" in HTMLElement.prototype,
  anchorPositioning: () => CSS.supports("position-anchor", "--any"),
  cssScope: () => CSS.supports("selector(:scope)") && CSS.supports("@scope (div) { }".length > 0 ? "@scope" : ""),
  setMethods: () => typeof Set.prototype.union === "function",
  objectGroupBy: () => typeof Object.groupBy === "function",
  promiseWithResolvers: () => typeof Promise.withResolvers === "function",
} as const;

type BaselineFeature = keyof typeof BASELINE_CHECKS;

interface BaselineReport {
  supported: BaselineFeature[];
  unsupported: BaselineFeature[];
  details: Record<BaselineFeature, boolean>;
}

function checkBaseline(features?: BaselineFeature[]): BaselineReport {
  const targets = features ?? (Object.keys(BASELINE_CHECKS) as BaselineFeature[]);
  const details = {} as Record<BaselineFeature, boolean>;

  for (const feature of targets) {
    try {
      details[feature] = BASELINE_CHECKS[feature]();
    } catch {
      details[feature] = false;
    }
  }

  return {
    supported: targets.filter((f) => details[f]),
    unsupported: targets.filter((f) => !details[f]),
    details,
  };
}

/** 런타임에 호출하여 콘솔에 지원 현황을 출력한다 */
function logBaselineReport(): void {
  const report = checkBaseline();
  console.group("[Baseline 2026] 브라우저 기능 지원 현황");
  for (const [feature, supported] of Object.entries(report.details)) {
    console.log(`${supported ? "  OK" : "  NG"} ${feature}`);
  }
  console.groupEnd();
}

export { checkBaseline, logBaselineReport, BASELINE_CHECKS };
export type { BaselineFeature, BaselineReport };
```

## 3. CSS 2026: Anchor Positioning / Popover / @scope

### 왜 중요한가

CSS 2026의 세 가지 핵심 신기능은 **지금까지 JavaScript로 해결하던 UI 패턴을 CSS만으로 처리**할 수 있게 한다. Anchor Positioning은 Floating UI 같은 포지셔닝 라이브러리를 대체하고, Popover API는 모달/드롭다운의 접근성과 Light Dismiss를 네이티브로 제공하며, @scope는 CSS Modules 없이 스타일 격리를 달성한다. 단, 아직 Newly Available 단계이므로 **반드시 `@supports`와 JS Fallback을 함께 작성**해야 한다. 아래 코드들은 모두 Fallback 포함 패턴을 보여준다.

### 3.1 CSS Anchor Positioning

요소를 다른 요소에 상대적으로 위치시키는 네이티브 CSS 기능이다. JavaScript 기반 포지셔닝 라이브러리(Floating UI 등)를 대체한다.

```css
/* 앵커 요소 등록 */
.trigger-button {
  anchor-name: --menu-anchor;
}

/* 앵커 대상 위치 지정 */
.dropdown-menu {
  position: fixed;
  position-anchor: --menu-anchor;

  /* 앵커 아래쪽, 왼쪽 정렬 */
  top: anchor(bottom);
  left: anchor(left);

  /* 뷰포트를 벗어날 때 반대쪽으로 전환 */
  position-try-fallbacks: flip-block, flip-inline;

  /* 크기 제한: 앵커와 뷰포트 가장자리 사이 */
  max-height: calc(anchor(--menu-anchor, bottom, 100vh) - 16px);
}

/* Fallback: Anchor Positioning 미지원 브라우저 */
@supports not (position-anchor: --any) {
  .dropdown-menu {
    position: absolute;
    top: 100%;
    left: 0;
    max-height: 300px;
  }
}
```

```typescript
// components/AnchoredPopup.ts -- Feature Detection 기반 분기

interface PopupPosition {
  top: number;
  left: number;
  maxHeight: number;
}

function supportsAnchorPositioning(): boolean {
  return CSS.supports("position-anchor", "--test");
}

/**
 * Anchor Positioning 미지원 시 JavaScript Fallback으로 위치를 계산한다.
 * 지원 브라우저에서는 CSS가 처리하므로 호출하지 않는다.
 */
function calculateFallbackPosition(
  anchor: HTMLElement,
  popup: HTMLElement,
): PopupPosition {
  const anchorRect = anchor.getBoundingClientRect();
  const viewportHeight = window.innerHeight;

  const spaceBelow = viewportHeight - anchorRect.bottom;
  const spaceAbove = anchorRect.top;

  const placeBelow = spaceBelow >= 200 || spaceBelow >= spaceAbove;

  return {
    top: placeBelow ? anchorRect.bottom + 4 : anchorRect.top - popup.offsetHeight - 4,
    left: anchorRect.left,
    maxHeight: (placeBelow ? spaceBelow : spaceAbove) - 16,
  };
}

function initAnchoredPopup(
  anchorSelector: string,
  popupSelector: string,
): void {
  if (supportsAnchorPositioning()) {
    // CSS가 모든 위치 계산을 처리한다. JS에서는 아무것도 하지 않는다.
    return;
  }

  const anchor = document.querySelector<HTMLElement>(anchorSelector);
  const popup = document.querySelector<HTMLElement>(popupSelector);
  if (!anchor || !popup) return;

  function updatePosition(): void {
    const pos = calculateFallbackPosition(anchor!, popup!);
    popup!.style.position = "fixed";
    popup!.style.top = `${pos.top}px`;
    popup!.style.left = `${pos.left}px`;
    popup!.style.maxHeight = `${pos.maxHeight}px`;
  }

  anchor.addEventListener("click", updatePosition);
  window.addEventListener("resize", updatePosition);
  window.addEventListener("scroll", updatePosition, { passive: true });
}

export { initAnchoredPopup, supportsAnchorPositioning, calculateFallbackPosition };
```

```typescript
 * Popover API 미지원 브라우저에서 <dialog>를 Fallback으로 사용한다.

function initPopoverFallback(): void {
  if (supportsPopover()) return;

  document.querySelectorAll<HTMLElement>("[popover]").forEach((el) => {
    el.style.display = "none";
    el.dataset.popoverFallback = "true";
  });

  document.querySelectorAll<HTMLElement>("[popovertarget]").forEach((trigger) => {
    const targetId = trigger.getAttribute("popovertarget");
    if (!targetId) return;

    const target = document.getElementById(targetId);
    if (!target) return;

    const action = trigger.getAttribute("popovertargetaction") ?? "toggle";

    trigger.addEventListener("click", () => {
      const isVisible = target.style.display !== "none";

      if (action === "hide" || (action === "toggle" && isVisible)) {
        target.style.display = "none";
      } else {
        target.style.display = "block";
        target.style.position = "fixed";
        target.style.zIndex = "9999";
        target.style.top = "50%";
        target.style.left = "50%";
        target.style.transform = "translate(-50%, -50%)";

    });
  });
}
```

## 4. 멀티 베타 + Preview별 BrowserStack 매트릭스

### 왜 중요한가

"배포 후 Safari에서 깨졌다"는 사고는 **테스트 환경이 Chrome 하나뿐**일 때 발생한다. 멀티 베타 전략은 **환경(Production/Beta/PR Preview)별로 테스트 범위를 차등 적용**하여, PR 단계에서 빠르게 핵심 브라우저를 검증하고, Production 배포 전에는 실기기까지 포함한 전체 매트릭스를 돌리는 것이다. BrowserStack은 실제 디바이스에서의 렌더링을 확인할 수 있어, 에뮬레이터만으로는 잡을 수 없는 이슈를 발견한다. [11. CI/CD 파이프라인 표준](./11_CICD_파이프라인_표준.md)과 연계하여 자동화한다.

```typescript
// scripts/run-compat-matrix.ts
import { execSync } from "node:child_process";

type EnvironmentTier = "production" | "beta" | "preview" | "preview-full";

interface MatrixConfig {
  environments: string[];
  tier: EnvironmentTier;
  testPattern: string;
}

interface MatrixResult {
  environment: string;
  browser: string;
  passed: number;
  failed: number;
  skipped: number;
  duration: number;
}

const BROWSER_MATRIX: Record<EnvironmentTier, string[]> = {
  production: [
    "chromium", "firefox", "webkit",
    "bs-chrome-win", "bs-firefox-win", "bs-safari-mac", "bs-edge-win",
    "bs-chrome-android", "bs-safari-ios", "bs-samsung-android",
  ],
  beta: ["chromium", "firefox", "webkit", "bs-chrome-win", "bs-firefox-win", "bs-safari-mac", "bs-edge-win"],
  preview: ["chromium", "webkit"],
  "preview-full": [
    "chromium", "firefox", "webkit",
    "bs-chrome-win", "bs-safari-mac", "bs-chrome-android", "bs-safari-ios",
  ],
};

function runTestMatrix(config: MatrixConfig): MatrixResult[] {
  const browsers = BROWSER_MATRIX[config.tier];
  const results: MatrixResult[] = [];

  for (const env of config.environments) {
    for (const browser of browsers) {
      const startTime = Date.now();
      const isBrowserStack = browser.startsWith("bs-");

      try {
        const envVars: Record<string, string> = {
          ...process.env as Record<string, string>,
          TEST_ENVIRONMENTS: env,
          PLAYWRIGHT_JSON_OUTPUT_FILE: `test-results/${env}-${browser}.json`,
        };

        if (isBrowserStack) {
          envVars.USE_BROWSERSTACK = "true";
        }

        const output = execSync(
          `npx playwright test --project=${browser} --grep="${config.testPattern}" --reporter=json`,
          { encoding: "utf-8", env: envVars, maxBuffer: 50 * 1024 * 1024 },
        );

        const report = JSON.parse(output);
        results.push({
          environment: env,
          browser,
          passed: report.stats?.expected ?? 0,
          failed: report.stats?.unexpected ?? 0,
          skipped: report.stats?.skipped ?? 0,
          duration: Date.now() - startTime,
        });
      } catch {
        results.push({
          environment: env,
          browser,
          passed: 0,
          failed: 1,
          skipped: 0,
          duration: Date.now() - startTime,
        });
      }
    }
  }

  return results;
}

function generateMarkdownReport(results: MatrixResult[]): string {
  const header = "| 환경 | 브라우저 | 성공 | 실패 | 스킵 | 시간 |";
  const separator = "|------|---------|------|------|------|------|";
  const rows = results.map(
    (r) =>
      `| ${r.environment} | ${r.browser} | ${r.passed} | ${r.failed} | ${r.skipped} | ${(r.duration / 1000).toFixed(1)}s |`,
  );

  const totalPassed = results.reduce((sum, r) => sum + r.passed, 0);
  const totalFailed = results.reduce((sum, r) => sum + r.failed, 0);
  const status = totalFailed === 0 ? "ALL PASSED" : `${totalFailed} FAILURES`;

  return [`## 크로스 브라우저 테스트 매트릭스 결과 (${status})`, "", header, separator, ...rows].join("\n");
}

// 실행
const tier = (process.env.MATRIX_TIER as EnvironmentTier) ?? "preview";
const config: MatrixConfig = {
  environments: (process.env.TEST_ENVIRONMENTS ?? "production").split(","),
  tier,
  testPattern: "compatibility|visual",
};

const results = runTestMatrix(config);
console.log(generateMarkdownReport(results));

export { runTestMatrix, generateMarkdownReport, BROWSER_MATRIX };
export type { MatrixConfig, MatrixResult, EnvironmentTier };
```

## 5. Import Maps

### 왜 중요한가

번들러(Webpack, Vite)는 `import React from "react"` 같은 bare specifier를 해석해 주지만, **브라우저는 기본적으로 bare specifier를 이해하지 못한다.** Import Maps는 이 매핑을 브라우저 네이티브로 제공하여, 번들러 없이도 ESM 모듈을 직접 로드할 수 있게 한다. 마이크로 프론트엔드에서 공유 의존성 버전을 통일하거나, CDN 기반 배포에서 번들 없이 모듈을 제공할 때 핵심적인 기술이다. [21. 마이크로 프론트엔드 가이드](./21_마이크로_프론트엔드_가이드.md)와 함께 참고하면 좋다.

번들러 없이 브라우저에서 직접 bare module specifier를 해석할 수 있다. 개발 환경, 마이크로 프론트엔드, CDN 기반 배포에서 유용하다.

## 6. UA Client Hints

### 왜 중요한가

`navigator.userAgent`는 **브라우저가 서로를 흉내 내며 복잡해진 레거시 문자열**이다. 파싱이 취약하고, 브라우저 업데이트마다 깨지며, 프라이버시 문제도 있다. Chrome은 이미 UA 문자열을 축소(UA Reduction)하고 있으며, 곧 다른 브라우저도 따를 것이다. UA Client Hints는 **구조화된 데이터로 브라우저 정보를 제공**하여, 안정적이고 프라이버시를 존중하는 방식으로 브라우저를 감지할 수 있다. [06. 웹 보안 심화 가이드](./06_웹_보안_심화_가이드.md)의 프라이버시 섹션도 함께 참고하자.

`navigator.userAgent` 문자열 파싱을 대체하는 구조화된 API이다. Privacy-preserving이며 서버/클라이언트 양쪽에서 사용할 수 있다.

### 6.1 클라이언트 측 사용

```typescript
// utils/client-hints.ts

interface ParsedClientHints {
  brands: Array<{ brand: string; version: string }>;
  mobile: boolean;
  platform: string;
  platformVersion: string;
  architecture: string;
  model: string;
  fullVersionList: Array<{ brand: string; version: string }>;
}

/**
 * UA Client Hints API를 사용하여 브라우저 정보를 가져온다.
 * 미지원 브라우저에서는 navigator.userAgent를 파싱하여 Fallback한다.
 */
async function getClientHints(): Promise<ParsedClientHints> {
  // Client Hints High-Entropy API 지원 확인
  if ("userAgentData" in navigator && navigator.userAgentData) {
    const uaData = navigator.userAgentData as NavigatorUAData;

    // Low-entropy: 항상 사용 가능
    const lowEntropy: Partial<ParsedClientHints> = {
      brands: uaData.brands.map((b) => ({ brand: b.brand, version: b.version })),
      mobile: uaData.mobile,
      platform: uaData.platform,
    };

    // High-entropy: 사용자 동의 필요 (Permissions-Policy)
    try {
      const highEntropy = await uaData.getHighEntropyValues([
        "platformVersion",
        "architecture",
        "model",
        "fullVersionList",
      ]);

      return {
        ...lowEntropy,
        platformVersion: highEntropy.platformVersion ?? "",
        architecture: highEntropy.architecture ?? "",
        model: highEntropy.model ?? "",
        fullVersionList: highEntropy.fullVersionList ?? [],
      } as ParsedClientHints;
    } catch {
      return {
        ...lowEntropy,
        platformVersion: "",
        architecture: "",
        model: "",
        fullVersionList: [],
      } as ParsedClientHints;
    }
  }

  // Fallback: UA 문자열 파싱 (Safari, Firefox 등)
  return parseUserAgentFallback(navigator.userAgent);
}

function parseUserAgentFallback(ua: string): ParsedClientHints {
  const mobile = /Mobi|Android/i.test(ua);
  const platform = /Mac/i.test(ua) ? "macOS"
    : /Win/i.test(ua) ? "Windows"
    : /Linux/i.test(ua) ? "Linux"
    : /Android/i.test(ua) ? "Android"
    : /iPhone|iPad/i.test(ua) ? "iOS"
    : "Unknown";

  const brands: Array<{ brand: string; version: string }> = [];
  const chromeMatch = ua.match(/Chrome\/([\d.]+)/);
  const firefoxMatch = ua.match(/Firefox\/([\d.]+)/);
  const safariMatch = ua.match(/Version\/([\d.]+).*Safari/);

  if (chromeMatch) brands.push({ brand: "Chromium", version: chromeMatch[1] });
  if (firefoxMatch) brands.push({ brand: "Firefox", version: firefoxMatch[1] });
  if (safariMatch) brands.push({ brand: "Safari", version: safariMatch[1] });

  return {
    brands,
    mobile,
    platform,
    platformVersion: "",
    architecture: "",
    model: "",
    fullVersionList: brands,
  };
}

/** UA Client Hints를 사용하여 Tier 1 브라우저인지 판별한다 */
async function isTier1Browser(): Promise<boolean> {
  const hints = await getClientHints();
  const tier1Brands = ["Chromium", "Google Chrome", "Firefox", "Safari", "Microsoft Edge"];
  return hints.brands.some((b) => tier1Brands.includes(b.brand));
}

// TypeScript 타입 보강
interface NavigatorUABrandVersion {
  brand: string;
  version: string;
}

interface NavigatorUAData {
  brands: NavigatorUABrandVersion[];
  mobile: boolean;
  platform: string;
  getHighEntropyValues(hints: string[]): Promise<Record<string, unknown>>;
}

export { getClientHints, isTier1Browser, parseUserAgentFallback };
export type { ParsedClientHints };
```

```typescript
  critical: ["Sec-CH-UA", "Sec-CH-UA-Mobile", "Sec-CH-UA-Platform"],
  accept: [
    "Sec-CH-UA",
    "Sec-CH-UA-Mobile",
    "Sec-CH-UA-Platform",
    "Sec-CH-UA-Platform-Version",
    "Sec-CH-UA-Full-Version-List",
    "Sec-CH-UA-Arch",
    "Sec-CH-UA-Model",
  ],
};

/**
 * Client Hints 요청 헤더를 응답에 설정하는 헬퍼이다.
 * Permissions-Policy와 함께 사용한다.
 */
function buildClientHintsHeaders(
  config: ClientHintsMiddlewareConfig = DEFAULT_CONFIG,
): Record<string, string> {
  return {
    "Accept-CH": config.accept.join(", "),
    "Critical-CH": config.critical.join(", "),
    "Permissions-Policy": config.accept
      .map((h) => `${h.toLowerCase().replace("sec-ch-", "ch-")}=(self)`)
      .join(", "),
  };
}

/**
 * 요청 헤더에서 Client Hints를 파싱한다.
 */
function parseClientHintsFromHeaders(headers: Record<string, string | undefined>): {
  brands: string;
  mobile: boolean;
  platform: string;
} {
  return {
    brands: headers["sec-ch-ua"] ?? "",
    mobile: headers["sec-ch-ua-mobile"] === "?1",
    platform: headers["sec-ch-ua-platform"]?.replace(/"/g, "") ?? "",
  };
}

export { buildClientHintsHeaders, parseClientHintsFromHeaders };
export type { ClientHintsMiddlewareConfig };
```

## 7. 환경별 Polyfill 전략

### 왜 중요한가

Polyfill은 **구형 브라우저에서 최신 API를 사용할 수 있게 해 주는 코드**이다. 하지만 무분별한 Polyfill은 번들 크기를 수십 KB 이상 증가시킨다. 핵심 원칙은 세 가지다: (1) **Feature Detection으로 필요할 때만 로드**한다 -- 이미 지원하는 브라우저에서는 0바이트, (2) **동적 import()로 번들에서 분리**한다 -- 메인 번들이 커지지 않는다, (3) **Differential Serving으로 모던/레거시 빌드를 나눈다** -- 95% 이상의 모던 브라우저 사용자는 Polyfill 없이 가벼운 빌드를 받는다. [08. 성능 최적화 가이드](./08_성능_최적화_가이드.md)의 번들 최적화 섹션과 직결되는 내용이다.

### 7.1 전략 개요

| 전략 | 적용 환경 | 방식 | 장점 | 단점 |
|------|----------|------|------|------|
| **조건부 동적 import** | SPA, CSR | Feature Detection + `import()` | 필요한 Polyfill만 로드, 번들 분리 | 초기 로드 시 딜레이 |
| **Differential Serving** | SSR, MPA | `<script type="module">` + `<script nomodule>` | 모던 브라우저는 Polyfill 0개 | 빌드 파이프라인 복잡 |
| **CDN Polyfill Service** | 빠른 프로토타이핑 | cdnjs(Cloudflare) / polyfill-fastly.io | 설정 간단, 무료 | 외부 의존성, 신뢰 가능한 호스트만 사용 |
| **core-js 선택적 임포트** | 모든 환경 | 필요한 모듈만 `import` | 정밀 제어, 트리셰이킹 | 수동 관리 필요 |

> **중요 -- polyfill.io 종료 후 대안 (2024.06 이후 필수)**
>
> 2024년 2월 polyfill.io 도메인이 Funnull(중국 CDN)에 매각된 뒤, 6월에 악성 스크립트가 주입되는 공급망 공격이 발생하여 100,000개 이상의 사이트가 영향을 받았다. **`cdn.polyfill.io` / `polyfill.io` 도메인은 절대 사용하지 말 것.** 2026년 5월 기준 권장 대안은 다음과 같다.
>
> | 대안 | URL 패턴 | 운영 주체 | 특징 |
> |------|---------|----------|------|
> | **Cloudflare cdnjs** | `https://cdnjs.cloudflare.com/polyfill/v3/polyfill.min.js` | Cloudflare | 기존 polyfill.io API와 100% 호환, drop-in 교체 |
> | **Fastly Polyfill** | `https://polyfill-fastly.io/v3/polyfill.min.js` | Fastly | 원 polyfill.io 개발자가 운영, 무료 |
> | **셀프 호스팅 + core-js** | `import "core-js/actual/..."` | 자체 빌드 | 외부 의존 0, 가장 안전 (권장) |
>
> Cloudflare로 프록시되는 사이트는 자동 URL Rewriting이 적용되지만, 그에 의존하지 말고 **소스 코드에서 직접 URL을 교체**해야 한다. 가장 안전한 옵션은 외부 CDN을 끊고 `core-js`를 번들에 포함시키는 것이다 (섹션 7.2, 7.3 참고).

### 7.2 조건부 동적 import 패턴

```typescript
// polyfills/index.ts

interface PolyfillEntry {
  name: string;
  test: () => boolean;
  load: () => Promise<unknown>;
  /** gzip 기준 예상 크기 (KB) */
  estimatedSize: number;
}

const POLYFILL_REGISTRY: PolyfillEntry[] = [
  {
    name: "Object.groupBy",
    test: () => typeof Object.groupBy === "function",
    load: () => import("core-js/actual/object/group-by"),
    estimatedSize: 0.5,
  },
  {
    name: "Promise.withResolvers",
    test: () => typeof Promise.withResolvers === "function",
    load: () => import("core-js/actual/promise/with-resolvers"),
    estimatedSize: 0.3,
  },
  {
    name: "Set methods",
    test: () => typeof Set.prototype.union === "function",
    load: () => import("core-js/actual/set"),
    estimatedSize: 2.1,
  },
  {
    name: "Map.groupBy",
    test: () => typeof Map.groupBy === "function",
    load: () => import("core-js/actual/map/group-by"),
    estimatedSize: 0.5,
  },
  {
    name: "Iterator helpers",
    test: () => typeof Iterator.prototype?.map === "function",
    load: () => import("core-js/actual/iterator"),
    estimatedSize: 3.2,
  },
  {
    name: "Array.fromAsync",
    test: () => typeof Array.fromAsync === "function",
    load: () => import("core-js/actual/array/from-async"),
    estimatedSize: 0.4,
  },
];

async function loadPolyfills(): Promise<{ loaded: string[]; skipped: string[] }> {
  const needed = POLYFILL_REGISTRY.filter((p) => {
    try {
      return !p.test();
    } catch {
      return true;
    }
  });

  if (needed.length === 0) {
    return { loaded: [], skipped: POLYFILL_REGISTRY.map((p) => p.name) };
  }

  const totalSize = needed.reduce((sum, p) => sum + p.estimatedSize, 0);
  console.info(
    `[Polyfill] ${needed.length}개 로드 시작 (예상 ${totalSize.toFixed(1)}KB gzip)`,
  );

  await Promise.all(needed.map((p) => p.load()));

  const loadedNames = needed.map((p) => p.name);
  console.info(`[Polyfill] 로드 완료: ${loadedNames.join(", ")}`);

  return {
    loaded: loadedNames,
    skipped: POLYFILL_REGISTRY.filter((p) => !needed.includes(p)).map((p) => p.name),
  };
}

export { loadPolyfills, POLYFILL_REGISTRY };
export type { PolyfillEntry };
```

```typescript
async function loadCssPolyfills(): Promise<void> {
  const polyfills: Promise<unknown>[] = [];

  // Container Queries Polyfill
  if (!CSS.supports("container-type", "inline-size")) {
    polyfills.push(
      import("container-query-polyfill").then((mod) => mod.default?.()),
    );
  }

  if (!CSS.supports("selector(:has(*))")) {

  }

  if (polyfills.length > 0) {
    await Promise.all(polyfills);
  }
}
```

## 9. AI 기반 호환성 이슈 자동 탐지

### 왜 중요한가

호환성 이슈는 **코드를 작성하는 시점에 발견하는 것이 가장 비용이 적다.** `eslint-plugin-compat`은 browserslist 설정을 기반으로 미지원 API 사용을 정적 분석 단계에서 잡아낸다. 여기에 AI를 결합하면, 발견된 이슈에 대한 Polyfill 추천, 대체 구현 코드, 사용자 영향 범위 추정까지 자동화할 수 있다. [16. AI 협업 코드리뷰 가이드](./16_AI_협업_코드리뷰_가이드.md)와 연계하여 코드리뷰 시에도 호환성 검증을 자동화하자.

### 9.1 eslint-plugin-compat + AI 분석 파이프라인

```typescript
// scripts/ai-compat-check.ts
import { execSync } from "node:child_process";
import { writeFileSync } from "node:fs";

interface CompatIssue {
  file: string;
  line: number;
  rule: string;
  message: string;
  api: string;
}

function runEslintCompat(): CompatIssue[] {
  const result = execSync(
    'npx eslint --format json --rule \'{"compat/compat": "error"}\' src/',
    { encoding: "utf-8", maxBuffer: 10 * 1024 * 1024 },
  );

  const parsed = JSON.parse(result) as Array<{
    filePath: string;
    messages: Array<{ line: number; ruleId: string; message: string }>;
  }>;

  return parsed.flatMap((file) =>
    file.messages
      .filter((msg) => msg.ruleId === "compat/compat")
      .map((msg) => ({
        file: file.filePath,
        line: msg.line,
        rule: msg.ruleId,
        message: msg.message,
        api: msg.message.match(/'([^']+)'/)?.[1] ?? "unknown",
      })),
  );
}

function generateAiPrompt(issues: CompatIssue[]): string {
  const grouped = Object.groupBy(issues, (i) => i.api);
  const summary = Object.entries(grouped)
    .map(
      ([api, items]) =>
        `- ${api}: ${items!.length}건 (${items!.map((i) => `${i.file}:${i.line}`).join(", ")})`,
    )
    .join("\n");

  return [
    "다음 호환성 이슈에 대해 각각:",
    "(1) Polyfill이 있으면 npm 패키지명과 설치 코드",
    "(2) Polyfill이 없으면 대체 구현 (TypeScript)",
    "(3) Feature Detection 코드",
    "(4) 사용자 영향 범위 추정",
    "",
    `--- 이슈 목록 (${issues.length}건) ---`,
    summary,
  ].join("\n");
}

const issues = runEslintCompat();
const prompt = generateAiPrompt(issues);
writeFileSync("compat-ai-prompt.txt", prompt, "utf-8");
```

```typescript
// scripts/analyze-test-results.ts
import { readFileSync } from "node:fs";

interface TestResult {
  title: string;
  status: "passed" | "failed" | "skipped";
  browser: string;
  duration: number;
  error?: string;
  environment?: string;
}

function parsePlaywrightReport(reportPath: string): TestResult[] {
  const raw = readFileSync(reportPath, "utf-8");
  const report = JSON.parse(raw) as {
    suites: Array<{
      specs: Array<{
        title: string;
        tests: Array<{
          projectName: string;
          status: string;
          duration: number;
          results: Array<{ error?: { message: string } }>;
        }>;
      }>;
    }>;
  };

  return report.suites.flatMap((suite) =>
    suite.specs.flatMap((spec) =>
      spec.tests.map((t) => ({
        title: spec.title,
        status: t.status as TestResult["status"],
        browser: t.projectName,
        duration: t.duration,
        error: t.results[0]?.error?.message,
      })),
    ),
  );
}

function buildAnalysisPrompt(results: TestResult[]): string {
  const failures = results.filter((r) => r.status === "failed");
  const byBrowser = Object.groupBy(failures, (f) => f.browser);

  return [
    "다음은 Playwright 크로스 브라우저 테스트 실패 결과이다.",
    "브라우저별 실패 원인을 분석하고, 호환성 문제 / 로직 버그 / 환경 차이로 분류해줘.",
    "각 문제에 대해 수정 코드를 TypeScript로 제시하고 우선순위(P0~P3)를 매겨줘.",
    "",
    `--- 실패 목록 (${failures.length}건) ---`,
    failures.map((f) => `[${f.browser}] ${f.title}: ${f.error ?? "unknown"}`).join("\n"),
    "",
    "--- 브라우저별 실패 수 ---",
    Object.entries(byBrowser).map(([b, fs]) => `${b}: ${fs!.length}건`).join("\n"),
  ].join("\n");
}

const results = parsePlaywrightReport("test-results/report.json");
console.log(buildAnalysisPrompt(results));
```

## 11. AI와 함께하는 브라우저 호환성 워크플로우

### 11.1 일상적인 AI 협업 패턴

브라우저 호환성은 AI가 **가장 큰 생산성 향상**을 가져다주는 영역 중 하나이다. 수백 개의 API-브라우저 조합을 사람이 외우는 것은 불가능하지만, AI는 caniuse 데이터를 기반으로 즉시 답변할 수 있다.

```typescript
// AI 협업 예시: 새로운 API를 사용하기 전 호환성 확인 워크플로우

/**
 * [1단계] 코드 작성 전 -- AI에게 호환성 확인 요청
 *
 * 프롬프트: "Object.groupBy()를 우리 browserslist 범위에서 사용해도 되는지 확인해줘.
 *           미지원 브라우저가 있으면 Feature Detection + Polyfill 코드를 작성해줘."
 */

/**
 * [2단계] AI가 생성한 Feature Detection + Polyfill 래퍼
 * -- AI 출력을 그대로 사용하지 말고, 반드시 코드 리뷰 후 적용한다
 */
function safeGroupBy<K extends PropertyKey, T>(
  items: Iterable<T>,
  keySelector: (item: T, index: number) => K,
): Partial<Record<K, T[]>> {
  // Feature Detection: 네이티브 지원 시 바로 사용
  if (typeof Object.groupBy === "function") {
    return Object.groupBy(items, keySelector);
  }

  // Fallback: 수동 구현 (core-js 동적 로드 전까지 사용)
  const result = {} as Record<K, T[]>;
  let index = 0;
  for (const item of items) {
    const key = keySelector(item, index++);
    (result[key] ??= []).push(item);
  }
  return result;
}

/**
 * [3단계] AI에게 테스트 코드 생성 요청
 *
 * 프롬프트: "위 safeGroupBy 함수에 대해 Vitest 테스트를 작성해줘.
 *           네이티브 Object.groupBy가 있을 때와 없을 때 모두 테스트해줘."
 */

/**
 * [4단계] eslint-plugin-compat이 CI에서 경고를 발생시키면,
 * AI 분석 스크립트(섹션 9)가 자동으로 수정 제안을 생성한다
 */

export { safeGroupBy };
```
