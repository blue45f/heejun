```typescript
// stores/signupFormStore.ts
import { create } from 'zustand';
import { persist } from 'zustand/middleware';

interface BasicInfo {
  name: string;
  email: string;
  phone: string;
}

interface AddressInfo {
  zipCode: string;
  address: string;
  addressDetail: string;
}

interface AgreementInfo {
  termsOfService: boolean;
  privacyPolicy: boolean;
  marketing: boolean;
}

interface SignupFormState {
  currentStep: 1 | 2 | 3 | 4;
  basicInfo: BasicInfo;
  addressInfo: AddressInfo;
  agreementInfo: AgreementInfo;
  isSubmitting: boolean;
  submitError: string | null;

  setStep: (step: SignupFormState['currentStep']) => void;
  updateBasicInfo: (info: Partial<BasicInfo>) => void;
  updateAddressInfo: (info: Partial<AddressInfo>) => void;
  updateAgreementInfo: (info: Partial<AgreementInfo>) => void;
  submit: () => Promise<void>;
  reset: () => void;
}

const initialState = {
  currentStep: 1 as const,
  basicInfo: { name: '', email: '', phone: '' },
  addressInfo: { zipCode: '', address: '', addressDetail: '' },
  agreementInfo: { termsOfService: false, privacyPolicy: false, marketing: false },
  isSubmitting: false,
  submitError: null,
};

export const useSignupFormStore = create<SignupFormState>()(
  persist(
    (set, get) => ({
      ...initialState,
      setStep: (step) => set({ currentStep: step }),
      updateBasicInfo: (info) =>
        set((s) => ({ basicInfo: { ...s.basicInfo, ...info } })),
      updateAddressInfo: (info) =>
        set((s) => ({ addressInfo: { ...s.addressInfo, ...info } })),
      updateAgreementInfo: (info) =>
        set((s) => ({ agreementInfo: { ...s.agreementInfo, ...info } })),
      submit: async () => {
        set({ isSubmitting: true, submitError: null });
        try {
          const { basicInfo, addressInfo, agreementInfo } = get();
          await api.signup({ ...basicInfo, ...addressInfo, ...agreementInfo });
          set({ currentStep: 4 });
        } catch (e) {
          set({ submitError: (e as Error).message });
        } finally {
          set({ isSubmitting: false });
        }
      },
      reset: () => set(initialState),
    }),
    {
      name: 'signup-form',
      partialize: (state) => ({
        basicInfo: state.basicInfo,
        addressInfo: state.addressInfo,
        agreementInfo: state.agreementInfo,
        currentStep: state.currentStep,
      }),
    }
  )
);
```

### 1.2 AI로 Redux에서 Zustand 자동 마이그레이션

Claude Code나 Cursor AI에 아래 프롬프트를 사용하면 Redux 코드를 Zustand로 변환할 수 있다.

```
프롬프트:
"이 Redux 슬라이스를 Zustand v5 스토어로 변환해줘.
- createAsyncThunk → 일반 async 함수
- 불변성 로직 단순화
- TypeScript 타입 유지
- immer 미들웨어는 사용하지 않고 전개 연산자 사용"
```

**변환 전 (Redux Toolkit):**

```typescript
// Redux Toolkit
import { createSlice, createAsyncThunk, PayloadAction } from '@reduxjs/toolkit';

interface Todo {
  id: string;
  text: string;
  completed: boolean;
}

interface TodoState {
  items: Todo[];
  loading: boolean;
  error: string | null;
  filter: 'all' | 'active' | 'completed';
}

export const fetchTodos = createAsyncThunk('todos/fetch', async () => {
  const response = await fetch('/api/todos');
  return response.json() as Promise<Todo[]>;
});

export const addTodo = createAsyncThunk('todos/add', async (text: string) => {
  const response = await fetch('/api/todos', {
    method: 'POST',
    body: JSON.stringify({ text }),
  });
  return response.json() as Promise<Todo>;
});

const todoSlice = createSlice({
  name: 'todos',
  initialState: {
    items: [],
    loading: false,
    error: null,
    filter: 'all',
  } as TodoState,
  reducers: {
    toggleTodo(state, action: PayloadAction<string>) {
      const todo = state.items.find((t) => t.id === action.payload);
      if (todo) todo.completed = !todo.completed;
    },
    setFilter(state, action: PayloadAction<TodoState['filter']>) {
      state.filter = action.payload;
    },
  },
  extraReducers: (builder) => {
    builder
      .addCase(fetchTodos.pending, (state) => { state.loading = true; })
      .addCase(fetchTodos.fulfilled, (state, action) => {
        state.loading = false;
        state.items = action.payload;
      })
      .addCase(fetchTodos.rejected, (state, action) => {
        state.loading = false;
        state.error = action.error.message ?? 'Failed';
      })
      .addCase(addTodo.fulfilled, (state, action) => {
        state.items.push(action.payload);
      });
  },
});
```

**변환 후 (Zustand v5):**

```typescript
// Zustand v5
import { create } from 'zustand';
import { devtools } from 'zustand/middleware';

interface Todo {
  id: string;
  text: string;
  completed: boolean;
}

interface TodoStore {
  items: Todo[];
  loading: boolean;
  error: string | null;
  filter: 'all' | 'active' | 'completed';

  fetchTodos: () => Promise<void>;
  addTodo: (text: string) => Promise<void>;
  toggleTodo: (id: string) => void;
  setFilter: (filter: TodoStore['filter']) => void;
}

export const useTodoStore = create<TodoStore>()(
  devtools(
    (set) => ({
      items: [],
      loading: false,
      error: null,
      filter: 'all',

      fetchTodos: async () => {
        set({ loading: true, error: null });
        try {
          const response = await fetch('/api/todos');
          const items: Todo[] = await response.json();
          set({ items, loading: false });
        } catch (e) {
          set({ error: (e as Error).message, loading: false });
        }
      },

      addTodo: async (text) => {
        try {
          const response = await fetch('/api/todos', {
            method: 'POST',
            body: JSON.stringify({ text }),
          });
          const newTodo: Todo = await response.json();
          set((s) => ({ items: [...s.items, newTodo] }));
        } catch (e) {
          set({ error: (e as Error).message });
        }
      },

      toggleTodo: (id) =>
        set((s) => ({
          items: s.items.map((t) =>
            t.id === id ? { ...t, completed: !t.completed } : t
          ),
        })),

      setFilter: (filter) => set({ filter }),
    }),
    { name: 'TodoStore' }
  )
);
```

**핵심 변화:**
- `createAsyncThunk` 보일러플레이트 제거 → 일반 `async` 함수
- `extraReducers` 빌더 패턴 제거 → `set()` 직접 호출
- Immer 프록시 의존 제거 → 명시적 전개 연산자
- 코드량 약 40% 감소

## 2. 2026 상태관리 패러다임

### 2.1 상태 유형 완전 분리

2026년 상태관리의 핵심 원칙은 **상태의 소유권(ownership)에 따른 완전한 분리**다. 하나의 전역 스토어에 모든 상태를 넣는 것은 안티패턴이다.

| 상태 유형 | 소유권 | 특성 | 권장 도구 |
|-----------|--------|------|-----------|
| **Server State** | 서버가 원본 | 비동기, 캐싱, stale 관리 | TanStack Query v5 |
| **Client State** | 클라이언트 원본 | 동기, 즉시 반영 | Zustand v5 / Jotai v2 |
| **URL State** | URL이 원본 | 공유/북마크 가능 | nuqs / TanStack Router |
| **Form State** | 폼 필드가 원본 | 유효성 검증 연동 | React Hook Form / Conform |
| **Derived State** | 다른 상태에서 계산 | 메모이제이션 대상 | useMemo / computed |

**원칙:** 각 상태는 반드시 하나의 소유권만 가진다. 서버 데이터를 전역 스토어에 복제하지 않는다.

```tsx
  success: boolean;

      <input name="password" type="password" required />
```

### 2.3 nuqs로 URL State 관리

URL 파라미터를 타입 안전하게 상태로 관리한다. 필터, 정렬, 페이지네이션, 검색어 등은 URL State가 정답이다.

```tsx
import { useQueryState, useQueryStates, parseAsInteger, parseAsStringEnum } from 'nuqs';

type SortField = 'name' | 'date' | 'price';
type SortOrder = 'asc' | 'desc';

function ProductList() {
  // 개별 파라미터
  const [search, setSearch] = useQueryState('q', { defaultValue: '' });
  const [page, setPage] = useQueryState('page', parseAsInteger.withDefault(1));

  // 여러 파라미터를 묶어서 관리
  const [sort, setSort] = useQueryStates({
    field: parseAsStringEnum<SortField>(['name', 'date', 'price']).withDefault('date'),
    order: parseAsStringEnum<SortOrder>(['asc', 'desc']).withDefault('desc'),
  });

  // URL: /products?q=keyboard&page=2&field=price&order=asc
  // 뒤로가기, 북마크, 링크 공유 모두 자연스럽게 동작

  const { data } = useQuery({
    queryKey: ['products', { search, page, ...sort }],
    queryFn: () => fetchProducts({ search, page, ...sort }),
  });

  return (
    <div>
      <input
        value={search}
        onChange={(e) => {
          setSearch(e.target.value);
          setPage(1); // 검색 변경 시 1페이지로 리셋
        }}
        placeholder="검색..."
      />
      <select
        value={sort.field}
        onChange={(e) => setSort({ field: e.target.value as SortField })}
      >
        <option value="name">이름</option>
        <option value="date">날짜</option>
        <option value="price">가격</option>
      </select>
    </div>
  );
}
```

### 3.2 Optimistic Updates

```typescript
import { useMutation, useQueryClient } from '@tanstack/react-query';

function useUpdateTodo() {
  const queryClient = useQueryClient();

  return useMutation({
    mutationFn: updateTodo,

    onMutate: async (updatedTodo) => {
      // 1. 진행 중인 쿼리 취소 (덮어쓰기 방지)
      await queryClient.cancelQueries({ queryKey: ['todos'] });

      // 2. 이전 값 스냅샷
      const previousTodos = queryClient.getQueryData<Todo[]>(['todos']);

      // 3. 캐시를 낙관적으로 업데이트
      queryClient.setQueryData<Todo[]>(['todos'], (old) =>
        old?.map((todo) =>
          todo.id === updatedTodo.id ? { ...todo, ...updatedTodo } : todo
        )
      );

      // 4. 롤백용 컨텍스트 반환
      return { previousTodos };
    },

    onError: (_err, _variables, context) => {
      // 실패 시 이전 값으로 롤백
      if (context?.previousTodos) {
        queryClient.setQueryData(['todos'], context.previousTodos);
      }
    },

    onSettled: () => {
      // 성공/실패 무관하게 쿼리 갱신
      queryClient.invalidateQueries({ queryKey: ['todos'] });
    },
  });
}
```

## 4. Zustand v5 최신 패턴

### 4.1 createWithEqualityFn

기본 `create`는 `Object.is`로 비교한다. 커스텀 비교 함수가 필요한 경우 사용한다.

```typescript
import { createWithEqualityFn } from 'zustand/traditional';
import { shallow } from 'zustand/shallow';

interface FilterState {
  filters: {
    category: string[];
    priceRange: [number, number];
    inStock: boolean;
  };
  setCategory: (categories: string[]) => void;
  setPriceRange: (range: [number, number]) => void;
  toggleInStock: () => void;
}

// shallow 비교로 불필요한 리렌더링 방지
const useFilterStore = createWithEqualityFn<FilterState>(
  (set) => ({
    filters: {
      category: [],
      priceRange: [0, 100000],
      inStock: false,
    },
    setCategory: (categories) =>
      set((s) => ({ filters: { ...s.filters, category: categories } })),
    setPriceRange: (range) =>
      set((s) => ({ filters: { ...s.filters, priceRange: range } })),
    toggleInStock: () =>
      set((s) => ({ filters: { ...s.filters, inStock: !s.filters.inStock } })),
  }),
  shallow // 얕은 비교 적용
);

// 사용 시 - 선택적 구독으로 세밀한 리렌더링 제어
function CategoryFilter() {
  const categories = useFilterStore((s) => s.filters.category);
  const setCategory = useFilterStore((s) => s.setCategory);
  // filters.priceRange가 변경되어도 이 컴포넌트는 리렌더링되지 않음
  return <MultiSelect value={categories} onChange={setCategory} />;
}
```

### 4.2 persist 미들웨어와 버전 마이그레이션

스토어 구조가 변경될 때 기존 영속화 데이터를 안전하게 마이그레이션한다.

```typescript
import { create } from 'zustand';
import { persist, createJSONStorage } from 'zustand/middleware';

interface SettingsState {
  theme: 'light' | 'dark' | 'system';
  language: string;
  fontSize: number;
  notifications: {
    email: boolean;
    push: boolean;
    sms: boolean;
  };
}

export const useSettingsStore = create<SettingsState>()(
  persist(
    (set) => ({
      theme: 'system',
      language: 'ko',
      fontSize: 14,
      notifications: { email: true, push: true, sms: false },
    }),
    {
      name: 'app-settings',
      storage: createJSONStorage(() => localStorage),
      version: 2, // 현재 스키마 버전

      // 버전별 마이그레이션 함수
      migrate: (persistedState: unknown, version: number) => {
        const state = persistedState as Record<string, unknown>;

        if (version === 0) {
          // v0 → v1: theme 필드 추가
          state.theme = 'system';
        }

        if (version <= 1) {
          // v1 → v2: notifications 객체 구조 변경
          state.notifications = {
            email: true,
            push: (state as any).pushEnabled ?? true,
            sms: false,
          };
          delete (state as any).pushEnabled;
        }

        return state as SettingsState;
      },

      // 특정 필드만 영속화
      partialize: (state) => ({
        theme: state.theme,
        language: state.language,
        fontSize: state.fontSize,
        notifications: state.notifications,
      }),
    }
  )
);
```

```typescript
import { atom, useAtom } from 'jotai';

  id: string;
  label: string;
  value: string;
  error: string | null;

  return (
```

## 6. Signal 기반 상태관리

### 6.1 @preact/signals-react

Signal은 값이 변경될 때 구독하는 컴포넌트만 정확히 리렌더링한다. Virtual DOM diffing 없이 DOM을 직접 업데이트하여 성능 이점이 있다.

```typescript
import { signal, computed, effect, batch } from '@preact/signals-react';
import { useSignals } from '@preact/signals-react/runtime';

// 스토어 정의 - 컴포넌트 외부에서 선언
const count = signal(0);
const doubleCount = computed(() => count.value * 2);

const todos = signal<Todo[]>([]);
const filter = signal<'all' | 'active' | 'completed'>('all');
const filteredTodos = computed(() => {
  switch (filter.value) {
    case 'active':
      return todos.value.filter((t) => !t.completed);
    case 'completed':
      return todos.value.filter((t) => t.completed);
    default:
      return todos.value;
  }
});

// 컴포넌트에서 사용
function TodoApp() {
  useSignals(); // React 연동 활성화

  return (
    <div>
      <p>총 {filteredTodos.value.length}개</p>
      {filteredTodos.value.map((todo) => (
        <TodoItem key={todo.id} todo={todo} />
      ))}
    </div>
  );
}

// batch로 여러 signal을 한 번에 업데이트
function resetAll() {
  batch(() => {
    count.value = 0;
    todos.value = [];
    filter.value = 'all';
  });
}

// effect로 부수효과 처리
effect(() => {
  console.log(`현재 필터: ${filter.value}, 결과: ${filteredTodos.value.length}개`);
});
```

### 6.2 TC39 Signals Proposal

TC39 Signals Proposal(Stage 1)은 JavaScript 언어 수준에서 반응형 프리미티브를 표준화하려는 제안이다. 프레임워크 간 상호운용성을 목표로 한다.

```typescript
// TC39 Signals Proposal (Stage 1) - 향후 표준화 예정 문법
// 현재는 폴리필로 실험 가능

// Signal.State - 변경 가능한 값
const counter = new Signal.State(0);

// Signal.Computed - 파생 값 (자동 의존성 추적)
const isEven = new Signal.Computed(() => counter.get() % 2 === 0);

// 값 읽기/쓰기
counter.get();      // 0
counter.set(1);
isEven.get();       // false

// Signal.subtle.Watcher - 변경 감지 (라이브러리 저자용)
const watcher = new Signal.subtle.Watcher(() => {
  console.log('signal changed');
});
watcher.watch(counter);
```

**현재 권장:** TC39 Signals는 아직 Stage 1이므로 프로덕션에서는 `@preact/signals-react`를 사용하고, 표준화 진행 상황을 모니터링한다.

```tsx
// Client Component - 최소한의 인터랙션 상태만 관리
'use client';

function AddToCartButton({ productId, price }: { productId: string; price: number }) {
  const [quantity, setQuantity] = useState(1);
  const [isAdding, setIsAdding] = useState(false);

  const handleAddToCart = async () => {
    setIsAdding(true);
    await addToCart(productId, quantity);
    setIsAdding(false);
  };

  return (
    <div>
      <select
        value={quantity}
        onChange={(e) => setQuantity(Number(e.target.value))}
      >
        {[1, 2, 3, 4, 5].map((n) => (
          <option key={n} value={n}>{n}개</option>
        ))}
      </select>
      <button onClick={handleAddToCart} disabled={isAdding}>
        {isAdding ? '추가 중...' : `장바구니 담기 (${(price * quantity).toLocaleString()}원)`}
      </button>
    </div>
  );
}
```

### 7.2 Context + useReducer 패턴

전역 라이브러리 없이도 복잡한 상태를 관리할 수 있다. 작은 규모의 프로젝트에서는 이것만으로 충분하다.

```tsx
// 타입 정의
interface AppState {
  theme: 'light' | 'dark';
  language: 'ko' | 'en' | 'ja';
  notifications: Notification[];
}

type AppAction =
  | { type: 'SET_THEME'; payload: AppState['theme'] }
  | { type: 'SET_LANGUAGE'; payload: AppState['language'] }
  | { type: 'ADD_NOTIFICATION'; payload: Notification }
  | { type: 'DISMISS_NOTIFICATION'; payload: string };

function appReducer(state: AppState, action: AppAction): AppState {
  switch (action.type) {
    case 'SET_THEME':
      return { ...state, theme: action.payload };
    case 'SET_LANGUAGE':
      return { ...state, language: action.payload };
    case 'ADD_NOTIFICATION':
      return { ...state, notifications: [...state.notifications, action.payload] };
    case 'DISMISS_NOTIFICATION':
      return {
        ...state,
        notifications: state.notifications.filter((n) => n.id !== action.payload),
      };
    default:
      return state;
  }
}

// Context 분리 - 상태와 디스패치를 분리하여 불필요한 리렌더링 방지
const AppStateContext = createContext<AppState | null>(null);
const AppDispatchContext = createContext<React.Dispatch<AppAction> | null>(null);

function AppProvider({ children }: { children: React.ReactNode }) {
  const [state, dispatch] = useReducer(appReducer, {
    theme: 'light',
    language: 'ko',
    notifications: [],
  });

  return (
    <AppStateContext.Provider value={state}>
      <AppDispatchContext.Provider value={dispatch}>
        {children}
      </AppDispatchContext.Provider>
    </AppStateContext.Provider>
  );
}

// 커스텀 훅
function useAppState() {
  const context = useContext(AppStateContext);
  if (!context) throw new Error('useAppState must be used within AppProvider');
  return context;
}

function useAppDispatch() {
  const context = useContext(AppDispatchContext);
  if (!context) throw new Error('useAppDispatch must be used within AppProvider');
  return context;
}

// dispatch만 사용하는 컴포넌트는 상태 변경 시 리렌더링되지 않음
function ThemeToggle() {
  const dispatch = useAppDispatch();
  return (
    <button onClick={() => dispatch({ type: 'SET_THEME', payload: 'dark' })}>
      다크 모드
    </button>
  );
}
```

## 8. 의사결정 트리

### 8.1 프로젝트 규모별 권장 스택

```
이 상태는 서버 데이터인가?
├── YES → TanStack Query v5
│         (캐싱, 리페치, 낙관적 업데이트 모두 해결)
│
└── NO → 이 상태는 URL로 표현 가능한가?
         ├── YES → nuqs / TanStack Router
         │         (필터, 정렬, 페이지, 검색어)
         │
         └── NO → 이 상태는 폼 데이터인가?
                  ├── YES → React Hook Form / Conform
                  │         (유효성 검증, 에러 처리)
                  │
                  └── NO → 여러 컴포넌트가 공유하는가?
                           ├── NO → useState / useReducer
                           │        (로컬 상태로 충분)
                           │
                           └── YES → 상태 구조가 복잡한가?
                                    ├── 단순 (2-3개 값) → Context + useReducer
                                    ├── 중간 (atom 단위 세밀 제어) → Jotai v2
                                    └── 복잡 (여러 도메인 슬라이스) → Zustand v5
```

### 8.2 규모별 권장 조합

| 프로젝트 규모 | Server State | Client State | URL State | Form State |
|---------------|-------------|-------------|-----------|------------|
| **소규모** (랜딩, 블로그) | fetch + use() | useState | 내장 searchParams | 네이티브 form |
| **중소규모** (관리자, 대시보드) | TanStack Query | Context + useReducer | nuqs | React Hook Form |
| **중대규모** (SaaS, 이커머스) | TanStack Query | Zustand v5 | nuqs | React Hook Form |
| **대규모** (복잡한 에디터, 실시간) | TanStack Query | Zustand v5 + Jotai v2 | TanStack Router | Conform |

### 8.3 라이브러리 비교표

| 기준 | Zustand v5 | Jotai v2 | Redux Toolkit | Signals |
|------|-----------|---------|---------------|---------|
| **패러다임** | Flux (단일 스토어) | Atomic (bottom-up) | Flux (단일 스토어) | Reactive (세밀한 구독) |
| **보일러플레이트** | 매우 적음 | 최소 | 보통 | 적음 |
| **번들 크기** | ~1KB | ~2KB | ~11KB | ~3KB |
| **학습 곡선** | 낮음 | 낮음 | 중간 | 낮음 |
| **DevTools** | Redux DevTools | 자체 DevTools | Redux DevTools | 없음 |
| **미들웨어** | persist, devtools, immer | 유틸리티 atom | 풍부한 미들웨어 | 없음 |
| **SSR 지원** | 좋음 | 좋음 | 좋음 | 제한적 |
| **React 외 사용** | 가능 (vanilla) | 불가 | 가능 | 가능 |
| **최적 사용처** | 중대규모 앱 전역 상태 | 세밀한 반응형 UI | 레거시/대규모 팀 | 고성능 인터랙션 |

### 8.4 안티패턴 체크리스트

| 안티패턴 | 문제점 | 개선 방향 |
|----------|--------|-----------|
| 서버 데이터를 Redux/Zustand에 저장 | 캐시 무효화, 동기화 버그 | TanStack Query로 이관 |
| 모든 상태를 단일 전역 스토어에 | 불필요한 리렌더링, 관심사 혼합 | 상태 유형별 분리 |
| URL로 표현 가능한 상태를 useState로 | 새로고침 시 초기화, 공유 불가 | nuqs 도입 |
| Context에 자주 변경되는 큰 객체 | 모든 소비자 리렌더링 | Context 분할 또는 Zustand |
| props drilling 해결을 위한 전역 상태 | 과도한 전역화 | 컴포넌트 합성(composition) 우선 |
