```typescript
// stores/orderFormStore.ts
import { create } from 'zustand';
import { persist, devtools } from 'zustand/middleware';
import { immer } from 'zustand/middleware/immer';

interface ProductSelection {
  productId: string;
  quantity: number;
  options: Record<string, string>;
}

interface ShippingInfo {
  recipientName: string;
  address: string;
  detailAddress: string;
  zipCode: string;
  phone: string;
  memo: string;
}

interface PaymentMethod {
  type: 'card' | 'bank' | 'mobile' | 'point';
  cardNumber?: string;
  bankCode?: string;
}

interface CouponInfo {
  couponId: string | null;
  discountAmount: number;
  pointUsage: number;
}

interface OrderFormState {
  currentStep: number;
  maxVisitedStep: number;
  product: ProductSelection;
  shipping: ShippingInfo;
  payment: PaymentMethod;
  coupon: CouponInfo;
  shippingFee: number;
  isSubmitting: boolean;
  error: string | null;

  // Actions
  setStep: (step: number) => void;
  updateProduct: (data: Partial<ProductSelection>) => void;
  updateShipping: (data: Partial<ShippingInfo>) => void;
  updatePayment: (data: Partial<PaymentMethod>) => void;
  applyCoupon: (couponId: string, discount: number) => void;
  calculateShippingFee: () => Promise<void>;
  submitOrder: () => Promise<{ orderId: string }>;
  reset: () => void;
}

const initialState = {
  currentStep: 0,
  maxVisitedStep: 0,
  product: { productId: '', quantity: 1, options: {} },
  shipping: { recipientName: '', address: '', detailAddress: '', zipCode: '', phone: '', memo: '' },
  payment: { type: 'card' as const },
  coupon: { couponId: null, discountAmount: 0, pointUsage: 0 },
  shippingFee: 0,
  isSubmitting: false,
  error: null,
};

export const useOrderFormStore = create<OrderFormState>()(
  devtools(
    persist(
      immer((set, get) => ({
        ...initialState,

        setStep: (step) => set((state) => {
          state.currentStep = step;
          state.maxVisitedStep = Math.max(state.maxVisitedStep, step);
        }),

        updateProduct: (data) => set((state) => {
          Object.assign(state.product, data);
        }),

        updateShipping: (data) => set((state) => {
          Object.assign(state.shipping, data);
        }),

        updatePayment: (data) => set((state) => {
          Object.assign(state.payment, data);
        }),

        applyCoupon: (couponId, discount) => set((state) => {
          state.coupon.couponId = couponId;
          state.coupon.discountAmount = discount;
        }),

        calculateShippingFee: async () => {
          const { shipping } = get();
          try {
            const res = await fetch('/api/shipping/fee', {
              method: 'POST',
              body: JSON.stringify({ zipCode: shipping.zipCode }),
            });
            const { fee } = await res.json();
            set((state) => { state.shippingFee = fee; });
          } catch {
            set((state) => { state.error = '배송비 계산 실패'; });
          }
        },

        submitOrder: async () => {
          set((state) => { state.isSubmitting = true; state.error = null; });
          const { product, shipping, payment, coupon } = get();
          try {
            const res = await fetch('/api/orders', {
              method: 'POST',
              body: JSON.stringify({ product, shipping, payment, coupon }),
            });
            if (!res.ok) throw new Error('주문 실패');
            const result = await res.json();
            set((state) => { state.isSubmitting = false; });
            return result;
          } catch (err) {
            set((state) => {
              state.isSubmitting = false;
              state.error = err instanceof Error ? err.message : '알 수 없는 에러';
            });
            throw err;
          }
        },

        reset: () => set(initialState),
      })),
      { name: 'order-form-storage', partialize: (state) => ({ product: state.product, shipping: state.shipping }) }
    ),
    { name: 'OrderForm' }
  )
);
```

```typescript
// === Before: Redux Toolkit ===
// features/cart/cartSlice.ts
const cartSlice = createSlice({
  name: 'cart',
  initialState: { items: [], total: 0 } as CartState,
  reducers: {
    addItem: (state, action: PayloadAction<CartItem>) => {
      state.items.push(action.payload);
      state.total += action.payload.price * action.payload.quantity;
    },
    removeItem: (state, action: PayloadAction<string>) => {
      const idx = state.items.findIndex(i => i.id === action.payload);
      if (idx !== -1) {
        state.total -= state.items[idx].price * state.items[idx].quantity;
        state.items.splice(idx, 1);
      }
    },
  },
});

// === After: Zustand ===
// stores/cartStore.ts
import { create } from 'zustand';
import { devtools, persist } from 'zustand/middleware';
import { immer } from 'zustand/middleware/immer';

interface CartStore {
  items: CartItem[];
  total: number;
  addItem: (item: CartItem) => void;
  removeItem: (id: string) => void;
  clearCart: () => void;
}

export const useCartStore = create<CartStore>()(
  devtools(
    persist(
      immer((set) => ({
        items: [],
        total: 0,

        addItem: (item) => set((state) => {
          const existing = state.items.find(i => i.id === item.id);
          if (existing) {
            existing.quantity += item.quantity;
          } else {
            state.items.push(item);
          }
          state.total = state.items.reduce((sum, i) => sum + i.price * i.quantity, 0);
        }),

        removeItem: (id) => set((state) => {
          state.items = state.items.filter(i => i.id !== id);
          state.total = state.items.reduce((sum, i) => sum + i.price * i.quantity, 0);
        }),

        clearCart: () => set({ items: [], total: 0 }),
      })),
      { name: 'cart-storage' }
    ),
    { name: 'CartStore' }
  )
);

// 기존 selector 호환 래퍼
export const cartSelectors = {
  selectItems: (state: CartStore) => state.items,
  selectTotal: (state: CartStore) => state.total,
  selectItemCount: (state: CartStore) => state.items.reduce((sum, i) => sum + i.quantity, 0),
  selectItemById: (id: string) => (state: CartStore) => state.items.find(i => i.id === id),
};

// 점진적 마이그레이션: 브리지 훅
// Redux에서 Zustand로 상태를 동기화하며 점진적으로 전환
export function useCartBridge() {
  const reduxItems = useSelector(selectCartItems);   // 기존 Redux
  const zustandItems = useCartStore(s => s.items);   // 새 Zustand

  // 마이그레이션 완료 여부에 따라 소스 전환
  const USE_ZUSTAND = process.env.NEXT_PUBLIC_USE_ZUSTAND_CART === 'true';
  return USE_ZUSTAND ? zustandItems : reduxItems;
}
```

#### 패턴 1: 인라인 리팩토링

```
Cursor 프롬프트 (Cmd+K로 코드 선택 후):
```

```typescript
// Before: 흩어진 useState
function OrderPage() {
  const [items, setItems] = useState<Item[]>([]);
  const [loading, setLoading] = useState(false);
  const [error, setError] = useState<string | null>(null);
  const [step, setStep] = useState(0);
  const [discount, setDiscount] = useState(0);

  const handleSubmit = async () => {
    setLoading(true);
    setError(null);
    try { /* ... */ } catch (e) { setError(e.message); }
    finally { setLoading(false); }
  };
}

// After: 응집된 useReducer
type OrderAction =
  | { type: 'SET_ITEMS'; payload: Item[] }
  | { type: 'NEXT_STEP' }
  | { type: 'PREV_STEP' }
  | { type: 'APPLY_DISCOUNT'; payload: number }
  | { type: 'SUBMIT_START' }
  | { type: 'SUBMIT_SUCCESS' }
  | { type: 'SUBMIT_ERROR'; payload: string };

interface OrderState {
  items: Item[];
  step: number;
  discount: number;
  loading: boolean;
  error: string | null;
}

function orderReducer(state: OrderState, action: OrderAction): OrderState {
  switch (action.type) {
    case 'SET_ITEMS':
      return { ...state, items: action.payload };
    case 'NEXT_STEP':
      return { ...state, step: Math.min(state.step + 1, 4) };
    case 'PREV_STEP':
      return { ...state, step: Math.max(state.step - 1, 0) };
    case 'APPLY_DISCOUNT':
      return { ...state, discount: action.payload };
    case 'SUBMIT_START':
      return { ...state, loading: true, error: null };
    case 'SUBMIT_SUCCESS':
      return { ...state, loading: false };
    case 'SUBMIT_ERROR':
      return { ...state, loading: false, error: action.payload };
  }
}
```

```typescript
import { Project, SyntaxKind } from 'ts-morph';

interface GlobalStateUsage {
  storeName: string;
  storeFile: string;
  properties: {
    name: string;
    consumers: string[];      // 사용하는 컴포넌트 파일 목록
    isCandidate: boolean;     // 로컬로 내릴 수 있는지
    reason: string;
  }[];
}

function analyzeGlobalStateUsage(projectPath: string): GlobalStateUsage[] {
  const project = new Project({ tsConfigFilePath: `${projectPath}/tsconfig.json` });
  const results: GlobalStateUsage[] = [];

  // Zustand store 파일 탐지
  const storeFiles = project.getSourceFiles('**/stores/**/*.ts');

  for (const file of storeFiles) {
    const exports = file.getExportedDeclarations();
    for (const [name, declarations] of exports) {
      if (!name.startsWith('use')) continue;

      // 해당 훅을 사용하는 모든 파일 탐색
      const refs = declarations[0]?.findReferencesAsNodes() ?? [];
      const consumers = [...new Set(refs.map(r => r.getSourceFile().getFilePath()))];

      results.push({
        storeName: name,
        storeFile: file.getFilePath(),
        properties: [{
          name,
          consumers,
          isCandidate: consumers.length <= 1,
          reason: consumers.length <= 1
            ? `1개 컴포넌트에서만 사용 - 로컬 상태로 전환 권장`
            : `${consumers.length}개 컴포넌트에서 사용 - 전역 유지`,
        }],
      });
    }
  }

  return results;
}
```

```typescript
class ApiClient {
  private baseUrl: string;

  constructor() {
    this.baseUrl = env.apiBaseUrl;
  }

  async get<T>(path: string, options?: RequestInit): Promise<T> {
    const res = await fetch(`${this.baseUrl}${path}`, {

    });

    return res.json();
```

## 3. 2026 상태관리 패러다임

상태를 5가지 카테고리로 분류하고 각각에 최적 도구를 매핑한다.

### 3.1 상태 분류 매트릭스

```
┌────────────────┬───────────────────┬──────────────────┬──────────────────┐
│ 상태 종류       │ 특성              │ 최적 도구         │ 예시             │
├────────────────┼───────────────────┼──────────────────┼──────────────────┤
│ Server State   │ 비동기, 캐시 필요  │ TanStack Query   │ API 데이터       │
│ Client State   │ 동기, UI 전용     │ Zustand / Jotai  │ 모달, 테마       │
│ URL State      │ 공유 가능, 북마크  │ nuqs / URLParams │ 필터, 페이지     │
│ Form State     │ 임시, 검증 필요   │ React Hook Form  │ 입력 폼         │
│ Derived State  │ 다른 상태에서 계산 │ useMemo / Signal │ 합계, 필터 결과  │
└────────────────┴───────────────────┴──────────────────┴──────────────────┘
```

### 3.2 실전 상태 분리 패턴

```typescript
// 하나의 페이지에서 5가지 상태 종류를 올바르게 분리하는 예시
function ProductListPage() {
  // 1. URL State - 필터, 정렬, 페이지네이션 (공유/북마크 가능)
  const [filters, setFilters] = useQueryStates({
    category: parseAsString.withDefault('all'),

    maxPrice: parseAsInteger.withDefault(999999),
    sort: parseAsString.withDefault('newest'),
    page: parseAsInteger.withDefault(1),

  const { data: products, isLoading } = useQuery(
    productListOptions(filters)

  const viewMode = useAppStore((s) => s.productViewMode);

  const { register, handleSubmit } = useForm<{ search: string }>();

  const totalValue = useMemo(
    () => products?.items.reduce((sum, p) => sum + p.price, 0) ?? 0,
    [products]
  );
  const hasActiveFilters = filters.category !== 'all' || filters.minPrice > 0;

  return (
    <div>
      <SearchForm register={register} onSubmit={handleSubmit(onSearch)} />
      <FilterBar filters={filters} onChange={setFilters} hasActive={hasActiveFilters} />
      <ProductGrid items={products?.items ?? []} viewMode={viewMode} />
      <TotalBanner total={totalValue} />
      <Pagination page={filters.page} total={products?.totalPages ?? 0} />
    </div>
  );
```

### 3.3 상태 분류 의사결정 플로우

```
상태가 필요한가?
├── 서버에서 오는 데이터인가? → TanStack Query
├── URL에 반영해야 하는가? (공유/북마크) → nuqs / URL State
├── 폼 입력인가? → React Hook Form
├── 다른 상태에서 계산 가능한가? → useMemo (저장하지 말 것)
└── 순수 UI 상태인가?
    ├── 여러 컴포넌트에서 공유? → Zustand / Jotai
    └── 단일 컴포넌트? → useState / useReducer
```

### 4.2 Optimistic Updates

```typescript
// mutations/useUpdateProduct.ts
export function useUpdateProduct() {
  const queryClient = useQueryClient();

  return useMutation({
    mutationFn: (data: { id: string; updates: Partial<Product> }) =>
      api.patch<Product>(`/products/${data.id}`, data.updates),

    onMutate: async ({ id, updates }) => {
      // 1. 진행 중인 refetch 취소
      await queryClient.cancelQueries({ queryKey: ['products', 'detail', id] });

      // 2. 이전 데이터 백업
      const previous = queryClient.getQueryData(productDetailOptions(id).queryKey);

      // 3. 낙관적 업데이트
      queryClient.setQueryData(productDetailOptions(id).queryKey, (old: Product | undefined) =>
        old ? { ...old, ...updates } : old
      );

      return { previous };
    },

    onError: (_err, { id }, context) => {
      // 4. 에러 시 롤백
      if (context?.previous) {
        queryClient.setQueryData(productDetailOptions(id).queryKey, context.previous);
      }
    },

    onSettled: (_data, _err, { id }) => {
      // 5. 성공/실패 상관없이 재검증
      queryClient.invalidateQueries({ queryKey: ['products', 'detail', id] });
      queryClient.invalidateQueries({ queryKey: ['products', 'list'] });
    },
  });
}
```

## 8. 전역 상태 없는 아키텍처

Server Components와 함께 전역 상태를 최소화하는 설계.

### 8.1 핵심 원칙

```
전역 상태를 만들기 전에 자문:
1. 이 데이터가 서버에서 오는가? → Server Component에서 직접 fetch
2. URL로 표현 가능한가? → searchParams / nuqs
3. 단일 컴포넌트 트리 내에서만 필요한가? → props / Context
4. 폼 입력인가? → React Hook Form (로컬)
5. 위 모두 아닌 경우에만 → 전역 상태 (Zustand / Jotai)
```

### 8.2 Server Components 패턴

```typescript
// 전역 상태가 필요한 최소한의 케이스

// 1. 인증 토큰 (모든 API 요청에 필요)
const useAuthStore = create<{ token: string | null }>()( /* ... */ );

// 2. 테마/다크모드 (전체 앱 UI에 영향)
const useThemeStore = create<{ theme: 'light' | 'dark' }>()( /* ... */ );

// 3. 토스트/알림 큐 (어디서든 추가 가능해야 함)
const useToastStore = create<{ toasts: Toast[] }>()( /* ... */ );

// 4. 웹소켓 연결 상태 (전역 단일 연결)
const useWsStore = create<{ connected: boolean; lastMessage: unknown }>()( /* ... */ );

// 이 외의 대부분은 Server State, URL State, Local State로 해결 가능
```

## 9. 의사결정 트리

### 9.1 상태관리 도구 선택

```
프로젝트 규모?
├── 소규모 (5페이지 이하)
│   ├── 서버 데이터 있음 → TanStack Query + useState
│   └── 서버 데이터 없음 → useState + useReducer
│
├── 중규모 (5~20페이지)
│   ├── Server Components 사용 → TanStack Query + Zustand (최소)
│   ├── SPA → TanStack Query + Zustand
│   └── 복잡한 상태 의존성 → TanStack Query + Jotai
│
└── 대규모 (20페이지+)
    ├── 팀 규모 5인+ → TanStack Query + Zustand (슬라이스)
    ├── 실시간 반응형 UI → TanStack Query + Jotai + Signals
    └── 마이크로프론트엔드 → TanStack Query + Zustand (독립 스토어)
```

### 9.2 TanStack Query vs Zustand vs Jotai

```
┌──────────────┬──────────────────┬──────────────────┬──────────────────┐
│ 기준          │ TanStack Query   │ Zustand          │ Jotai            │
├──────────────┼──────────────────┼──────────────────┼──────────────────┤
│ 주 용도      │ 서버 상태 캐시    │ 전역 클라이언트    │ 원자적 상태       │
│ 학습 곡선    │ 중               │ 낮음              │ 중               │
│ 보일러플레이트│ 낮음             │ 매우 낮음          │ 매우 낮음        │
│ DevTools     │ 공식 지원        │ 공식 지원          │ 공식 지원        │
│ 번들 크기    │ ~39KB            │ ~1KB              │ ~3KB             │
│ 리렌더링     │ 쿼리 키 기반     │ selector 기반      │ atom 기반 (세밀) │
│ SSR          │ hydration 지원   │ persist 주의       │ Provider 방식    │
│ 최적 사용처  │ API 데이터       │ UI 상태, 설정      │ 복잡한 파생 상태  │
└──────────────┴──────────────────┴──────────────────┴──────────────────┘
```

### 9.3 조합 권장 패턴

```
권장 조합 (2026):
1. TanStack Query + Zustand (가장 범용적)
   - Server State → TanStack Query
   - Client State → Zustand
   - URL State → nuqs
   - Form State → React Hook Form

2. TanStack Query + Jotai (복잡한 파생 상태가 많은 경우)
   - Server State → TanStack Query (또는 atomWithQuery)
   - Client State → Jotai atoms
   - Derived State → computed atoms

3. TanStack Query + Signals (퍼포먼스 크리티컬)
   - Server State → TanStack Query
   - UI State → @preact/signals-react
   - 세밀한 DOM 업데이트 필요 시
```
