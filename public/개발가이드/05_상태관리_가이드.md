```typescript
// Phase 1: 서버 상태 분리 - RTK Query → TanStack Query
// hooks/queries/useProducts.ts
import { queryOptions, useQuery } from '@tanstack/react-query';

export const productsQueryOptions = queryOptions({
  queryKey: ['products'],
  queryFn: async () => {
    const res = await fetch('/api/products');
    if (!res.ok) throw new Error('Failed to fetch products');
    return res.json() as Promise<Product[]>;
  },
  staleTime: 5 * 60 * 1000,
});

export const useProducts = () => useQuery(productsQueryOptions);

// Phase 2: 클라이언트 상태 → Zustand v5
// stores/cartStore.ts
import { create } from 'zustand';
import { persist, devtools } from 'zustand/middleware';
import { immer } from 'zustand/middleware/immer';

interface CartItem {
  productId: string;
  quantity: number;
  price: number;
}

interface CartState {
  items: CartItem[];
  addItem: (item: CartItem) => void;
  removeItem: (productId: string) => void;
  updateQuantity: (productId: string, quantity: number) => void;
  clear: () => void;
}

export const useCartStore = create<CartState>()(
  devtools(
    persist(
      immer((set) => ({
        items: [],
        addItem: (item) =>
          set((state) => {
            const existing = state.items.find(
              (i) => i.productId === item.productId
            );
            if (existing) {
              existing.quantity += item.quantity;
            } else {
              state.items.push(item);
            }
          }),
        removeItem: (productId) =>
          set((state) => {
            state.items = state.items.filter(
              (i) => i.productId !== productId
            );
          }),
        updateQuantity: (productId, quantity) =>
          set((state) => {
            const item = state.items.find(
              (i) => i.productId === productId
            );
            if (item) item.quantity = quantity;
          }),
        clear: () => set({ items: [] }),
      })),
      { name: 'cart-storage' }
    ),
    { name: 'CartStore' }
  )
);

// Phase 3: 브릿지 훅 (공존 기간)
// hooks/useCartCompat.ts
// Redux와 Zustand 공존 기간 동안 사용하는 호환 훅
import { useCartStore } from '@/stores/cartStore';

// 기존 Redux useSelector 시그니처와 동일한 인터페이스
export const useCartItems = () => useCartStore((s) => s.items);
export const useCartTotal = () =>
  useCartStore((s) =>
    s.items.reduce((sum, item) => sum + item.price * item.quantity, 0)
  );
export const useCartActions = () =>
  useCartStore((s) => ({
    addItem: s.addItem,
    removeItem: s.removeItem,
    updateQuantity: s.updateQuantity,
    clear: s.clear,
  }));
```

## 2. Server Components 시대의 상태 최소화

React Server Components(RSC)와 Server Actions가 보편화된 2026년, **클라이언트 상태를 최소화**하는 것이 핵심 전략이다.

### 2.1 상태 계층 모델

```
서버 컴포넌트 (상태 없음)
  ├── 데이터 페칭: async/await 직접 호출
  ├── 정적 UI: props만으로 렌더링
  └── 클라이언트 바운더리 ('use client')
        ├── 서버 상태: TanStack Query (캐시 + 동기화)
        ├── URL 상태: nuqs / searchParams
        ├── UI 상태: useState / useReducer (로컬)
        └── 크로스 컴포넌트 상태: Zustand / Jotai (최소한)
```

### 2.2 상태 최소화 패턴

```typescript

'use client';

import { useQueryState, parseAsStringEnum } from 'nuqs';

type OrderStatus = 'all' | 'pending' | 'completed' | 'cancelled';

export function OrderFilterPanel({
  initialOrders,
}: {
  initialOrders: Order[];
}) {
  // URL searchParams가 곧 상태 → 공유 가능, 북마크 가능
  const [status, setStatus] = useQueryState(
    'status',
    parseAsStringEnum<OrderStatus>(['all', 'pending', 'completed', 'cancelled'])
      .withDefault('all')
  );

  const [sort, setSort] = useQueryState('sort', {
    defaultValue: 'date',
  });

  // 서버 상태는 TanStack Query (URL 상태 기반 queryKey)
  const { data: orders } = useQuery({
    queryKey: ['orders', { status, sort }],
    queryFn: () => fetchOrders({ status, sort }),
    initialData: status === 'all' ? initialOrders : undefined,
  });

  return (
    <div>
      <StatusFilter value={status} onChange={setStatus} />
      <SortSelect value={sort} onChange={setSort} />
      <OrderList orders={orders ?? []} />
    </div>
  );
}
```

### 3.2 streamedQuery (서버 스트리밍 데이터)

```typescript
// v5.80+ streamedQuery: SSE/스트리밍 응답을 Query 캐시와 통합
import { useQuery } from '@tanstack/react-query';
import { streamedQuery } from '@tanstack/react-query';

// AI 응답처럼 점진적으로 도착하는 데이터 처리
export function useAIAnalysis(prompt: string) {
  return useQuery({
    queryKey: ['ai-analysis', prompt],
    queryFn: streamedQuery({
      queryFn: async function* () {
        const response = await fetch('/api/ai/analyze', {
          method: 'POST',
          body: JSON.stringify({ prompt }),
        });

        const reader = response.body!.getReader();
        const decoder = new TextDecoder();
        let accumulated = '';

        while (true) {
          const { done, value } = await reader.read();
          if (done) break;

          accumulated += decoder.decode(value, { stream: true });
          // yield할 때마다 data가 업데이트됨
          yield accumulated;
        }

        return accumulated;
      },
    }),
  });
}

// 컴포넌트에서 사용
function AnalysisPanel({ prompt }: { prompt: string }) {
  const { data, isLoading } = useAIAnalysis(prompt);

  return (
    <div>
      {isLoading && <StreamingIndicator />}
      <MarkdownRenderer content={data ?? ''} />
    </div>
  );
}
```

### 3.4 Optimistic Update + Mutation 조합

```typescript
import { useMutation, useQueryClient } from '@tanstack/react-query';
import { productsQueryOptions } from '@/queries/products';

export function useToggleFavorite() {
  const queryClient = useQueryClient();

  return useMutation({
    mutationFn: (productId: string) =>
      fetch(`/api/products/${productId}/favorite`, { method: 'POST' }),

    onMutate: async (productId) => {
      // 진행 중인 refetch 취소
      await queryClient.cancelQueries({ queryKey: ['products'] });

      // 이전 상태 스냅샷
      const previousProducts = queryClient.getQueryData(
        productsQueryOptions({}).queryKey
      );

      // 낙관적 업데이트
      queryClient.setQueryData(
        productsQueryOptions({}).queryKey,
        (old: Product[] | undefined) =>
          old?.map((p) =>
            p.id === productId ? { ...p, isFavorite: !p.isFavorite } : p
          )
      );

      return { previousProducts };
    },

    onError: (_err, _productId, context) => {
      // 실패 시 롤백
      if (context?.previousProducts) {
        queryClient.setQueryData(
          productsQueryOptions({}).queryKey,
          context.previousProducts
        );
      }
    },

    onSettled: () => {
      // 성공/실패 모두 서버 데이터로 동기화
      queryClient.invalidateQueries({ queryKey: ['products'] });
    },
  });
}
```

```typescript
interface Notification {
```

### 6.1 2026 진행 상황

TC39 Signals Proposal은 2026년 4월 현재 **Stage 1**에 위치한다. JavaScript 엔진 레벨에서 반응성 프리미티브를 제공하려는 제안이다.

```typescript
// TC39 Signals Proposal (Stage 1) - 아직 실험적
// 프레임워크 독립적인 반응성 프리미티브

// 향후 표준화 시 예상 API
namespace Signal {
  // State: 쓸 수 있는 반응적 값
  class State<T> {
    constructor(initialValue: T);
    get(): T;
    set(value: T): void;
  }

  // Computed: 파생 값 (자동 의존성 추적)
  class Computed<T> {
    constructor(fn: () => T);
    get(): T;
  }

  // Effect: 부수효과
  function effect(fn: () => void): () => void;
}

// 사용 예시 (향후)
const count = new Signal.State(0);
const doubled = new Signal.Computed(() => count.get() * 2);

Signal.effect(() => {
  console.log(`Count: ${count.get()}, Doubled: ${doubled.get()}`);
});

count.set(5); // 자동 로그: "Count: 5, Doubled: 10"
```

```typescript
// hooks/useReactiveValue.ts (프레임워크 독립 인터페이스)

import { useSyncExternalStore } from 'react';

interface ReactiveValue<T> {
  get: () => T;
  set: (value: T) => void;
  subscribe: (callback: () => void) => () => void;
}

// 현재는 Zustand vanilla store 기반
export function createReactiveValue<T>(initial: T): ReactiveValue<T> {
  let value = initial;
  const listeners = new Set<() => void>();

  return {
    get: () => value,
    set: (newValue) => {
      value = newValue;
      listeners.forEach((l) => l());
    },
    subscribe: (callback) => {
      listeners.add(callback);
      return () => listeners.delete(callback);
    },
  };
}

// React 바인딩
export function useReactiveValue<T>(rv: ReactiveValue<T>) {
  return useSyncExternalStore(rv.subscribe, rv.get);
}
```

```typescript
import { useQueryClient } from '@tanstack/react-query';

interface StateSnapshot {
  version: string;
  timestamp: number;
  environment: string;
  queryCache: Record<string, unknown>;
  zustandStores: Record<string, unknown>;
}

// 스냅샷 내보내기
export function useExportSnapshot() {
  const queryClient = useQueryClient();

  return () => {
    const queryCache = queryClient.getQueryCache();
    const queries: Record<string, unknown> = {};

    queryCache.getAll().forEach((query) => {
      if (query.state.data !== undefined) {
        queries[JSON.stringify(query.queryKey)] = {
          data: query.state.data,
          dataUpdatedAt: query.state.dataUpdatedAt,
        };
      }
    });

    const snapshot: StateSnapshot = {
      version: '1.0',
      timestamp: Date.now(),
      environment: window.location.hostname,
      queryCache: queries,
      zustandStores: {
        ui: useUIStore.getState(),
        cart: useCartStore.getState(),
      },
    };

    // 클립보드에 복사 또는 파일 다운로드
    const blob = new Blob([JSON.stringify(snapshot, null, 2)], {
      type: 'application/json',
    });
    const url = URL.createObjectURL(blob);
    const a = document.createElement('a');
    a.href = url;
    a.download = `state-snapshot-${Date.now()}.json`;
    a.click();
    URL.revokeObjectURL(url);

    return snapshot;
  };
}

// 스냅샷 가져오기
export function useImportSnapshot() {
  const queryClient = useQueryClient();

  return async (file: File) => {
    const text = await file.text();
    const snapshot: StateSnapshot = JSON.parse(text);

    // Query 캐시 복원
    Object.entries(snapshot.queryCache).forEach(([keyStr, cached]) => {
      const queryKey = JSON.parse(keyStr);
      const { data } = cached as { data: unknown; dataUpdatedAt: number };
      queryClient.setQueryData(queryKey, data);
    });

    // Zustand 스토어 복원
    if (snapshot.zustandStores.ui) {
      useUIStore.setState(snapshot.zustandStores.ui as Partial<UIState>);
    }
    if (snapshot.zustandStores.cart) {
      useCartStore.setState(snapshot.zustandStores.cart as Partial<CartState>);
    }

    console.log(
      `Snapshot restored from ${snapshot.environment} at ${new Date(snapshot.timestamp).toISOString()}`
    );
  };
}
```

| **Next.js App Router** | TanStack Query + Server Component 직접 페칭 | Zustand (최소) | searchParams + nuqs |
