## 1. AI 기반 접근성 자동화

접근성 개선의 30% 이상을 AI 자동화로 처리한다. Claude, Copilot을 활용한 접근성 진단 --> 수정 --> 테스트 --> CI 통합 파이프라인을 구축한다.

### 1.1 AI 프롬프트 -- 접근성 개선

> **프롬프트 1: 컴포넌트 접근성 감사**
>
> ```

> 각 문제에 대해 (1) 위반 기준 번호(예: 1.1.1),
> (2) 심각도(Critical/Major/Minor),
> (3) 수정 코드를 제시해줘.
> 특히 키보드 네비게이션, 스크린 리더 호환성, ARIA 속성을 중점 점검해줘.

> {코드 붙여넣기}
> ```

>
> ```

### 1.2 axe-core + AI 자동 수정 제안

axe-core가 탐지한 위반 사항을 AI가 분석하여 구체적인 수정 코드를 자동 생성한다.

```typescript
// scripts/ai-a11y-fix.ts
import { execSync } from "node:child_process";
import { readFileSync, writeFileSync } from "node:fs";

interface AxeViolation {
  id: string;
  impact: "critical" | "serious" | "moderate" | "minor";
  description: string;
  helpUrl: string;
  nodes: Array<{
    html: string;
    target: string[];
    failureSummary: string;
  }>;
}

interface AiFixSuggestion {
  violationId: string;
  originalHtml: string;
  fixedHtml: string;
  explanation: string;
}

// axe-core 결과를 파싱하여 AI에 전달할 컨텍스트 생성
function buildA11yFixPrompt(violations: AxeViolation[]): string {
  const grouped = violations.reduce(
    (acc, v) => {
      const key = v.impact;
      if (!acc[key]) acc[key] = [];
      acc[key].push(v);
      return acc;
    },
    {} as Record<string, AxeViolation[]>,
  );

  let prompt = "다음 axe-core 접근성 위반 사항을 수정해줘.\n";
  prompt += "심각도 순서(critical > serious > moderate > minor)로 처리해줘.\n\n";

  for (const impact of ["critical", "serious", "moderate", "minor"]) {
    const items = grouped[impact];
    if (!items?.length) continue;
    prompt += `## ${impact.toUpperCase()} (${items.length}건)\n\n`;
    for (const v of items) {
      prompt += `### ${v.id}: ${v.description}\n`;
      prompt += `참고: ${v.helpUrl}\n`;
      for (const node of v.nodes) {
        prompt += `\`\`\`html\n${node.html}\n\`\`\`\n`;
        prompt += `문제: ${node.failureSummary}\n\n`;
      }
    }
  }

  return prompt;
}

// 수정 제안을 파일에 적용
function applySuggestions(suggestions: AiFixSuggestion[]): void {
  const report = suggestions.map((s) => ({
    위반: s.violationId,
    원본: s.originalHtml,
    수정: s.fixedHtml,
    설명: s.explanation,
  }));

  writeFileSync(
    "a11y-fix-report.json",
    JSON.stringify(report, null, 2),
    "utf-8",
  );

  console.log(`${suggestions.length}건의 접근성 수정 제안이 생성되었습니다.`);
}
```

### 1.3 Playwright 접근성 자동 테스트

```typescript
// e2e/a11y.spec.ts
import AxeBuilder from "@axe-core/playwright";
import { test, expect, type Page } from "@playwright/test";

// 공통 접근성 검증 헬퍼
async function checkA11y(
  page: Page,
  options?: {
    exclude?: string[];
    disableRules?: string[];
    standard?: "wcag2a" | "wcag2aa" | "wcag2aaa" | "wcag22aa";
  },
) {
  const builder = new AxeBuilder({ page }).withTags([
    options?.standard ?? "wcag22aa",
    "best-practice",
  ]);

  if (options?.exclude) {
    for (const selector of options.exclude) {
      builder.exclude(selector);
    }
  }

  if (options?.disableRules) {
    builder.disableRules(options.disableRules);
  }

  const results = await builder.analyze();

  // 위반 사항을 가독성 높은 형태로 변환
  const violations = results.violations.map((v) => ({
    id: v.id,
    impact: v.impact,
    description: v.description,
    nodes: v.nodes.length,
    elements: v.nodes.map((n) => n.html).slice(0, 3),
  }));

  expect(violations, `접근성 위반 ${violations.length}건 발견`).toEqual([]);
}

test.describe("접근성 자동 테스트", () => {
  test("메인 페이지 접근성 검증", async ({ page }) => {
    await page.goto("/");
    await checkA11y(page);
  });

  test("키보드 네비게이션 검증", async ({ page }) => {
    await page.goto("/");

    // Skip to content 링크 검증
    await page.keyboard.press("Tab");
    const skipLink = page.getByRole("link", { name: /본문으로 이동/i });
    await expect(skipLink).toBeFocused();
    await skipLink.click();

    // 메인 콘텐츠로 포커스 이동 확인
    const main = page.getByRole("main");
    await expect(main).toBeFocused();
  });

  test("포커스 트랩 검증 (모달)", async ({ page }) => {
    await page.goto("/");
    await page.getByRole("button", { name: /열기/i }).click();

    const dialog = page.getByRole("dialog");
    await expect(dialog).toBeVisible();

    // 모달 내부에서 Tab 순환 확인
    const focusableElements = dialog.locator(
      'button, [href], input, select, textarea, [tabindex]:not([tabindex="-1"])',
    );
    const count = await focusableElements.count();

    for (let i = 0; i < count + 1; i++) {
      await page.keyboard.press("Tab");
    }

    // 첫 번째 요소로 다시 포커스 돌아옴
    const firstElement = focusableElements.first();
    await expect(firstElement).toBeFocused();

    // Escape로 모달 닫기
    await page.keyboard.press("Escape");
    await expect(dialog).not.toBeVisible();
  });

  test("ARIA 라이브 리전 알림 검증", async ({ page }) => {
    await page.goto("/form");

    // 폼 제출 후 결과 알림
    await page.getByRole("textbox", { name: /이름/i }).fill("테스트");
    await page.getByRole("button", { name: /제출/i }).click();

    const alert = page.getByRole("alert");
    await expect(alert).toBeVisible();
    await expect(alert).toHaveText(/성공/);
  });

  test("스크린 리더 접근 가능한 이름 검증", async ({ page }) => {
    await page.goto("/");

    // 모든 인터랙티브 요소에 접근 가능한 이름이 있는지 확인
    const buttons = page.getByRole("button");
    const count = await buttons.count();

    for (let i = 0; i < count; i++) {
      const name = await buttons.nth(i).getAttribute("aria-label");
      const text = await buttons.nth(i).textContent();
      expect(name || text?.trim()).toBeTruthy();
    }
  });
});
```

  pull_request:

      - name: Start preview server
        run: npm run preview &
        env:
          PORT: 4173

      - name: Wait for server

      - name: Run axe-core audit
        run: |
          npx @axe-core/cli http://localhost:4173 \
            --tags wcag2aa,wcag22aa,best-practice \
            --exit \
            --save a11y-results.json

      - name: Run Playwright a11y tests
        run: npx playwright test e2e/a11y.spec.ts

      - name: Upload results
        if: always()
        uses: actions/upload-artifact@v4
        with:
          name: a11y-results
          path: |
            a11y-results.json
            test-results/

      - name: Comment PR with results
        if: failure()
        uses: actions/github-script@v7
        with:
          script: |
            const fs = require('fs');
            const results = JSON.parse(fs.readFileSync('a11y-results.json', 'utf8'));
            const violations = results.violations || [];
            let body = '## 접근성 위반 감지\n\n';
            body += `| 규칙 | 심각도 | 요소 수 | 설명 |\n|------|--------|---------|------|\n`;
            for (const v of violations) {
              body += `| ${v.id} | ${v.impact} | ${v.nodes.length} | ${v.description} |\n`;
            }
            await github.rest.issues.createComment({
              owner: context.repo.owner,
              repo: context.repo.repo,
              issue_number: context.issue.number,
              body,
            });

### 1.5 AI로 ARIA 속성 자동 생성

```typescript
// utils/aria-generator.ts

/**
 * 컴포넌트 역할에 따른 필수 ARIA 속성 매핑
 */
const ARIA_PATTERNS: Record<string, Record<string, string | boolean>> = {
  dialog: {
    role: "dialog",
    "aria-modal": "true",
    "aria-labelledby": "{titleId}",
    "aria-describedby": "{descriptionId}",
  },
  accordion: {
    role: "region",
    "aria-labelledby": "{headerId}",
  },
  "accordion-trigger": {
    role: "button",
    "aria-expanded": "{isOpen}",
    "aria-controls": "{panelId}",
  },
  "accordion-panel": {
    role: "region",
    "aria-labelledby": "{triggerId}",
    hidden: "{!isOpen}",
  },
  tabs: {
    role: "tablist",
    "aria-label": "{label}",
    "aria-orientation": "horizontal",
  },
  tab: {
    role: "tab",
    "aria-selected": "{isActive}",
    "aria-controls": "{panelId}",
    tabIndex: "{isActive ? 0 : -1}",
  },
  tabpanel: {
    role: "tabpanel",
    "aria-labelledby": "{tabId}",
    tabIndex: "0",
  },
  tooltip: {
    role: "tooltip",
    "aria-hidden": "{!isVisible}",
  },
  "tooltip-trigger": {
    "aria-describedby": "{tooltipId}",
  },
  combobox: {
    role: "combobox",
    "aria-expanded": "{isOpen}",
    "aria-controls": "{listboxId}",
    "aria-activedescendant": "{activeOptionId}",
    "aria-autocomplete": "list",
    "aria-haspopup": "listbox",
  },
  alert: {
    role: "alert",
    "aria-live": "assertive",
    "aria-atomic": "true",
  },
  status: {
    role: "status",
    "aria-live": "polite",
    "aria-atomic": "true",
  },
};

/**
 * 컴포넌트 유형에 맞는 ARIA 속성을 생성한다.
 */
function generateAriaProps(
  componentType: keyof typeof ARIA_PATTERNS,
  bindings: Record<string, string>,
): Record<string, string | boolean> {
  const pattern = ARIA_PATTERNS[componentType];
  if (!pattern) {
    throw new Error(`알 수 없는 컴포넌트 유형: ${componentType}`);
  }

  const result: Record<string, string | boolean> = {};

  for (const [key, value] of Object.entries(pattern)) {
    if (typeof value === "string" && value.startsWith("{")) {
      const bindingKey = value.slice(1, -1);
      result[key] = bindings[bindingKey] ?? value;
    } else {
      result[key] = value;
    }
  }

  return result;
}

export { generateAriaProps, ARIA_PATTERNS };
```

## 2. WCAG 2.2 AA/AAA 기준 상세

### 2.1 WCAG 2.2 핵심 원칙

| 원칙 | 설명 | 핵심 기준 (AA) |
|------|------|----------------|
| **인식의 용이성** | 모든 콘텐츠를 인식할 수 있어야 한다 | 1.1.1 대체 텍스트, 1.3.1 정보와 관계, 1.4.3 대비 |

| **이해의 용이성** | 콘텐츠를 이해할 수 있어야 한다 | 3.1.1 페이지 언어, 3.2.6 일관된 도움말, 3.3.7 중복 입력 |
| **견고성** | 다양한 보조 기술과 호환되어야 한다 | 4.1.2 이름/역할/값, 4.1.3 상태 메시지 |

### 2.3 AA vs AAA 비교

| 기준 | AA (필수) | AAA (권장) |
|------|-----------|------------|
| 텍스트 대비 | 4.5:1 (일반), 3:1 (대형) | 7:1 (일반), 4.5:1 (대형) |
| 타깃 크기 | 24x24px | 44x44px |
| 포커스 표시 | 가려지지 않아야 함 | 2px 두께, 3:1 대비 |
| 텍스트 간격 조정 | 콘텐츠 손실 없이 조정 가능 | - |
| 타이밍 | 사용자가 조정 가능 | 타이밍 의존 없음 |
| 에러 예방 | 법적/금융 데이터 | 모든 데이터 |

```tsx
// hooks/useFocusTrap.ts
import { useCallback, useEffect, useRef } from "react";

const FOCUSABLE_SELECTOR = [

  'select:not([disabled])',
  'textarea:not([disabled])',
  '[tabindex]:not([tabindex="-1"])',
  '[contenteditable="true"]',
].join(", ");

/**
 * 포커스 트랩 훅 -- 모달, 다이얼로그 등에서 사용
 */
function useFocusTrap(isActive: boolean) {
  const containerRef = useRef<HTMLDivElement>(null);
  const previousFocusRef = useRef<HTMLElement | null>(null);

  useEffect(() => {
    if (!isActive) return;

    // 현재 포커스 저장
    previousFocusRef.current = document.activeElement as HTMLElement;

    const container = containerRef.current;
    if (!container) return;

    // 트랩 내부의 첫 번째 포커스 가능 요소로 이동
    const focusableElements = container.querySelectorAll<HTMLElement>(FOCUSABLE_SELECTOR);
    focusableElements[0]?.focus();

    function handleKeyDown(event: KeyboardEvent) {
      if (event.key !== "Tab" || !container) return;

      const focusable = container.querySelectorAll<HTMLElement>(FOCUSABLE_SELECTOR);
      const firstElement = focusable[0];
      const lastElement = focusable[focusable.length - 1];

      if (event.shiftKey) {
        if (document.activeElement === firstElement) {
          event.preventDefault();
          lastElement?.focus();
        }
      } else {
        if (document.activeElement === lastElement) {
          event.preventDefault();
          firstElement?.focus();
        }
      }
    }

    document.addEventListener("keydown", handleKeyDown);

    return () => {
      document.removeEventListener("keydown", handleKeyDown);
      // 포커스 복원
      previousFocusRef.current?.focus();
    };
  }, [isActive]);

  return containerRef;
}

export { useFocusTrap };
```

### 3.2 키보드 네비게이션

```tsx
// hooks/useRovingFocus.ts
import { useState, useCallback, type KeyboardEvent } from "react";

/**
 * Roving tabindex 패턴 -- 탭 목록, 메뉴 바 등에서 사용
 * 그룹 내에서 방향키로 이동하고, Tab으로 그룹 진입/탈출
 */
function useRovingFocus(itemCount: number) {
  const [activeIndex, setActiveIndex] = useState(0);

  const handleKeyDown = useCallback(
    (event: KeyboardEvent) => {
      let nextIndex = activeIndex;

      switch (event.key) {
        case "ArrowRight":
        case "ArrowDown":
          event.preventDefault();
          nextIndex = (activeIndex + 1) % itemCount;
          break;
        case "ArrowLeft":
        case "ArrowUp":
          event.preventDefault();
          nextIndex = (activeIndex - 1 + itemCount) % itemCount;
          break;
        case "Home":
          event.preventDefault();
          nextIndex = 0;
          break;
        case "End":
          event.preventDefault();
          nextIndex = itemCount - 1;
          break;
        default:
          return;
      }

      setActiveIndex(nextIndex);
    },
    [activeIndex, itemCount],
  );

  const getItemProps = useCallback(
    (index: number) => ({
      tabIndex: index === activeIndex ? 0 : -1,
      "aria-selected": index === activeIndex,
      onFocus: () => setActiveIndex(index),
    }),
    [activeIndex],
  );

  return { activeIndex, handleKeyDown, getItemProps };
}

export { useRovingFocus };
```

/**
 * 스크린 리더에 동적 알림을 전달하는 라이브 리전
 */
function LiveRegion({
  politeness = "polite",
}: {
  politeness?: "polite" | "assertive";
}) {
  const [message, setMessage] = useState("");
  const timeoutRef = useRef<ReturnType<typeof setTimeout>>();

  const announce = useCallback(
    (text: string, clearAfter = 5000) => {
      // 동일 메시지 연속 발화를 위해 초기화 후 재설정
      setMessage("");
      requestAnimationFrame(() => {
        setMessage(text);
      });

      if (clearAfter > 0) {
        clearTimeout(timeoutRef.current);
        timeoutRef.current = setTimeout(() => setMessage(""), clearAfter);
      }
    },
    [],
  );

  useEffect(() => {
    return () => clearTimeout(timeoutRef.current);
  }, []);

  return (
    <div
      role={politeness === "assertive" ? "alert" : "status"}
      aria-live={politeness}
      aria-atomic="true"
      style={{
        position: "absolute",
        width: "1px",
        height: "1px",
        overflow: "hidden",
        clip: "rect(0, 0, 0, 0)",
      }}
    >
      {message}
    </div>

// hooks/useFormErrors.ts
import { useCallback, useRef } from "react";

/**
 * 폼 에러 발생 시 해당 필드로 포커스를 이동하고
 * 에러 요약을 라이브 리전으로 알린다.
 */
function useFormErrors() {
  const errorSummaryRef = useRef<HTMLDivElement>(null);

  const focusFirstError = useCallback((errors: Record<string, string>) => {
    const errorKeys = Object.keys(errors);
    if (errorKeys.length === 0) return;

    // 에러 요약에 포커스
    errorSummaryRef.current?.focus();

    // 또는 첫 번째 에러 필드로 포커스
    const firstErrorField = document.getElementById(errorKeys[0]);
    firstErrorField?.focus();
  }, []);

  const ErrorSummary = ({ errors }: { errors: Record<string, string> }) => {
    const errorEntries = Object.entries(errors);
    if (errorEntries.length === 0) return null;

    return (
      <div
        ref={errorSummaryRef}
        role="alert"
        tabIndex={-1}
        aria-label={`${errorEntries.length}개의 오류가 있습니다`}
      >
        <h2>입력 오류 {errorEntries.length}건</h2>
        <ul>
          {errorEntries.map(([fieldId, message]) => (
            <li key={fieldId}>
              <a href={`#${fieldId}`}>{message}</a>
            </li>
          ))}
        </ul>
      </div>
    );
  };

  return { focusFirstError, ErrorSummary };
}

// utils/color-contrast.ts

/**
 * 상대 휘도 계산 (WCAG 2.2 기준)
 */
function getRelativeLuminance(hex: string): number {
  const rgb = hexToRgb(hex);
  const [r, g, b] = rgb.map((c) => {
    const sRGB = c / 255;
    return sRGB <= 0.03928
      ? sRGB / 12.92
      : Math.pow((sRGB + 0.055) / 1.055, 2.4);
  });
  return 0.2126 * r + 0.7152 * g + 0.0722 * b;
}

/**
 * 두 색상 간 대비율 계산
 */
function getContrastRatio(color1: string, color2: string): number {
  const l1 = getRelativeLuminance(color1);
  const l2 = getRelativeLuminance(color2);
  const lighter = Math.max(l1, l2);
  const darker = Math.min(l1, l2);
  return (lighter + 0.05) / (darker + 0.05);
}

/**
 * WCAG 2.2 대비 기준 충족 여부 확인
 */
function meetsContrastRequirement(
  foreground: string,
  background: string,
  level: "AA" | "AAA" = "AA",
  isLargeText = false,
): { passes: boolean; ratio: number; required: number } {
  const ratio = getContrastRatio(foreground, background);

  const required =
    level === "AAA"
      ? isLargeText ? 4.5 : 7
      : isLargeText ? 3 : 4.5;

  return { passes: ratio >= required, ratio: Math.round(ratio * 100) / 100, required };
}

function hexToRgb(hex: string): [number, number, number] {
  const cleaned = hex.replace("#", "");
  return [
    parseInt(cleaned.slice(0, 2), 16),
    parseInt(cleaned.slice(2, 4), 16),
    parseInt(cleaned.slice(4, 6), 16),
  ];
}

export { getContrastRatio, meetsContrastRequirement, getRelativeLuminance };

// hooks/useReducedMotion.ts
import { useState, useEffect } from "react";

/**
 * 사용자의 모션 감소 선호 감지
 */
function useReducedMotion(): boolean {
  const [prefersReducedMotion, setPrefersReducedMotion] = useState(() => {
    if (typeof window === "undefined") return false;
    return window.matchMedia("(prefers-reduced-motion: reduce)").matches;
  });

  useEffect(() => {
    const mediaQuery = window.matchMedia("(prefers-reduced-motion: reduce)");

    function handleChange(event: MediaQueryListEvent) {
      setPrefersReducedMotion(event.matches);
    }

    mediaQuery.addEventListener("change", handleChange);
    return () => mediaQuery.removeEventListener("change", handleChange);
  }, []);

  return prefersReducedMotion;
}

export { useReducedMotion };

## 8. 접근성 체크리스트

### 8.1 개발 시 체크리스트

- [ ] 모든 이미지에 의미 있는 `alt` 텍스트 또는 `alt=""` (장식 이미지) 제공
- [ ] 제목 계층 (`h1` --> `h2` --> `h3`) 건너뛰기 없이 유지
- [ ] 모든 인터랙티브 요소가 키보드로 접근 가능
- [ ] 커스텀 컴포넌트에 적절한 ARIA role, state, property 부여
- [ ] 포커스 순서가 논리적이며 시각적 순서와 일치
- [ ] 포커스 인디케이터가 항상 보이며 가려지지 않음 (2.4.11)
- [ ] 색상만으로 정보를 전달하지 않음
- [ ] 텍스트 대비율 AA 기준 충족 (4.5:1 / 3:1)
- [ ] 터치 타깃 크기 최소 24x24px (2.5.8)
- [ ] 폼 필드에 `<label>` 연결 및 에러 메시지 `aria-describedby` 연결
- [ ] 동적 콘텐츠 변경 시 `aria-live` 리전으로 알림
- [ ] 모달/다이얼로그에 포커스 트랩 적용

### 8.2 배포 전 체크리스트

- [ ] axe-core 자동 스캔 위반 0건
- [ ] Lighthouse 접근성 점수 95점 이상
- [ ] 키보드만으로 모든 기능 완료 가능 (Tab, Enter, Space, Escape, Arrow)
- [ ] VoiceOver (macOS) 또는 NVDA (Windows)로 주요 플로우 테스트
- [ ] `prefers-reduced-motion: reduce` 시 애니메이션 비활성화 확인
- [ ] `prefers-color-scheme: dark` 시 대비 기준 유지 확인
- [ ] `forced-colors: active` (고대비 모드) 시 정보 손실 없음
- [ ] 200% 확대 시 콘텐츠 잘림/겹침 없음
- [ ] Skip to content 링크 동작 확인
- [ ] `lang` 속성이 `<html>`에 설정됨

### 8.3 코드 리뷰 체크리스트

- [ ] `<div onClick>` 대신 `<button>` 또는 적절한 시맨틱 요소 사용
- [ ] 아이콘 버튼에 `aria-label` 제공
- [ ] 조건부 렌더링 시 포커스 관리 처리
- [ ] 에러 상태에 `aria-invalid`, `aria-describedby` 적용
- [ ] 로딩 상태에 `aria-busy`, `aria-live` 적용
- [ ] `tabIndex` 양수값 사용하지 않음
- [ ] `autofocus` 속성 남용하지 않음
