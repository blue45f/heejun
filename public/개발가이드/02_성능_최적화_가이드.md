```typescript
// "이 함수를 Web Worker로 분리하고, Comlink으로 래핑하고,
//  React hook으로 만들어줘. 진행률 콜백도 추가."

// TO-BE: Worker 분리 + Comlink 래핑
// worker.ts
import { expose } from "comlink";

const api = {
  processLargeDataset(
    data: DataItem[],
    onProgress: (pct: number) => void
  ): ProcessedResult {
    const filtered = data.filter(item => item.score > threshold);
    const results: TransformedItem[] = [];

    for (let i = 0; i < filtered.length; i++) {
      results.push(heavyTransform(filtered[i]));
      if (i % 1000 === 0) onProgress((i / filtered.length) * 100);
    }

    return results.sort((a, b) => b.priority - a.priority);
  },
};

expose(api);

// useProcessData.ts
import { wrap } from "comlink";

export function useProcessData() {
  const [progress, setProgress] = useState(0);
  const [result, setResult] = useState<ProcessedResult | null>(null);

  const process = useCallback(async (data: DataItem[]) => {
    const worker = new Worker(
      new URL("./worker.ts", import.meta.url),
      { type: "module" }
    );
    const api = wrap<typeof import("./worker")["api"]>(worker);

    const processed = await api.processLargeDataset(
      data,
      proxy((pct: number) => setProgress(pct))
    );

    setResult(processed);
    worker.terminate();
  }, []);

  return { process, progress, result };
}
```

```typescript
interface VirtualListProps<T> {

  overscan = 5,

        }}
```

```yaml
# .github/workflows/lighthouse-ai-analysis.yml
name: Lighthouse AI Analysis

on:
  pull_request:
    branches: [main, develop]

          urls: |
            http://localhost:3000/

          temporaryPublicStorage: true
          configPath: .lighthouserc.json

      - name: Extract Lighthouse Results
        id: extract
        run: |
          # Lighthouse 결과 JSON 추출
          RESULTS=$(cat .lighthouseci/lhr-*.json | jq -s '{
            pages: [.[] | {
              url: .finalUrl,
              performance: .categories.performance.score,
              lcp: .audits["largest-contentful-paint"].numericValue,
              inp: .audits["experimental-interaction-to-next-paint"].numericValue,
              cls: .audits["cumulative-layout-shift"].numericValue,

            -H "content-type: application/json" \
            -d "$(jq -n --arg results "${{ steps.extract.outputs.results }}" '{
              model: "claude-sonnet-4-20250514",
              max_tokens: 2048,
              messages: [{
                role: "user",
                content: ("Lighthouse 분석 결과를 바탕으로 성능 개선 리포트를 작성해줘.\n\n## 요구사항\n1. 각 페이지별 Core Web Vitals 상태 (Good/Needs Improvement/Poor)\n2. 가장 시급한 개선 항목 Top 3 (구체적 코드 수정 방안 포함)\n3. 미사용 JS/CSS 제거 시 예상 절감량\n4. 전체 성능 점수 요약 테이블\n\n마크다운 형식으로 출력. 코드 예시 포함.\n\n데이터:\n" + $results)

      - name: Comment PR with Analysis

              repo: context.repo.repo,
              issue_number: context.issue.number,
              body: `## Lighthouse AI 성능 분석 리포트\n\n${analysis}\n\n---\n[전체 Lighthouse 리포트 보기](${lhciUrl})`
            });
```

### 1.4 AI 이미지 최적화 파이프라인

sharp 기반 이미지 최적화에 AI 품질 판단을 결합한 자동화 파이프라인입니다.

```typescript
// scripts/optimize-images.ts
import sharp from "sharp";
import { readdir, stat, writeFile } from "fs/promises";
import { join, extname, basename } from "path";

interface OptimizationResult {
  file: string;
  originalSize: number;
  optimizedSize: number;
  format: string;
  quality: number;
  reductionPct: number;
}

interface ImageProfile {
  maxWidth: number;
  quality: number;
  formats: ("avif" | "webp" | "original")[];
}

// 이미지 용도별 프로파일
const PROFILES: Record<string, ImageProfile> = {
  hero: { maxWidth: 1920, quality: 80, formats: ["avif", "webp", "original"] },
  thumbnail: { maxWidth: 400, quality: 70, formats: ["avif", "webp"] },
  icon: { maxWidth: 128, quality: 85, formats: ["webp", "original"] },
  og: { maxWidth: 1200, quality: 80, formats: ["original"] },
};

// AI 품질 판단: 이미지 복잡도에 따른 품질 계수 자동 결정
async function analyzeImageComplexity(
  buffer: Buffer
): Promise<{ complexity: "low" | "medium" | "high"; qualityAdjust: number }> {
  const metadata = await sharp(buffer).metadata();
  const stats = await sharp(buffer).stats();

  // 채널별 엔트로피 기반 복잡도 추정
  const avgEntropy =
    stats.channels.reduce((sum, ch) => sum + (ch.stdev / 255), 0) /
    stats.channels.length;

  if (avgEntropy < 0.15) {
    return { complexity: "low", qualityAdjust: -15 }; // 단순 이미지: 품질 낮춰도 OK
  }
  if (avgEntropy < 0.35) {
    return { complexity: "medium", qualityAdjust: 0 };
  }
  return { complexity: "high", qualityAdjust: 10 }; // 복잡한 이미지: 품질 유지
}

async function optimizeImage(
  inputPath: string,
  outputDir: string,
  profile: ImageProfile
): Promise<OptimizationResult[]> {
  const buffer = await sharp(inputPath).toBuffer();
  const originalSize = buffer.length;
  const { complexity, qualityAdjust } = await analyzeImageComplexity(buffer);
  const adjustedQuality = Math.min(
    95,
    Math.max(30, profile.quality + qualityAdjust)
  );

  const results: OptimizationResult[] = [];
  const name = basename(inputPath, extname(inputPath));

  for (const format of profile.formats) {
    let pipeline = sharp(buffer).resize({
      width: profile.maxWidth,
      withoutEnlargement: true,
    });

    let outputPath: string;

    switch (format) {
      case "avif":
        pipeline = pipeline.avif({ quality: adjustedQuality, effort: 6 });
        outputPath = join(outputDir, `${name}.avif`);
        break;
      case "webp":
        pipeline = pipeline.webp({ quality: adjustedQuality, effort: 5 });
        outputPath = join(outputDir, `${name}.webp`);
        break;
      default:
        pipeline = pipeline.jpeg({ quality: adjustedQuality, mozjpeg: true });
        outputPath = join(outputDir, `${name}.jpg`);
        break;
    }

    const optimized = await pipeline.toBuffer();
    await writeFile(outputPath, optimized);

    results.push({
      file: outputPath,
      originalSize,
      optimizedSize: optimized.length,
      format,
      quality: adjustedQuality,
      reductionPct: Math.round((1 - optimized.length / originalSize) * 100),
    });
  }

  console.log(
    `[${complexity}] ${basename(inputPath)}: quality=${adjustedQuality}, ` +
      `formats=${profile.formats.join(",")}`
  );

  return results;
}

// 디렉토리 일괄 최적화 + 리포트 생성
async function optimizeDirectory(
  inputDir: string,
  outputDir: string,
  profileName: string
): Promise<void> {
  const profile = PROFILES[profileName];
  if (!profile) throw new Error(`Unknown profile: ${profileName}`);

  const files = await readdir(inputDir);
  const imageFiles = files.filter((f) =>
    /\.(jpg|jpeg|png|webp)$/i.test(f)
  );

  const allResults: OptimizationResult[] = [];

  for (const file of imageFiles) {
    const results = await optimizeImage(
      join(inputDir, file),
      outputDir,
      profile
    );
    allResults.push(...results);
  }

  // 최적화 리포트 생성
  const totalOriginal = allResults.reduce((s, r) => s + r.originalSize, 0);
  const totalOptimized = allResults.reduce((s, r) => s + r.optimizedSize, 0);

  console.log("\n=== 이미지 최적화 리포트 ===");
  console.log(`처리 파일 수: ${imageFiles.length}`);
  console.log(`생성 파일 수: ${allResults.length}`);
  console.log(
    `총 절감: ${(totalOriginal / 1024).toFixed(0)}KB -> ` +
      `${(totalOptimized / 1024).toFixed(0)}KB ` +
      `(-${Math.round((1 - totalOptimized / totalOriginal) * 100)}%)`
  );
}
```

### 2.3 Feature Flag별 성능 영향 A/B 측정

Feature Flag를 기반으로 성능 영향을 격리 측정하는 시스템입니다.

```typescript
// lib/perf-feature-flag.ts

interface PerfMeasurement {
  flagName: string;
  variant: string;
  metrics: {
    lcp: number;
    inp: number;
    cls: number;
    customTimings: Record<string, number>;
  };
  timestamp: number;
  sessionId: string;
  url: string;
}

// Feature Flag에 성능 측정 래퍼를 결합
function withPerfTracking<T extends (...args: unknown[]) => unknown>(
  flagName: string,
  variant: string,
  fn: T
): T {
  return ((...args: Parameters<T>) => {
    const startMark = `ff_${flagName}_${variant}_start`;
    const endMark = `ff_${flagName}_${variant}_end`;

    performance.mark(startMark);
    const result = fn(...args);

    // Promise인 경우 비동기 처리
    if (result instanceof Promise) {
      return result.finally(() => {
        performance.mark(endMark);
        performance.measure(`ff_${flagName}_${variant}`, startMark, endMark);
        reportFlagPerformance(flagName, variant);
      });
    }

    performance.mark(endMark);
    performance.measure(`ff_${flagName}_${variant}`, startMark, endMark);
    reportFlagPerformance(flagName, variant);
    return result;
  }) as T;
}

// RUM 엔드포인트로 전송
async function reportFlagPerformance(
  flagName: string,
  variant: string
): Promise<void> {
  const entries = performance.getEntriesByName(
    `ff_${flagName}_${variant}`
  );
  const latestEntry = entries[entries.length - 1];

  if (!latestEntry) return;

  const measurement: PerfMeasurement = {
    flagName,
    variant,
    metrics: {
      lcp: getWebVital("LCP"),
      inp: getWebVital("INP"),
      cls: getWebVital("CLS"),
      customTimings: {
        duration: latestEntry.duration,
      },
    },
    timestamp: Date.now(),
    sessionId: getSessionId(),
    url: location.href,
  };

  // Beacon API로 비차단 전송
  navigator.sendBeacon(
    "/api/rum/feature-perf",
    JSON.stringify(measurement)
  );
}

function getWebVital(name: string): number {
  // web-vitals 라이브러리 연동
  return (globalThis as Record<string, number>)[`__wv_${name}`] ?? 0;
}

function getSessionId(): string {
  let id = sessionStorage.getItem("perf_session");
  if (!id) {
    id = crypto.randomUUID();
    sessionStorage.setItem("perf_session", id);
  }
  return id;
}

// 사용 예시
export function useFeatureWithPerf<T>(
  flagName: string,
  variants: Record<string, () => T>
): T {
  const activeVariant = getFeatureFlagVariant(flagName); // 외부 FF 시스템 연동
  const fn = variants[activeVariant] ?? variants["control"];

  const tracked = withPerfTracking(flagName, activeVariant, fn);
  return tracked();
}
```

### 2.4 동적 환경 생성 시 CDN 캐시 워밍 자동화

Preview/Beta 환경이 생성될 때 주요 에셋을 CDN에 미리 캐시하여 첫 방문 성능 저하를 방지합니다.

```typescript
// scripts/cdn-cache-warm.ts

interface WarmTarget {
  path: string;
  priority: "critical" | "high" | "normal";
}

// 크리티컬 에셋 목록 자동 추출 (빌드 manifest 기반)
async function extractCriticalAssets(
  buildDir: string
): Promise<WarmTarget[]> {
  const manifestPath = `${buildDir}/.vite/manifest.json`;
  const manifest: Record<string, { file: string; css?: string[] }> =
    JSON.parse(await Bun.file(manifestPath).text());

  const targets: WarmTarget[] = [];

  for (const [entry, info] of Object.entries(manifest)) {
    // 엔트리 포인트 JS = critical
    if (entry.includes("index") || entry.includes("main")) {
      targets.push({ path: `/${info.file}`, priority: "critical" });
    }

    // CSS = critical
    if (info.css) {
      for (const css of info.css) {
        targets.push({ path: `/${css}`, priority: "critical" });
      }
    }

    // 주요 라우트 청크 = high
    if (entry.includes("pages/")) {
      targets.push({ path: `/${info.file}`, priority: "high" });
    }
  }

  return targets;
}

// CDN 캐시 워밍 실행
async function warmCDNCache(
  baseUrl: string,
  targets: WarmTarget[]
): Promise<void> {
  // 우선순위순 정렬
  const sorted = targets.sort((a, b) => {
    const order = { critical: 0, high: 1, normal: 2 };
    return order[a.priority] - order[b.priority];
  });

  // 동시 요청 제한 (CDN 부하 방지)
  const concurrency = 5;
  const results: { path: string; status: number; time: number }[] = [];

  for (let i = 0; i < sorted.length; i += concurrency) {
    const batch = sorted.slice(i, i + concurrency);
    const batchResults = await Promise.all(
      batch.map(async (target) => {
        const start = performance.now();
        const res = await fetch(`${baseUrl}${target.path}`, {
          method: "GET",
          headers: { "X-Cache-Warm": "true" },
        });
        const time = performance.now() - start;
        return { path: target.path, status: res.status, time };
      })
    );
    results.push(...batchResults);
  }

  // 결과 리포트
  console.log("\n=== CDN Cache Warming Report ===");
  console.log(`Environment: ${baseUrl}`);
  console.log(`Total assets: ${results.length}`);
  console.log(
    `Success: ${results.filter((r) => r.status === 200).length}`
  );
  console.log(
    `Avg response: ${(results.reduce((s, r) => s + r.time, 0) / results.length).toFixed(0)}ms`
  );

  const failures = results.filter((r) => r.status !== 200);
  if (failures.length > 0) {
    console.log(`\nFailures:`);
    failures.forEach((f) => console.log(`  ${f.path}: ${f.status}`));
  }
}

// CI/CD 훅에서 실행
async function main(): Promise<void> {
  const envUrl = process.env.PREVIEW_URL;
  const buildDir = process.env.BUILD_DIR ?? "dist";

  if (!envUrl) {
    throw new Error("PREVIEW_URL environment variable required");
  }

  const targets = await extractCriticalAssets(buildDir);
  await warmCDNCache(envUrl, targets);
}

main().catch(console.error);
```

### 2.5 N개 베타 환경 동시 성능 모니터링 대시보드

여러 베타 환경의 성능 메트릭을 단일 대시보드에서 통합 모니터링합니다.

```typescript
// lib/multi-env-monitor.ts

interface EnvironmentMetrics {
  envId: string;
  envUrl: string;
  branch: string;
  prNumber?: number;
  timestamp: number;
  vitals: {
    lcp: { p50: number; p75: number; p95: number };
    inp: { p50: number; p75: number; p95: number };
    cls: { p50: number; p75: number; p95: number };
  };
  bundle: {
    totalJS: number;
    totalCSS: number;
    largestChunk: number;
  };
  lighthouseScore: number;
}

interface MonitorConfig {
  environments: {
    id: string;
    url: string;
    branch: string;
    prNumber?: number;
  }[];
  thresholds: {
    lcp: number;   // ms
    inp: number;   // ms
    cls: number;
    bundleJS: number; // KB
  };
  checkInterval: number; // minutes
}

// 환경 목록 자동 수집 (GitHub API 기반)
async function discoverPreviewEnvironments(
  owner: string,
  repo: string,
  token: string
): Promise<MonitorConfig["environments"]> {
  const res = await fetch(
    `https://api.github.com/repos/${owner}/${repo}/deployments?per_page=20`,
    { headers: { Authorization: `Bearer ${token}` } }
  );
  const deployments = await res.json();

  const environments: MonitorConfig["environments"] = [];

  for (const deployment of deployments) {
    const statusRes = await fetch(deployment.statuses_url, {
      headers: { Authorization: `Bearer ${token}` },
    });
    const statuses = await statusRes.json();

    if (statuses[0]?.state === "success" && statuses[0]?.environment_url) {
      environments.push({
        id: `deploy-${deployment.id}`,
        url: statuses[0].environment_url,
        branch: deployment.ref,
        prNumber: deployment.payload?.pr_number,
      });
    }
  }

  return environments;
}

// 환경별 Synthetic 성능 체크
async function checkEnvironmentPerf(
  env: MonitorConfig["environments"][0]
): Promise<EnvironmentMetrics> {
  // Lighthouse Node API 사용
  const { default: lighthouse } = await import("lighthouse");
  const chromeLauncher = await import("chrome-launcher");

  const chrome = await chromeLauncher.launch({ chromeFlags: ["--headless"] });

  const result = await lighthouse(env.url, {
    port: chrome.port,
    output: "json",
    onlyCategories: ["performance"],
  });

  await chrome.kill();

  const lhr = result?.lhr;
  if (!lhr) throw new Error(`Lighthouse failed for ${env.url}`);

  return {
    envId: env.id,
    envUrl: env.url,
    branch: env.branch,
    prNumber: env.prNumber,
    timestamp: Date.now(),
    vitals: {
      lcp: {
        p50: lhr.audits["largest-contentful-paint"].numericValue * 0.8,
        p75: lhr.audits["largest-contentful-paint"].numericValue,
        p95: lhr.audits["largest-contentful-paint"].numericValue * 1.3,
      },
      inp: {
        p50: lhr.audits["total-blocking-time"].numericValue * 0.6,
        p75: lhr.audits["total-blocking-time"].numericValue * 0.8,
        p95: lhr.audits["total-blocking-time"].numericValue,
      },
      cls: {
        p50: lhr.audits["cumulative-layout-shift"].numericValue * 0.7,
        p75: lhr.audits["cumulative-layout-shift"].numericValue,
        p95: lhr.audits["cumulative-layout-shift"].numericValue * 1.5,
      },
    },
    bundle: {
      totalJS: lhr.audits["total-byte-weight"]?.numericValue ?? 0,
      totalCSS: 0,
      largestChunk: 0,
    },
    lighthouseScore: (lhr.categories.performance?.score ?? 0) * 100,
  };
}

// 전체 환경 스캔 + 이상치 알림
async function scanAllEnvironments(
  config: MonitorConfig
): Promise<{
  results: EnvironmentMetrics[];
  alerts: string[];
}> {
  const results: EnvironmentMetrics[] = [];
  const alerts: string[] = [];

  for (const env of config.environments) {
    try {
      const metrics = await checkEnvironmentPerf(env);
      results.push(metrics);

      // 임계값 초과 체크
      if (metrics.vitals.lcp.p75 > config.thresholds.lcp) {
        alerts.push(
          `[${env.branch}] LCP ${metrics.vitals.lcp.p75.toFixed(0)}ms > ${config.thresholds.lcp}ms`
        );
      }
      if (metrics.vitals.inp.p75 > config.thresholds.inp) {
        alerts.push(
          `[${env.branch}] INP ${metrics.vitals.inp.p75.toFixed(0)}ms > ${config.thresholds.inp}ms`
        );
      }
      if (metrics.lighthouseScore < 90) {
        alerts.push(
          `[${env.branch}] Lighthouse ${metrics.lighthouseScore.toFixed(0)} < 90`
        );
      }
    } catch (err) {
      alerts.push(`[${env.branch}] Check failed: ${(err as Error).message}`);
    }
  }

  return { results, alerts };
}

export {
  discoverPreviewEnvironments,
  checkEnvironmentPerf,
  scanAllEnvironments,
  type MonitorConfig,
  type EnvironmentMetrics,
};
```

### 2.6 Edge Function 기반 환경별 라우팅 + 성능 메트릭 수집

Edge Function에서 환경별 라우팅과 실시간 성능 메트릭 수집을 동시에 처리합니다.

```typescript
// edge-functions/env-router.ts
// Vercel Edge Middleware 또는 Cloudflare Workers 호환

interface EnvRoute {
  pattern: RegExp;
  targetOrigin: string;
  envId: string;
}

interface PerfEntry {
  envId: string;
  path: string;
  edgeLatency: number;
  originLatency: number;
  cacheStatus: string;
  region: string;
  timestamp: number;
}

// 환경별 라우팅 테이블 (KV/Edge Config에서 동적 로드 가능)
const ENV_ROUTES: EnvRoute[] = [
  {
    pattern: /^beta-(\d+)\./,
    targetOrigin: "https://beta-$1.app.example.com",
    envId: "beta-$1",
  },
  {
    pattern: /^pr-(\d+)\./,
    targetOrigin: "https://pr-$1.preview.example.com",
    envId: "pr-$1",
  },
];

export default {
  async fetch(request: Request): Promise<Response> {
    const url = new URL(request.url);
    const hostname = url.hostname;
    const startTime = performance.now();

    // 환경 라우팅 결정
    let targetOrigin = "https://app.example.com"; // default: production
    let envId = "production";

    for (const route of ENV_ROUTES) {
      const match = hostname.match(route.pattern);
      if (match) {
        targetOrigin = route.targetOrigin.replace(
          /\$(\d+)/g,
          (_, i) => match[parseInt(i)]
        );
        envId = route.envId.replace(
          /\$(\d+)/g,
          (_, i) => match[parseInt(i)]
        );
        break;
      }
    }

    // Origin으로 프록시
    const originUrl = `${targetOrigin}${url.pathname}${url.search}`;
    const originStart = performance.now();

    const originResponse = await fetch(originUrl, {
      method: request.method,
      headers: request.headers,
      body: request.body,
    });

    const originLatency = performance.now() - originStart;
    const edgeLatency = performance.now() - startTime;
    const cacheStatus =
      originResponse.headers.get("x-cache") ?? "MISS";

    // 성능 메트릭 비차단 수집
    const perfEntry: PerfEntry = {
      envId,
      path: url.pathname,
      edgeLatency,
      originLatency,
      cacheStatus,
      region: request.headers.get("cf-ray")?.split("-")[1] ?? "unknown",
      timestamp: Date.now(),
    };

    // 비동기로 메트릭 저장 (요청 완료를 블로킹하지 않음)
    const ctx = (globalThis as { waitUntil?: (p: Promise<unknown>) => void });
    if (ctx.waitUntil) {
      ctx.waitUntil(storeMetric(perfEntry));
    }

    // Server-Timing 헤더 추가 (DevTools에서 확인 가능)
    const response = new Response(originResponse.body, originResponse);
    response.headers.set(
      "Server-Timing",
      `edge;dur=${edgeLatency.toFixed(1)}, origin;dur=${originLatency.toFixed(1)}, env;desc="${envId}"`
    );

    return response;
  },
};

async function storeMetric(entry: PerfEntry): Promise<void> {
  // Analytics 엔드포인트로 전송 (Cloudflare Analytics Engine, Vercel Analytics 등)
  await fetch("https://analytics.example.com/api/edge-metrics", {
    method: "POST",
    headers: { "Content-Type": "application/json" },
    body: JSON.stringify(entry),
  });
```

### 3.3 Speculation Rules API - 프리렌더링

Speculation Rules API로 사용자가 방문할 가능성이 높은 페이지를 미리 렌더링합니다.

```typescript
// 동적 Speculation Rules 주입
function injectSpeculationRules(rules: SpeculationRule[]): void {
  const existing = document.querySelector(
    'script[type="speculationrules"]'
  );
  if (existing) existing.remove();

  const script = document.createElement("script");
  script.type = "speculationrules";
  script.textContent = JSON.stringify({ prerender: rules });
  document.head.appendChild(script);
}

interface SpeculationRule {
  source: "list" | "document";
  urls?: string[];
  where?: { href_matches: string };
  eagerness?: "immediate" | "eager" | "moderate" | "conservative";
}

// 사용자 행동 기반 동적 프리렌더링
function setupSmartPrerender(): void {
  // 1단계: 보수적 프리렌더링 (호버/포커스 시)
  injectSpeculationRules([
    {
      source: "document",
      where: { href_matches: "/*" },
      eagerness: "moderate",
    },
  ]);

  // 2단계: 네비게이션 패턴 분석 후 적극적 프리렌더링
  const navHistory: string[] = [];

  const originalPushState = history.pushState.bind(history);
  history.pushState = function (...args) {
    navHistory.push(location.pathname);
    originalPushState(...args);

    // 상위 3개 다음 페이지 예측
    const predictions = predictNextPages(navHistory, 3);
    if (predictions.length > 0) {
      injectSpeculationRules([
        {
          source: "list",
          urls: predictions,
          eagerness: "eager",
        },
      ]);
    }
  };
}

function predictNextPages(
  history: string[],
  topN: number
): string[] {
  const current = history[history.length - 1];
  const transitions = new Map<string, number>();

  for (let i = 0; i < history.length - 1; i++) {
    if (history[i] === current) {
      const next = history[i + 1];
      transitions.set(next, (transitions.get(next) ?? 0) + 1);
    }
  }

  return [...transitions.entries()]
    .sort((a, b) => b[1] - a[1])
    .slice(0, topN)
    .map(([url]) => url);
}
```

```typescript
async function loadRemoteModule<T>(
  remoteName: string,
  modulePath: string
): Promise<T> {
  const startTime = performance.now();

  try {
    const module = await import(
      /* webpackChunkName: "[request]" */
      `${remoteName}/${modulePath}`
    );

    const loadTime = performance.now() - startTime;
    performance.measure(`mf-load-${remoteName}`, {
      start: startTime,
      duration: loadTime,
    });

    if (loadTime > 3000) {
      console.warn(
        `[MF] Slow remote load: ${remoteName} took ${loadTime.toFixed(0)}ms`
      );
    }

    return module as T;
  } catch (err) {
    console.error(`[MF] Failed to load ${remoteName}/${modulePath}`, err);
    throw err;
  }
}

// React 컴포넌트로 래핑
function RemoteComponent({
```

## 6. RUM & 성능 모니터링

### 6.1 자체 RUM (Real User Monitoring) 구축

외부 서비스 의존 없이 자체 RUM을 구축하여 실사용자 성능 데이터를 수집합니다.

```typescript
// lib/rum.ts
import { onCLS, onINP, onLCP, onFCP, onTTFB, type Metric } from "web-vitals";

interface RUMPayload {
  sessionId: string;
  pageUrl: string;
  referrer: string;
  userAgent: string;
  connection?: string;
  deviceMemory?: number;
  vitals: Record<string, number>;
  customTimings: Record<string, number>;
  resources: ResourceEntry[];
  errors: ErrorEntry[];
  timestamp: number;
}

interface ResourceEntry {
  name: string;
  type: string;
  duration: number;
  transferSize: number;
  cacheHit: boolean;
}

interface ErrorEntry {
  message: string;
  source: string;
  line: number;
  column: number;
  timestamp: number;
}

class RUMCollector {
  private payload: RUMPayload;
  private buffer: Metric[] = [];
  private flushTimer: ReturnType<typeof setTimeout> | null = null;
  private endpoint: string;

  constructor(endpoint: string) {
    this.endpoint = endpoint;
    this.payload = this.createInitialPayload();
    this.setupVitals();
    this.setupResourceTiming();
    this.setupErrorTracking();
    this.setupFlush();
  }

  private createInitialPayload(): RUMPayload {
    const nav = navigator as Navigator & {
      connection?: { effectiveType: string };
      deviceMemory?: number;
    };

    return {
      sessionId: this.getSessionId(),
      pageUrl: location.href,
      referrer: document.referrer,
      userAgent: navigator.userAgent,
      connection: nav.connection?.effectiveType,
      deviceMemory: nav.deviceMemory,
      vitals: {},
      customTimings: {},
      resources: [],
      errors: [],
      timestamp: Date.now(),
    };
  }

  private getSessionId(): string {
    let id = sessionStorage.getItem("rum_sid");
    if (!id) {
      id = crypto.randomUUID();
      sessionStorage.setItem("rum_sid", id);
    }
    return id;
  }

  private setupVitals(): void {
    const reportVital = (metric: Metric) => {
      this.payload.vitals[metric.name] = metric.value;
      this.buffer.push(metric);
    };

    onCLS(reportVital);
    onINP(reportVital);
    onLCP(reportVital);
    onFCP(reportVital);
    onTTFB(reportVital);
  }

  private setupResourceTiming(): void {
    const observer = new PerformanceObserver((list) => {
      for (const entry of list.getEntries()) {
        const resource = entry as PerformanceResourceTiming;

        // 주요 리소스만 수집 (API, JS, CSS, 이미지)
        if (
          resource.initiatorType === "xmlhttprequest" ||
          resource.initiatorType === "fetch" ||
          resource.initiatorType === "script" ||
          resource.initiatorType === "link" ||
          resource.initiatorType === "img"
        ) {
          this.payload.resources.push({
            name: resource.name,
            type: resource.initiatorType,
            duration: resource.duration,
            transferSize: resource.transferSize,
            cacheHit: resource.transferSize === 0 && resource.decodedBodySize > 0,
          });
        }
      }
    });

    observer.observe({ type: "resource", buffered: true });
  }

  private setupErrorTracking(): void {
    window.addEventListener("error", (event) => {
      this.payload.errors.push({
        message: event.message,
        source: event.filename,
        line: event.lineno,
        column: event.colno,
        timestamp: Date.now(),
      });
    });

    window.addEventListener("unhandledrejection", (event) => {
      this.payload.errors.push({
        message: String(event.reason),
        source: "unhandledrejection",
        line: 0,
        column: 0,
        timestamp: Date.now(),
      });
    });
  }

  // 커스텀 타이밍 측정
  measure(name: string, fn: () => void): void {
    const start = performance.now();
    fn();
    this.payload.customTimings[name] = performance.now() - start;
  }

  async measureAsync<T>(name: string, fn: () => Promise<T>): Promise<T> {
    const start = performance.now();
    const result = await fn();
    this.payload.customTimings[name] = performance.now() - start;
    return result;
  }

  private setupFlush(): void {
    // visibilitychange에서 최종 전송
    document.addEventListener("visibilitychange", () => {
      if (document.visibilityState === "hidden") {
        this.flush();
      }
    });

    // 30초마다 중간 전송 (장기 세션 대응)
    this.flushTimer = setInterval(() => this.flush(), 30_000);
  }

  private flush(): void {
    if (
      Object.keys(this.payload.vitals).length === 0 &&
      this.payload.resources.length === 0
    ) {
      return;
    }

    const data = JSON.stringify(this.payload);

    // Beacon API (비차단, 페이지 언로드에서도 동작)
    const sent = navigator.sendBeacon(this.endpoint, data);

    if (!sent) {
      // Beacon 실패 시 fetch fallback
      fetch(this.endpoint, {
        method: "POST",
        body: data,
        headers: { "Content-Type": "application/json" },
        keepalive: true,
      }).catch(() => {
        // 최후 수단: localStorage 버퍼링
        const buffered = JSON.parse(
          localStorage.getItem("rum_buffer") ?? "[]"
        );
        buffered.push(this.payload);
        localStorage.setItem(
          "rum_buffer",
          JSON.stringify(buffered.slice(-10))
        );
      });
    }

    // 리소스 데이터 초기화 (vitals는 유지)
    this.payload.resources = [];
    this.payload.errors = [];
    this.payload.customTimings = {};
  }

  destroy(): void {
    if (this.flushTimer) clearInterval(this.flushTimer);
    this.flush();
  }
}

// 싱글톤 인스턴스
let rumInstance: RUMCollector | null = null;

export function initRUM(endpoint: string): RUMCollector {
  if (!rumInstance) {
    rumInstance = new RUMCollector(endpoint);
  }
  return rumInstance;
}

export function getRUM(): RUMCollector {
  if (!rumInstance) throw new Error("RUM not initialized. Call initRUM first.");
  return rumInstance;
}
```

### 6.2 성능 버짓 CI 검증 통합

CI 파이프라인에서 성능 버짓을 자동 검증하고, 위반 시 PR을 블로킹합니다.

```typescript
// scripts/perf-budget-check.ts

interface BudgetRule {
  metric: string;
  threshold: number;
  unit: string;
  severity: "error" | "warning";
}

const PERFORMANCE_BUDGET: BudgetRule[] = [
  { metric: "total-js-size", threshold: 200, unit: "KB (gzip)", severity: "error" },
  { metric: "total-css-size", threshold: 50, unit: "KB (gzip)", severity: "error" },
  { metric: "largest-chunk", threshold: 80, unit: "KB (gzip)", severity: "error" },
  { metric: "lcp", threshold: 2500, unit: "ms", severity: "error" },
  { metric: "inp", threshold: 200, unit: "ms", severity: "error" },
  { metric: "cls", threshold: 0.1, unit: "", severity: "error" },
  { metric: "tbt", threshold: 200, unit: "ms", severity: "warning" },
  { metric: "fcp", threshold: 1800, unit: "ms", severity: "warning" },
  { metric: "unused-js-count", threshold: 3, unit: "items", severity: "warning" },
  { metric: "image-total", threshold: 500, unit: "KB", severity: "warning" },
];

interface CheckResult {
  metric: string;
  value: number;
  threshold: number;
  unit: string;
  severity: "error" | "warning";
  passed: boolean;
}

function checkBudgets(
  measurements: Record<string, number>
): { results: CheckResult[]; passed: boolean } {
  const results: CheckResult[] = [];
  let allPassed = true;

  for (const rule of PERFORMANCE_BUDGET) {
    const value = measurements[rule.metric];
    if (value === undefined) continue;

    const passed = value <= rule.threshold;
    if (!passed && rule.severity === "error") {
      allPassed = false;
    }

    results.push({
      metric: rule.metric,
      value,
      threshold: rule.threshold,
      unit: rule.unit,
      severity: rule.severity,
      passed,
    });
  }

  return { results, passed: allPassed };
}

function formatResults(results: CheckResult[]): string {
  const header = "| Status | Metric | Value | Budget | Severity |";
  const separator = "|--------|--------|-------|--------|----------|";

  const rows = results.map((r) => {
    const status = r.passed ? "PASS" : "FAIL";
    return `| ${status} | ${r.metric} | ${r.value}${r.unit} | ${r.threshold}${r.unit} | ${r.severity} |`;
  });

  return [header, separator, ...rows].join("\n");
}

// CI에서 실행
async function main(): Promise<void> {
  // size-limit, Lighthouse 등에서 수집된 측정값
  const measurements: Record<string, number> = JSON.parse(
    process.env.PERF_MEASUREMENTS ?? "{}"
  );

  const { results, passed } = checkBudgets(measurements);

  console.log("\n=== Performance Budget Check ===\n");
  console.log(formatResults(results));

  const failures = results.filter((r) => !r.passed);
  if (failures.length > 0) {
    console.log(`\nBudget violations: ${failures.length}`);
    const errors = failures.filter((r) => r.severity === "error");
    const warnings = failures.filter((r) => r.severity === "warning");
    if (errors.length > 0) console.log(`  Errors: ${errors.length}`);
    if (warnings.length > 0) console.log(`  Warnings: ${warnings.length}`);
  }

  if (!passed) {
    console.error("\nPerformance budget check FAILED. Fix errors before merging.");
    process.exit(1);
  }

  console.log("\nPerformance budget check PASSED.");
}

main().catch((err) => {
  console.error(err);
  process.exit(1);
});
```
