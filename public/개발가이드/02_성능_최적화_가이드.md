### 1.2 AI 기반 Lighthouse 리포트 자동 분석

Lighthouse CI 결과를 AI에게 전달하여 실행 가능한 개선 제안을 자동 생성합니다.

```typescript
// scripts/lighthouse-ai-analyzer.ts
import Anthropic from "@anthropic-ai/sdk";
import { readFileSync } from "fs";

interface LighthouseAnalysis {
  score: number;
  criticalIssues: Array<{
    metric: string;
    currentValue: string;
    targetValue: string;
    fix: string;
    priority: "P0" | "P1" | "P2";
  }>;
  estimatedImprovement: string;
}

async function analyzeLighthouseReport(
  reportPath: string
): Promise<LighthouseAnalysis> {
  const report = JSON.parse(readFileSync(reportPath, "utf-8"));
  const client = new Anthropic();

  const { audits, categories } = report;
  const perfScore = categories.performance.score * 100;

  const failedAudits = Object.entries(audits)
    .filter(([, audit]: [string, any]) => audit.score !== null && audit.score < 0.9)
    .map(([id, audit]: [string, any]) => ({
      id,
      title: audit.title,
      score: audit.score,
      displayValue: audit.displayValue,
      description: audit.description,
    }));

  const response = await client.messages.create({
    model: "claude-sonnet-4-20250514",
    max_tokens: 4096,
    messages: [
      {
        role: "user",
        content: `Lighthouse 성능 점수: ${perfScore}점

실패한 감사 항목:
${JSON.stringify(failedAudits, null, 2)}

다음 형식으로 분석 결과를 JSON으로 반환해줘:
1. 각 이슈의 우선순위 (P0: 즉시 수정, P1: 이번 스프린트, P2: 백로그)
2. 구체적인 수정 방법 (코드 레벨)
3. 수정 시 예상 점수 개선폭`,
      },
    ],
  });

  const text =
    response.content[0].type === "text" ? response.content[0].text : "";
  return JSON.parse(text);
}

// CI 파이프라인에서 실행
async function main() {
  const analysis = await analyzeLighthouseReport(
    "./lighthouse-results/report.json"
  );

  if (analysis.criticalIssues.some((i) => i.priority === "P0")) {
    console.error("P0 성능 이슈 발견 - 빌드 실패 처리");
    process.exit(1);
  }

  console.log("분석 완료:", JSON.stringify(analysis, null, 2));
}

main();
```

### 1.4 AI 이미지 최적화 파이프라인

빌드 시점에 이미지를 자동으로 분석하고, 최적 포맷 변환과 압축을 수행합니다.

```typescript
// scripts/image-optimizer.ts
import sharp from "sharp";
import { glob } from "glob";
import { stat } from "fs/promises";
import { basename, extname, join } from "path";

interface OptimizationResult {
  originalPath: string;
  originalSize: number;
  outputs: Array<{
    format: string;
    size: number;
    path: string;
  }>;
  savings: string;
}

interface ImageConfig {
  breakpoints: number[];
  formats: Array<"avif" | "webp" | "jpg">;
  quality: { avif: number; webp: number; jpg: number };
}

const config: ImageConfig = {
  breakpoints: [640, 768, 1024, 1280, 1920],
  formats: ["avif", "webp", "jpg"],
  quality: { avif: 50, webp: 75, jpg: 80 },
};

async function optimizeImage(
  inputPath: string
): Promise<OptimizationResult> {
  const originalStat = await stat(inputPath);
  const name = basename(inputPath, extname(inputPath));
  const outputs: OptimizationResult["outputs"] = [];

  for (const width of config.breakpoints) {
    for (const format of config.formats) {
      const outputPath = join(
        "public/optimized",
        `${name}-${width}.${format}`
      );

      const pipeline = sharp(inputPath).resize(width, null, {
        withoutEnlargement: true,
      });

      switch (format) {
        case "avif":
          pipeline.avif({ quality: config.quality.avif, effort: 4 });
          break;
        case "webp":
          pipeline.webp({ quality: config.quality.webp });
          break;
        case "jpg":
          pipeline.jpeg({ quality: config.quality.jpg, progressive: true });
          break;
      }

      const info = await pipeline.toFile(outputPath);
      outputs.push({ format, size: info.size, path: outputPath });
    }
  }

  const totalOriginal = originalStat.size * config.breakpoints.length;
  const totalOptimized = outputs.reduce((sum, o) => sum + o.size, 0);
  const savings = (
    ((totalOriginal - totalOptimized) / totalOriginal) *
    100
  ).toFixed(1);

  return {
    originalPath: inputPath,
    originalSize: originalStat.size,
    outputs,
    savings: `${savings}%`,
  };
}

async function main() {
  const images = await glob("src/assets/images/**/*.{png,jpg,jpeg}");
  console.log(`${images.length}개 이미지 최적화 시작...`);

  const results = await Promise.all(images.map(optimizeImage));

  const totalSaved = results.reduce((sum, r) => {
    const original = r.originalSize;
    const smallest = Math.min(...r.outputs.map((o) => o.size));
    return sum + (original - smallest);
  }, 0);

  console.log(`총 절감: ${(totalSaved / 1024 / 1024).toFixed(2)} MB`);
}

main();
```

## 2. Core Web Vitals 2026

2026년 Core Web Vitals의 핵심 변화는 INP(Interaction to Next Paint)의 안정화, Long Animation Frames API의 표준화, 그리고 Speculation Rules API를 통한 프리렌더링 고도화입니다.

### 2.1 INP 최적화 심화

#### scheduler.yield() - 메인 스레드 양보 표준 API

`scheduler.yield()`는 긴 태스크를 분할하여 브라우저에 제어권을 반환하는 표준 API입니다. `setTimeout(0)` 해킹 대비 우선순위가 보존됩니다.

```typescript
// 긴 작업을 분할하여 INP 개선
async function processLargeDataset<T>(
  items: T[],
  processor: (item: T) => void,
  chunkSize = 50
): Promise<void> {
  for (let i = 0; i < items.length; i += chunkSize) {
    const chunk = items.slice(i, i + chunkSize);
    chunk.forEach(processor);

    // 매 청크 후 브라우저에 제어권 양보
    await scheduler.yield();
  }
}

// 사용 예: 대량 리스트 필터링
async function filterAndRenderProducts(
  products: Product[],
  filters: FilterCriteria
): Promise<void> {
  const results: Product[] = [];

  for (const product of products) {
    if (matchesFilters(product, filters)) {
      results.push(product);
    }

    // 매 100개마다 yield하여 입력 응답성 유지
    if (results.length % 100 === 0) {
      await scheduler.yield();
    }
  }

  renderProducts(results);
}
```

#### Prioritized Task Scheduling API

태스크에 우선순위를 부여하여 중요한 사용자 인터랙션이 백그라운드 작업에 의해 지연되지 않도록 합니다.

```typescript
// 우선순위 기반 태스크 스케줄링
type TaskPriority = "user-blocking" | "user-visible" | "background";

interface ScheduledTask {
  id: string;
  priority: TaskPriority;
  callback: () => void | Promise<void>;
}

class PriorityTaskScheduler {
  private controller = new AbortController();

  async schedule(task: ScheduledTask): Promise<void> {
    return scheduler.postTask(task.callback, {
      priority: task.priority,
      signal: this.controller.signal,
    });
  }

  // 사용자 클릭 -> 즉시 반응 + 백그라운드 분석
  async handleUserInteraction(event: MouseEvent): Promise<void> {
    // P0: 즉시 UI 피드백 (user-blocking)
    await this.schedule({
      id: "ui-feedback",
      priority: "user-blocking",
      callback: () => {
        showLoadingState(event.target as HTMLElement);
      },
    });

    // P1: 데이터 페칭 (user-visible)
    await this.schedule({
      id: "data-fetch",
      priority: "user-visible",
      callback: async () => {
        const data = await fetchProductDetails();
        renderProductDetails(data);
      },
    });

    // P2: 분석 이벤트 전송 (background)
    this.schedule({
      id: "analytics",
      priority: "background",
      callback: () => {
        sendAnalyticsEvent("product_click", { target: event.target });
      },
    });
  }

  abort(): void {
    this.controller.abort();
  }
}
```

### 2.2 Long Animation Frames (LoAF) API

LoAF는 기존 Long Tasks API의 한계(어떤 스크립트가 원인인지 모름)를 해결합니다. 50ms를 초과하는 애니메이션 프레임에 대해 상세한 스크립트 어트리뷰션을 제공합니다.

```typescript
// Long Animation Frames 모니터링
interface LoAFEntry extends PerformanceEntry {
  duration: number;
  renderStart: number;
  styleAndLayoutStart: number;
  firstUIEventTimestamp: number;
  blockingDuration: number;
  scripts: LoAFScriptEntry[];
}

interface LoAFScriptEntry {
  invoker: string;
  invokerType: string;
  sourceURL: string;
  sourceCharPosition: number;
  sourceFunctionName: string;
  duration: number;
  executionStart: number;
  forcedStyleAndLayoutDuration: number;
  pauseDuration: number;
  windowAttribution: string;
}

function observeLongAnimationFrames(): void {
  const observer = new PerformanceObserver((list) => {
    for (const entry of list.getEntries() as LoAFEntry[]) {
      // 100ms 이상만 보고 (50ms 기본 임계값보다 높게 설정)
      if (entry.duration < 100) continue;

      const scripts = entry.scripts.map((s) => ({
        source: `${s.sourceURL}:${s.sourceCharPosition}`,
        function: s.sourceFunctionName,
        duration: Math.round(s.duration),
        layoutForced: s.forcedStyleAndLayoutDuration > 0,
      }));

      console.warn("[LoAF] 긴 프레임 감지:", {
        totalDuration: Math.round(entry.duration),
        blockingDuration: Math.round(entry.blockingDuration),
        scripts,
        hadUserInput: entry.firstUIEventTimestamp > 0,
      });

      // RUM 서버로 전송
      sendToRUM({
        type: "loaf",
        duration: entry.duration,
        blockingDuration: entry.blockingDuration,
        scripts,
        url: location.href,
        timestamp: Date.now(),
      });
    }
  });

  observer.observe({ type: "long-animation-frame", buffered: true });
}

// 개발 환경에서 LoAF 기반 자동 경고
function enableDevLoAFWarnings(): void {
  if (process.env.NODE_ENV !== "development") return;

  const observer = new PerformanceObserver((list) => {
    for (const entry of list.getEntries() as LoAFEntry[]) {
      for (const script of entry.scripts) {
        if (script.forcedStyleAndLayoutDuration > 10) {
          console.error(
            `[Layout Thrashing] ${script.sourceFunctionName} in ${script.sourceURL}\n` +
              `강제 레이아웃 재계산: ${script.forcedStyleAndLayoutDuration}ms\n` +
              `DOM 읽기/쓰기를 분리하세요.`
          );
        }
      }
    }
  });

  observer.observe({ type: "long-animation-frame", buffered: true });
}
```

### 2.3 Speculation Rules API (Prerendering 2.0)

Speculation Rules API는 `<link rel="prefetch">`를 대체하는 선언적 프리렌더링/프리페치 API입니다. 사용자가 링크를 클릭하기 전에 페이지를 미리 렌더링하여 체감 내비게이션 속도를 극적으로 개선합니다.

```typescript
// 동적 Speculation Rules 관리
interface SpeculationRule {
  prerender?: Array<{
    where?: { href_matches: string };
    urls?: string[];
    eagerness?: "immediate" | "eager" | "moderate" | "conservative";
    expects_no_vary_search?: string;
  }>;
  prefetch?: Array<{
    where?: { href_matches: string };
    urls?: string[];
    eagerness?: "immediate" | "eager" | "moderate" | "conservative";
  }>;
}

function injectSpeculationRules(rules: SpeculationRule): void {
  // 기존 speculation rules 제거
  document
    .querySelectorAll('script[type="speculationrules"]')
    .forEach((el) => el.remove());

  const script = document.createElement("script");
  script.type = "speculationrules";
  script.textContent = JSON.stringify(rules);
  document.head.appendChild(script);
}

// 페이지별 최적 프리렌더링 규칙 설정
function setupSpeculationRules(): void {
  const rules: SpeculationRule = {
    prerender: [
      {
        // 같은 도메인 내 네비게이션 링크 - 호버 시 프리렌더
        where: { href_matches: "/*" },
        eagerness: "moderate",
      },
      {
        // 주요 랜딩 페이지 - 즉시 프리렌더
        urls: ["/products", "/pricing", "/docs"],
        eagerness: "eager",
      },
    ],
    prefetch: [
      {
        // 외부 링크는 프리페치만
        where: { href_matches: "https://api.example.com/*" },
        eagerness: "conservative",
      },
    ],
  };

  injectSpeculationRules(rules);
}

// SPA에서 경로 변경 시 규칙 업데이트
function updateRulesForRoute(route: string): void {
  const routeRules: Record<string, SpeculationRule> = {
    "/products": {
      prerender: [
        {
          // 상품 상세 페이지 프리렌더
          where: { href_matches: "/products/*" },
          eagerness: "moderate",
        },
      ],
    },
    "/docs": {
      prerender: [
        {
          where: { href_matches: "/docs/*" },
          eagerness: "conservative",
        },
      ],
    },
  };

  if (routeRules[route]) {
    injectSpeculationRules(routeRules[route]);
  }
}
```

```typescript
// View Transitions 성능 최적화 유틸리티
function performViewTransition(
  updateCallback: () => void | Promise<void>
): void {
  if (!document.startViewTransition) {
    // 미지원 브라우저는 즉시 업데이트
    updateCallback();
    return;
  }

  const transition = document.startViewTransition(async () => {
    await updateCallback();
  });

  // 전환 중 성능 모니터링
  transition.finished.then(() => {
    performance.measure("view-transition", {
      start: performance.now(),
      detail: { type: "navigation" },
    });
  });
}

// React에서의 View Transition 통합
function useViewTransition() {
  const navigate = useNavigate();

  const transitionNavigate = useCallback(
    (to: string) => {
      if (!document.startViewTransition) {
        navigate(to);
        return;
      }

      document.startViewTransition(() => {
        navigate(to);
      });
    },
    [navigate]
  );

  return { navigate: transitionNavigate };
}
```

### 3.4 Server Actions + useOptimistic 패턴

서버에 요청을 보내는 동안 UI를 즉시 업데이트하여 체감 속도를 개선합니다.

```typescript
// app/components/TodoList.tsx
"use client";

import { useOptimistic, useTransition } from "react";
import { addTodo, toggleTodo, deleteTodo } from "@/app/actions/todos";

interface Todo {
  id: string;
  text: string;
  completed: boolean;
}

type OptimisticAction =
  | { type: "add"; todo: Todo }
  | { type: "toggle"; id: string }
  | { type: "delete"; id: string };

export function TodoList({ todos }: { todos: Todo[] }) {
  const [isPending, startTransition] = useTransition();

  const [optimisticTodos, dispatchOptimistic] = useOptimistic(
    todos,
    (state: Todo[], action: OptimisticAction) => {
      switch (action.type) {
        case "add":
          return [...state, action.todo];
        case "toggle":
          return state.map((t) =>
            t.id === action.id ? { ...t, completed: !t.completed } : t
          );
        case "delete":
          return state.filter((t) => t.id !== action.id);
      }
    }
  );

  async function handleAdd(formData: FormData) {
    const text = formData.get("text") as string;
    const tempId = `temp-${Date.now()}`;

    startTransition(async () => {
      // 즉시 UI 업데이트 (낙관적)
      dispatchOptimistic({
        type: "add",
        todo: { id: tempId, text, completed: false },
      });

      // 서버에 실제 저장
      await addTodo(text);
    });
  }

  async function handleToggle(id: string) {
    startTransition(async () => {
      dispatchOptimistic({ type: "toggle", id });
      await toggleTodo(id);
    });
  }

  async function handleDelete(id: string) {
    startTransition(async () => {
      dispatchOptimistic({ type: "delete", id });
      await deleteTodo(id);
    });
  }

  return (
    <div>
      <form action={handleAdd}>
        <input name="text" placeholder="할 일 입력..." required />
        <button type="submit">추가</button>
      </form>

      <ul>
        {optimisticTodos.map((todo) => (
          <li key={todo.id} style={{ opacity: todo.id.startsWith("temp-") ? 0.6 : 1 }}>
            <input
              type="checkbox"
              checked={todo.completed}
              onChange={() => handleToggle(todo.id)}
            />
            <span style={{ textDecoration: todo.completed ? "line-through" : "none" }}>
              {todo.text}
            </span>
            <button onClick={() => handleDelete(todo.id)}>삭제</button>
          </li>
        ))}
      </ul>
    </div>
  );
}
```

```typescript
        timerRef.current = setTimeout(() => {
          prefetchModule(moduleLoader, key);
        }, 100); // 100ms 후 프리페치 (의도적 호버 판별)
      },
      onMouseLeave: () => {
        if (timerRef.current) clearTimeout(timerRef.current);
      },
    }),
    [moduleLoader, key]
  );

  return handlers;
}

// 뷰포트 진입 시 프리페치
export function usePrefetchOnViewport(
  moduleLoader: () => Promise<any>,
  key: string,
  threshold = 0.1
) {
  const ref = useRef<HTMLElement>(null);

  useEffect(() => {
    const element = ref.current;
    if (!element) return;

    const observer = new IntersectionObserver(
      (entries) => {
        if (entries[0].isIntersecting) {
          prefetchModule(moduleLoader, key);
          observer.disconnect();
        }
      },
      { threshold, rootMargin: "200px" }
    );

    observer.observe(element);
    return () => observer.disconnect();
  }, [moduleLoader, key, threshold]);

  return ref;
}

// 브라우저 유휴 시간에 프리페치
export function prefetchOnIdle(
  modules: Array<{ loader: () => Promise<any>; key: string }>
): void {
  if ("requestIdleCallback" in window) {
    requestIdleCallback(
      (deadline) => {
        for (const { loader, key } of modules) {
          if (deadline.timeRemaining() < 10) break;
          prefetchModule(loader, key);
        }
      },
      { timeout: 5000 }
    );
  }
}
```

```typescript
export function middleware(request: NextRequest) {

  );

    response.headers.set("x-render-mode", "full-ssr");
    response.headers.set(

    // 일반 사용자에게는 PPR 셸 + 스트리밍
    response.headers.set("x-render-mode", "ppr-streaming");
  }

  response.headers.set("X-Content-Type-Options", "nosniff");
  response.headers.set("X-Frame-Options", "DENY");
  response.headers.set("Referrer-Policy", "strict-origin-when-cross-origin");

  return response;
}

function getABGroup(request: NextRequest): string {
  const existing = request.cookies.get("ab-group")?.value;
  if (existing) return existing;

  // 일관된 분배를 위해 IP 기반 해시
  const ip = request.headers.get("x-forwarded-for") ?? "unknown";
  const hash = simpleHash(ip);
  return hash % 2 === 0 ? "control" : "variant";
}

function simpleHash(str: string): number {
  let hash = 0;
  for (let i = 0; i < str.length; i++) {
    hash = (hash << 5) - hash + str.charCodeAt(i);
    hash |= 0;
  }
  return Math.abs(hash);
}
```

```typescript
interface CDNConfig {
  name: string;
  baseUrl: string;
  priority: number;
  regions: string[]; // 강점 지역
  healthCheckUrl: string;
}

const cdnConfigs: CDNConfig[] = [
  {
    name: "cloudfront",
    baseUrl: "https://d1234.cloudfront.net",
    priority: 1,
    regions: ["ap-northeast-2", "ap-northeast-1", "us-east-1"],
    healthCheckUrl: "https://d1234.cloudfront.net/health.txt",
  },
  {
    name: "cloudflare",
    baseUrl: "https://assets.example.com",

  },

class MultiCDNRouter {
  private healthStatus = new Map<string, boolean>();
  private latencyMap = new Map<string, number>();

  async initialize(): Promise<void> {
    // 각 CDN 헬스체크 및 레이턴시 측정
    await Promise.all(
      cdnConfigs.map(async (cdn) => {
        try {
          const start = performance.now();
          const res = await fetch(cdn.healthCheckUrl, {
            method: "HEAD",
            signal: AbortSignal.timeout(3000),
          });
          const latency = performance.now() - start;

          this.healthStatus.set(cdn.name, res.ok);
          this.latencyMap.set(cdn.name, latency);
        } catch {
          this.healthStatus.set(cdn.name, false);
          this.latencyMap.set(cdn.name, Infinity);
        }
      })
    );
  }

  getOptimalCDN(): CDNConfig {
    const healthy = cdnConfigs.filter(
      (cdn) => this.healthStatus.get(cdn.name) === true
    );

    if (healthy.length === 0) {
      // 모든 CDN 다운 시 origin 직접 사용
      console.warn("모든 CDN 장애 감지 - origin fallback");
      return {
        name: "origin",
        baseUrl: process.env.NEXT_PUBLIC_API_ORIGIN ?? "",
        priority: 99,
        regions: [],
        healthCheckUrl: "",
      };
    }

    // 레이턴시 기준 정렬
    return healthy.sort(
      (a, b) =>
        (this.latencyMap.get(a.name) ?? Infinity) -
        (this.latencyMap.get(b.name) ?? Infinity)
    )[0];
  }

  buildUrl(path: string): string {
    const cdn = this.getOptimalCDN();
    return `${cdn.baseUrl}${path}`;
  }
}
```

```typescript
interface WarmingConfig {
  urls: string[];
  regions: string[];
  concurrency: number;
}

async function warmCDNCache(config: WarmingConfig): Promise<void> {
  const { urls, concurrency } = config;
  const results: Array<{ url: string; status: number; duration: number }> = [];

  // 동시 요청 제한을 위한 세마포어
  let running = 0;
  const queue = [...urls];

  async function processUrl(url: string): Promise<void> {
    const start = Date.now();
    try {
      const res = await fetch(url, {
        headers: {
          "User-Agent": "CacheWarmer/1.0",
          "X-Cache-Warm": "true",
        },
      });
      results.push({
        url,
        status: res.status,
        duration: Date.now() - start,
      });
    } catch (error) {
      results.push({ url, status: 0, duration: Date.now() - start });
    }
  }

  // 동시성 제어
  const workers = Array.from({ length: concurrency }, async () => {
    while (queue.length > 0) {
      const url = queue.shift();
      if (url) await processUrl(url);
    }
  });

  await Promise.all(workers);

  // 결과 요약
  const succeeded = results.filter((r) => r.status === 200);
  const avgDuration = succeeded.reduce((s, r) => s + r.duration, 0) / succeeded.length;

  console.log(`캐시 워밍 완료: ${succeeded.length}/${results.length} 성공`);
  console.log(`평균 응답 시간: ${Math.round(avgDuration)}ms`);
}
```

### 6.1 자체 RUM 수집기 구현 (< 3KB)

외부 서비스 의존 없이 핵심 성능 지표를 수집하는 경량 RUM 라이브러리입니다.

```typescript
// lib/rum-collector.ts
interface RUMMetrics {
  // Core Web Vitals
  lcp: number | null;
  inp: number | null;
  cls: number | null;
  // 추가 지표
  fcp: number | null;
  ttfb: number | null;
  // 메타데이터
  url: string;
  userAgent: string;
  connectionType: string;
  deviceMemory: number;
  timestamp: number;
}

interface RUMConfig {
  endpoint: string;
  sampleRate: number; // 0-1, 예: 0.1 = 10% 샘플링
  debug: boolean;
}

class RUMCollector {
  private metrics: Partial<RUMMetrics> = {};
  private config: RUMConfig;
  private observers: PerformanceObserver[] = [];

  constructor(config: RUMConfig) {
    this.config = config;

    // 샘플링 체크
    if (Math.random() > config.sampleRate) return;

    this.collectNavigationTiming();
    this.observeLCP();
    this.observeINP();
    this.observeCLS();
    this.observeFCP();

    // 페이지 이탈 시 전송
    this.setupBeaconSend();
  }

  private collectNavigationTiming(): void {
    const nav = performance.getEntriesByType(
      "navigation"
    )[0] as PerformanceNavigationTiming;
    if (nav) {
      this.metrics.ttfb = nav.responseStart - nav.requestStart;
    }
  }

  private observeLCP(): void {
    const observer = new PerformanceObserver((list) => {
      const entries = list.getEntries();
      const last = entries[entries.length - 1];
      this.metrics.lcp = last.startTime;
    });
    observer.observe({ type: "largest-contentful-paint", buffered: true });
    this.observers.push(observer);
  }

  private observeINP(): void {
    let maxDuration = 0;

    const observer = new PerformanceObserver((list) => {
      for (const entry of list.getEntries()) {
        const duration = (entry as any).duration;
        if (duration > maxDuration) {
          maxDuration = duration;
          this.metrics.inp = duration;
        }
      }
    });
    observer.observe({ type: "event", buffered: true, durationThreshold: 16 });
    this.observers.push(observer);
  }

  private observeCLS(): void {
    let clsValue = 0;
    let sessionValue = 0;
    let sessionEntries: PerformanceEntry[] = [];

    const observer = new PerformanceObserver((list) => {
      for (const entry of list.getEntries() as any[]) {
        if (!entry.hadRecentInput) {
          const firstEntry = sessionEntries[0];
          const lastEntry = sessionEntries[sessionEntries.length - 1];

          if (
            sessionValue &&
            entry.startTime - lastEntry.startTime < 1000 &&
            entry.startTime - firstEntry.startTime < 5000
          ) {
            sessionValue += entry.value;
            sessionEntries.push(entry);
          } else {
            sessionValue = entry.value;
            sessionEntries = [entry];
          }

          if (sessionValue > clsValue) {
            clsValue = sessionValue;
            this.metrics.cls = clsValue;
          }
        }
      }
    });
    observer.observe({ type: "layout-shift", buffered: true });
    this.observers.push(observer);
  }

  private observeFCP(): void {
    const observer = new PerformanceObserver((list) => {
      for (const entry of list.getEntries()) {
        if (entry.name === "first-contentful-paint") {
          this.metrics.fcp = entry.startTime;
        }
      }
    });
    observer.observe({ type: "paint", buffered: true });
    this.observers.push(observer);
  }

  private setupBeaconSend(): void {
    const sendMetrics = () => {
      const payload: RUMMetrics = {
        lcp: this.metrics.lcp ?? null,
        inp: this.metrics.inp ?? null,
        cls: this.metrics.cls ?? null,
        fcp: this.metrics.fcp ?? null,
        ttfb: this.metrics.ttfb ?? null,
        url: location.href,
        userAgent: navigator.userAgent,
        connectionType:
          (navigator as any).connection?.effectiveType ?? "unknown",
        deviceMemory: (navigator as any).deviceMemory ?? 0,
        timestamp: Date.now(),
      };

      if (this.config.debug) {
        console.log("[RUM]", payload);
      }

      // Beacon API로 페이지 이탈 시에도 확실히 전송
      const blob = new Blob([JSON.stringify(payload)], {
        type: "application/json",
      });
      navigator.sendBeacon(this.config.endpoint, blob);
    };

    // visibilitychange가 pagehide보다 안정적
    document.addEventListener("visibilitychange", () => {
      if (document.visibilityState === "hidden") {
        sendMetrics();
      }
    });

    // Safari iOS 대응
    window.addEventListener("pagehide", sendMetrics);
  }

  destroy(): void {
    this.observers.forEach((o) => o.disconnect());
    this.observers = [];
  }
}

// 초기화
export function initRUM(config?: Partial<RUMConfig>): RUMCollector {
  return new RUMCollector({
    endpoint: "/api/rum",
    sampleRate: 0.1,
    debug: process.env.NODE_ENV === "development",
    ...config,
  });
}
```

```typescript
import { readFileSync, readdirSync, statSync } from "fs";
import { join } from "path";
import { perfBudget } from "../perf-budget.config";

interface BudgetViolation {
  category: string;
  metric: string;
  limit: number;
  actual: number;
  severity: "error" | "warning";
}

function checkBundleBudget(buildDir: string): BudgetViolation[] {
  const violations: BudgetViolation[] = [];
  const jsFiles = findFiles(buildDir, ".js");

  const totalSize = jsFiles.reduce(
    (sum, f) => sum + statSync(f).size / 1024,
    0
  );

  if (totalSize > perfBudget.bundles.maxTotalSize) {
    violations.push({
      category: "bundle",
      metric: "totalSize",
      limit: perfBudget.bundles.maxTotalSize,
      actual: Math.round(totalSize),
      severity: "error",
    });
  }

  for (const file of jsFiles) {
    const size = statSync(file).size / 1024;
    if (size > perfBudget.bundles.maxChunkSize) {
      violations.push({
        category: "bundle",
        metric: `chunkSize (${file})`,
        limit: perfBudget.bundles.maxChunkSize,
        actual: Math.round(size),
        severity: "error",
      });
    }
  }

  return violations;
}

function findFiles(dir: string, ext: string): string[] {
  const results: string[] = [];

  for (const entry of readdirSync(dir, { withFileTypes: true })) {
    const fullPath = join(dir, entry.name);
    if (entry.isDirectory()) {
      results.push(...findFiles(fullPath, ext));
    } else if (entry.name.endsWith(ext)) {
      results.push(fullPath);
    }
  }

  return results;
}

      );
    }

    const hasErrors = violations.some((v) => v.severity === "error");
```

### 6.3 PerformanceObserver 통합 모니터링

여러 성능 지표를 하나의 통합 모니터링 시스템으로 수집합니다.

```typescript
// lib/perf-monitor.ts
type MetricName = "LCP" | "INP" | "CLS" | "FCP" | "TTFB" | "LoAF" | "Resource";

interface PerfEvent {
  metric: MetricName;
  value: number;
  rating: "good" | "needs-improvement" | "poor";
  attribution?: Record<string, unknown>;
  timestamp: number;
}

type PerfEventHandler = (event: PerfEvent) => void;

const thresholds: Record<string, [number, number]> = {
  LCP: [2500, 4000],
  INP: [200, 500],
  CLS: [0.1, 0.25],
  FCP: [1800, 3000],
  TTFB: [800, 1800],
};

function getRating(
  metric: string,
  value: number
): "good" | "needs-improvement" | "poor" {
  const [good, poor] = thresholds[metric] ?? [Infinity, Infinity];
  if (value <= good) return "good";
  if (value <= poor) return "needs-improvement";
  return "poor";
}

class PerformanceMonitor {
  private handlers: PerfEventHandler[] = [];
  private observers: PerformanceObserver[] = [];

  onMetric(handler: PerfEventHandler): () => void {
    this.handlers.push(handler);
    return () => {
      this.handlers = this.handlers.filter((h) => h !== handler);
    };
  }

  private emit(event: PerfEvent): void {
    this.handlers.forEach((h) => h(event));
  }

  start(): void {
    // LCP
    this.observe("largest-contentful-paint", (entry) => {
      this.emit({
        metric: "LCP",
        value: entry.startTime,
        rating: getRating("LCP", entry.startTime),
        attribution: {
          element: (entry as any).element?.tagName,
          url: (entry as any).url,
          size: (entry as any).size,
        },
        timestamp: Date.now(),
      });
    });

    // INP (event timing)
    this.observe(
      "event",
      (entry) => {
        const duration = (entry as any).duration;
        if (duration > 0) {
          this.emit({
            metric: "INP",
            value: duration,
            rating: getRating("INP", duration),
            attribution: {
              type: entry.name,
              target: (entry as any).target?.tagName,
              processingStart: (entry as any).processingStart,
              processingEnd: (entry as any).processingEnd,
            },
            timestamp: Date.now(),
          });
        }
      },
      { durationThreshold: 16 }
    );

    // CLS
    this.observe("layout-shift", (entry) => {
      if (!(entry as any).hadRecentInput) {
        this.emit({
          metric: "CLS",
          value: (entry as any).value,
          rating: getRating("CLS", (entry as any).value),
          attribution: {
            sources: (entry as any).sources?.map((s: any) => ({
              node: s.node?.tagName,
              previousRect: s.previousRect,
              currentRect: s.currentRect,
            })),
          },
          timestamp: Date.now(),
        });
      }
    });

    // Long Animation Frames
    this.observe("long-animation-frame", (entry) => {
      this.emit({
        metric: "LoAF",
        value: entry.duration,
        rating: entry.duration > 100 ? "poor" : "needs-improvement",
        attribution: {
          scripts: (entry as any).scripts?.map((s: any) => ({
            sourceURL: s.sourceURL,
            functionName: s.sourceFunctionName,
            duration: s.duration,
          })),
          blockingDuration: (entry as any).blockingDuration,
        },
        timestamp: Date.now(),
      });
    });

    // 리소스 로딩
    this.observe("resource", (entry) => {
      const resource = entry as PerformanceResourceTiming;
      // 1초 이상 걸리는 리소스만 보고
      if (resource.duration > 1000) {
        this.emit({
          metric: "Resource",
          value: resource.duration,
          rating: "poor",
          attribution: {
            name: resource.name,
            type: resource.initiatorType,
            transferSize: resource.transferSize,
            protocol: resource.nextHopProtocol,
          },
          timestamp: Date.now(),
        });
      }
    });
  }

  private observe(
    type: string,
    callback: (entry: PerformanceEntry) => void,
    options?: Record<string, any>
  ): void {
    try {
      const observer = new PerformanceObserver((list) => {
        list.getEntries().forEach(callback);
      });
      observer.observe({ type, buffered: true, ...options });
      this.observers.push(observer);
    } catch {
      // 미지원 타입은 무시
    }
  }

  stop(): void {
    this.observers.forEach((o) => o.disconnect());
    this.observers = [];
  }
}

// 사용 예시
export function setupMonitoring(): PerformanceMonitor {
  const monitor = new PerformanceMonitor();

  monitor.onMetric((event) => {
    // 콘솔 로깅 (개발 환경)
    if (process.env.NODE_ENV === "development") {
      const emoji =
        event.rating === "good"
          ? "[OK]"
          : event.rating === "needs-improvement"
            ? "[WARN]"
            : "[BAD]";
      console.log(
        `${emoji} ${event.metric}: ${typeof event.value === "number" ? event.value.toFixed(2) : event.value}`
      );
    }

    // RUM 서버 전송 (프로덕션)
    if (process.env.NODE_ENV === "production" && event.rating === "poor") {
      navigator.sendBeacon(
        "/api/rum",
        JSON.stringify(event)
      );
    }
  });

  monitor.start();
  return monitor;
}
```
