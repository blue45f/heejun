```typescript
import {
  useRef,
  useState,
  useTransition,
  useDeferredValue,
  useCallback,
} from "react";

interface VirtualListProps<T> {
  items: T[];
  renderItem: (item: T, index: number) => React.ReactNode;
  estimateSize?: number;
  overscan?: number;
  searchable?: boolean;
  filterFn?: (item: T, query: string) => boolean;
}

export function OptimizedVirtualList<T>({
  items,
  renderItem,
  estimateSize = 60,
  overscan = 5,
  searchable = false,
  filterFn,
}: VirtualListProps<T>) {
  const parentRef = useRef<HTMLDivElement>(null);
  const [searchQuery, setSearchQuery] = useState("");
  const [, startTransition] = useTransition();
  const deferredQuery = useDeferredValue(searchQuery);

  const filteredItems =
    searchable && filterFn && deferredQuery
      ? items.filter((item) => filterFn(item, deferredQuery))
      : items;

  const virtualizer = useVirtualizer({
    count: filteredItems.length,
    getScrollElement: () => parentRef.current,
    estimateSize: () => estimateSize,
    overscan,
    measureElement: (el) => el.getBoundingClientRect().height,
  });

  const handleSearch = useCallback((value: string) => {
    startTransition(() => {
      setSearchQuery(value);
    });
  }, []);

  return (
    <div>
      {searchable && (
        <input
          type="search"
          placeholder="검색..."
          onChange={(e) => handleSearch(e.target.value)}
        />
      )}
      <div
        ref={parentRef}
        style={{ height: "600px", overflow: "auto", contain: "strict" }}
      >
        <div
          style={{
            height: `${virtualizer.getTotalSize()}px`,
            width: "100%",
            position: "relative",
          }}
        >
          {virtualizer.getVirtualItems().map((virtualItem) => (
            <div
              key={virtualItem.key}
              data-index={virtualItem.index}
              ref={virtualizer.measureElement}
              style={{
                position: "absolute",
                top: 0,
                left: 0,
                width: "100%",
                transform: `translateY(${virtualItem.start}px)`,
              }}
            >
              {renderItem(filteredItems[virtualItem.index], virtualItem.index)}
            </div>
          ))}
        </div>
      </div>
    </div>
  );
}
```

```typescript
import { useState, useRef, useEffect, useCallback } from "react";

interface OptimizedImageProps {
  src: string;
  alt: string;
  width: number;
  height: number;
  blurHash?: string;
  priority?: boolean;
  sizes?: string;
  srcSet?: string;
}

export function OptimizedImage({
  src,
  alt,
  width,
  height,
  blurHash,
  priority = false,
  sizes = "(max-width: 768px) 100vw, (max-width: 1200px) 50vw, 33vw",
  srcSet,
}: OptimizedImageProps) {
  const [loaded, setLoaded] = useState(false);
  const [inView, setInView] = useState(priority);
  const imgRef = useRef<HTMLDivElement>(null);

  useEffect(() => {
    if (priority || !imgRef.current) return;

    const observer = new IntersectionObserver(
      ([entry]) => {
        if (entry.isIntersecting) {
          setInView(true);
          observer.disconnect();
        }
      },
      { rootMargin: "200px" }
    );
    observer.observe(imgRef.current);
    return () => observer.disconnect();
  }, [priority]);

  const aspectRatio = `${width} / ${height}`;

  return (
    <div
      ref={imgRef}
      style={{
        aspectRatio,
        width: "100%",
        position: "relative",
        overflow: "hidden",
        backgroundColor: "#f0f0f0",
        contain: "layout style paint",
      }}
    >
      {blurHash && !loaded && (
        <canvas
          style={{
            position: "absolute",
            inset: 0,
            width: "100%",
            height: "100%",
            filter: "blur(20px)",
            transform: "scale(1.1)",
          }}
        />
      )}
      {inView && (
        <img
          src={src}
          alt={alt}
          width={width}
          height={height}
          sizes={sizes}
          srcSet={srcSet}
          loading={priority ? "eager" : "lazy"}
          decoding={priority ? "sync" : "async"}
          fetchPriority={priority ? "high" : "auto"}
          onLoad={() => setLoaded(true)}
          style={{
            width: "100%",
            height: "100%",
            objectFit: "cover",
            opacity: loaded ? 1 : 0,
            transition: "opacity 0.3s ease",
          }}
        />
      )}
    </div>
  );
}
```

```typescript
// utils/native-lodash.ts - lodash 네이티브 대체

export function debounce<T extends (...args: unknown[]) => unknown>(
  fn: T,
  ms: number,
  options?: { leading?: boolean; trailing?: boolean }
): T & { cancel: () => void; flush: () => void } {
  const { leading = false, trailing = true } = options ?? {};
  let timeoutId: ReturnType<typeof setTimeout> | null = null;
  let lastArgs: Parameters<T> | null = null;
  let lastResult: ReturnType<T>;

  const debounced = function (this: unknown, ...args: Parameters<T>) {
    lastArgs = args;
    const invokeLeading = leading && timeoutId === null;

    if (timeoutId !== null) clearTimeout(timeoutId);

    timeoutId = setTimeout(() => {
      timeoutId = null;
      if (trailing && lastArgs) {
        lastResult = fn.apply(this, lastArgs) as ReturnType<T>;
        lastArgs = null;
      }
    }, ms);

    if (invokeLeading) {
      lastResult = fn.apply(this, args) as ReturnType<T>;
    }

    return lastResult;
  } as T & { cancel: () => void; flush: () => void };

  debounced.cancel = () => {
    if (timeoutId !== null) clearTimeout(timeoutId);
    timeoutId = null;
    lastArgs = null;
  };

  debounced.flush = () => {
    if (timeoutId !== null && lastArgs) {
      clearTimeout(timeoutId);
      timeoutId = null;
      lastResult = fn.apply(null, lastArgs) as ReturnType<T>;
      lastArgs = null;
    }
  };

  return debounced;
}

export function throttle<T extends (...args: unknown[]) => unknown>(
  fn: T,
  ms: number
): T & { cancel: () => void } {
  let lastCall = 0;
  let timeoutId: ReturnType<typeof setTimeout> | null = null;

  const throttled = function (this: unknown, ...args: Parameters<T>) {
    const now = Date.now();
    const remaining = ms - (now - lastCall);

    if (remaining <= 0) {
      if (timeoutId !== null) {
        clearTimeout(timeoutId);
        timeoutId = null;
      }
      lastCall = now;
      return fn.apply(this, args);
    }

    if (timeoutId === null) {
      timeoutId = setTimeout(() => {
        lastCall = Date.now();
        timeoutId = null;
        fn.apply(this, args);
      }, remaining);
    }
  } as T & { cancel: () => void };

  throttled.cancel = () => {
    if (timeoutId !== null) clearTimeout(timeoutId);
    timeoutId = null;
    lastCall = 0;
  };

  return throttled;
}

export function groupBy<T>(
  arr: T[],
  keyFn: ((item: T) => string) | keyof T
): Record<string, T[]> {
  const getKey = typeof keyFn === "function"
    ? keyFn
    : (item: T) => String(item[keyFn]);

  return arr.reduce<Record<string, T[]>>((acc, item) => {
    const key = getKey(item);
    (acc[key] ??= []).push(item);
    return acc;
  }, {});
}

export function pick<T extends Record<string, unknown>, K extends keyof T>(
  obj: T,
  keys: K[]
): Pick<T, K> {
  const result = {} as Pick<T, K>;
  for (const key of keys) {
    if (key in obj) result[key] = obj[key];
  }
  return result;
}

export function omit<T extends Record<string, unknown>, K extends keyof T>(
  obj: T,
  keys: K[]
): Omit<T, K> {
  const keySet = new Set<string | number | symbol>(keys);
  return Object.fromEntries(
    Object.entries(obj).filter(([k]) => !keySet.has(k))
  ) as Omit<T, K>;
}

export function deepMerge<T extends Record<string, unknown>>(
  target: T,
  ...sources: Partial<T>[]
): T {
  const result = { ...target };
  for (const source of sources) {
    for (const key of Object.keys(source) as (keyof T)[]) {
      const sourceVal = source[key];
      const targetVal = result[key];
      if (
        sourceVal &&
        typeof sourceVal === "object" &&
        !Array.isArray(sourceVal) &&
        targetVal &&
        typeof targetVal === "object" &&
        !Array.isArray(targetVal)
      ) {
        (result as Record<string, unknown>)[key as string] = deepMerge(
          targetVal as Record<string, unknown>,
          sourceVal as Record<string, unknown>
        );
      } else {
        (result as Record<string, unknown>)[key as string] = sourceVal;
      }
    }
  }
  return result;
}
```

## 2. Navigation API + Speculation Rules API

### 2.1 Navigation API: SPA 네비게이션 현대화

Navigation API는 기존 `popstate` + `History.pushState` 조합을 완전히 대체하는 새로운 표준입니다. View Transitions API와 결합하면 네이티브 앱 수준의 페이지 전환을 구현할 수 있습니다.

```typescript
// navigation-manager.ts
interface NavigationConfig {
  onNavigate?: (url: URL) => void;
  transitions?: boolean;
  scrollRestoration?: "manual" | "auto";
  prefetchOnHover?: boolean;
}

class NavigationManager {
  private abortControllers = new Map<string, AbortController>();
  private prefetchCache = new Map<string, Promise<Response>>();

  constructor(private config: NavigationConfig = {}) {
    this.init();
  }

  private init() {
    if (!("navigation" in window)) {
      console.warn("Navigation API not supported, falling back to History API");
      return;
    }

    navigation.addEventListener("navigate", (event: NavigateEvent) => {
      // 외부 링크, 다운로드, form submission(GET 제외)은 무시
      if (!event.canIntercept || event.hashChange) return;

      const url = new URL(event.destination.url);

      // 같은 origin만 처리
      if (url.origin !== location.origin) return;

      event.intercept({
        scroll: this.config.scrollRestoration === "manual" ? "manual" : "after-transition",
        focusReset: "after-transition",
        handler: async () => {
          // 이전 네비게이션 취소
          this.abortPending(url.pathname);

          const controller = new AbortController();
          this.abortControllers.set(url.pathname, controller);

          try {
            // View Transition 적용
            if (this.config.transitions && document.startViewTransition) {
              const transition = document.startViewTransition(async () => {
                await this.loadPage(url, controller.signal);
                this.config.onNavigate?.(url);
              });
              await transition.finished;
            } else {
              await this.loadPage(url, controller.signal);
              this.config.onNavigate?.(url);
            }
          } finally {
            this.abortControllers.delete(url.pathname);
          }
        },
      });
    });

    // 호버 시 프리페치
    if (this.config.prefetchOnHover) {
      this.setupHoverPrefetch();
    }
  }

  private async loadPage(url: URL, signal: AbortSignal): Promise<void> {
    // 프리페치 캐시 확인
    const cached = this.prefetchCache.get(url.pathname);
    const response = cached ?? fetch(url.toString(), {
      signal,
      headers: { "X-Navigation": "true" },
    });

    const res = await response;
    if (!res.ok) throw new Error(`Navigation failed: ${res.status}`);

    const html = await res.text();
    const parser = new DOMParser();
    const doc = parser.parseFromString(html, "text/html");

    // main 콘텐츠만 교체
    const newMain = doc.querySelector("main");
    const currentMain = document.querySelector("main");
    if (newMain && currentMain) {
      currentMain.replaceWith(newMain);
    }

    // 제목 업데이트
    document.title = doc.title;
  }

  private setupHoverPrefetch() {
    document.addEventListener("pointerenter", (e) => {
      const link = (e.target as Element).closest("a[href]");
      if (!link) return;

      const href = link.getAttribute("href");
      if (!href || href.startsWith("http")) return;

      if (!this.prefetchCache.has(href)) {
        this.prefetchCache.set(href, fetch(href, { priority: "low" }));

        // 30초 후 캐시 정리
        setTimeout(() => this.prefetchCache.delete(href), 30_000);
      }
    }, { capture: true, passive: true });
  }

  private abortPending(pathname: string) {
    for (const [path, controller] of this.abortControllers) {
      if (path !== pathname) {
        controller.abort();
        this.abortControllers.delete(path);
      }
    }
  }

  // 프로그래매틱 네비게이션
  navigate(url: string, options?: { state?: unknown; replace?: boolean }) {
    if ("navigation" in window) {
      navigation.navigate(url, {
        state: options?.state,
        history: options?.replace ? "replace" : "push",
      });
    } else {
      if (options?.replace) {
        history.replaceState(options?.state, "", url);
      } else {
        history.pushState(options?.state, "", url);
      }
      window.dispatchEvent(new PopStateEvent("popstate"));
    }
  }
}

export const navManager = new NavigationManager({
  transitions: true,
  scrollRestoration: "auto",
  prefetchOnHover: true,
});
```

Speculation Rules API는 브라우저에게 "다음에 사용자가 방문할 가능성이 높은 페이지"를 알려줘서 미리 prefetch/prerender하게 만드는 선언적 API입니다.

```typescript
// speculation-rules.ts
interface SpeculationRule {
  source?: "list" | "document";
  urls?: string[];
  where?: {
    and?: SpeculationCondition[];
    or?: SpeculationCondition[];
    not?: SpeculationCondition;
    href_matches?: string;
    selector_matches?: string;
  };
  eagerness?: "immediate" | "eager" | "moderate" | "conservative";
  expects_no_vary_search?: string;
  referrer_policy?: string;
  requires?: string[];
}

interface SpeculationCondition {
  href_matches?: string;
  selector_matches?: string;
  not?: SpeculationCondition;
}

type SpeculationAction = "prefetch" | "prerender";

// 동적 Speculation Rules 관리자
class SpeculationRulesManager {
  private scriptElement: HTMLScriptElement | null = null;
  private rules: Record<SpeculationAction, SpeculationRule[]> = {
    prefetch: [],
    prerender: [],
  };

  constructor() {
    this.applyDefaults();
  }

  private applyDefaults() {
    // 기본 규칙: 같은 사이트 링크를 conservative하게 prefetch
    this.addRule("prefetch", {
      source: "document",
      where: {
        and: [
          { href_matches: "/*" },
          { not: { href_matches: "/api/*" } },
          { not: { href_matches: "/auth/*" } },
          { not: { selector_matches: "[data-no-prefetch]" } },
        ],
      },
      eagerness: "moderate",
    });

    // 네비게이션 영역 링크는 eager하게 prerender
    this.addRule("prerender", {
      source: "document",
      where: {
        selector_matches: "nav a[href], [data-prerender] a[href]",
      },
      eagerness: "moderate",
    });
  }

  addRule(action: SpeculationAction, rule: SpeculationRule) {
    this.rules[action].push(rule);
    this.inject();
  }

  // 특정 URL 목록을 즉시 prerender
  prerenderUrls(urls: string[]) {
    this.addRule("prerender", {
      source: "list",
      urls,
      eagerness: "immediate",
    });
  }

  // 사용자 행동 기반 동적 추가 (예: 검색 결과 상위 3개)
  addPredictive(urls: string[], action: SpeculationAction = "prefetch") {
    this.addRule(action, {
      source: "list",
      urls,
      eagerness: "eager",
    });
  }

  private inject() {
    if (this.scriptElement) {
      this.scriptElement.remove();
    }

    this.scriptElement = document.createElement("script");
    this.scriptElement.type = "speculationrules";
    this.scriptElement.textContent = JSON.stringify(this.rules);
    document.head.appendChild(this.scriptElement);
  }

  // 현재 활성 규칙 확인 (디버깅용)
  getActiveRules(): Record<SpeculationAction, SpeculationRule[]> {
    return structuredClone(this.rules);
  }
}

export const speculationRules = new SpeculationRulesManager();
```

```typescript
// scripts/perf-audit.ts
import { execSync } from "node:child_process";
import { readFileSync, writeFileSync, mkdirSync } from "node:fs";
import { budget, type PerformanceBudget } from "../performance-budget.config";

interface LighthouseResult {
  categories: {
    performance: { score: number };
  };
  audits: {
    "largest-contentful-paint": { numericValue: number };
    "cumulative-layout-shift": { numericValue: number };
    "interaction-to-next-paint": { numericValue: number };
    "first-contentful-paint": { numericValue: number };
    "server-response-time": { numericValue: number };
    "total-blocking-time": { numericValue: number };
    "speed-index": { numericValue: number };
  };
}

interface AuditResult {
  metric: string;
  value: number;
  budget: number;
  pass: boolean;
  delta?: number;
  deltaPercent?: number;
}

async function runLighthouse(url: string): Promise<LighthouseResult> {
  const outputDir = ".perf-reports";
  mkdirSync(outputDir, { recursive: true });

  execSync(
    `npx @lhci/cli collect --url="${url}" ` +
    `--numberOfRuns=3 ` +
    `--settings.preset=desktop ` +
    `--settings.output=json ` +
    `--settings.outputPath=${outputDir}/lhr.json`,
    { stdio: "pipe" }
  );

  const raw = readFileSync(`${outputDir}/lhr.json`, "utf-8");
  return JSON.parse(raw) as LighthouseResult;
}

function evaluateMetrics(
  result: LighthouseResult,
  budget: PerformanceBudget
): AuditResult[] {
  const mapping: [string, keyof typeof result.audits, keyof typeof budget.metrics][] = [
    ["LCP", "largest-contentful-paint", "LCP"],
    ["CLS", "cumulative-layout-shift", "CLS"],
    ["INP", "interaction-to-next-paint", "INP"],
    ["FCP", "first-contentful-paint", "FCP"],
    ["TTFB", "server-response-time", "TTFB"],
    ["TBT", "total-blocking-time", "TBT"],
    ["Speed Index", "speed-index", "speedIndex"],
  ];

  return mapping.map(([name, auditKey, budgetKey]) => {
    const value = result.audits[auditKey].numericValue;
    const limit = budget.metrics[budgetKey];
    const margin = limit * budget.thresholds.errorMargin;

    return {
      metric: name,
      value: Math.round(value * 100) / 100,
      budget: limit,
      pass: value <= limit + margin,
    };
  });
}

function evaluateBundle(statsPath: string, budget: PerformanceBudget): AuditResult[] {
  const stats = JSON.parse(readFileSync(statsPath, "utf-8"));
  const results: AuditResult[] = [];

  // 전체 번들 사이즈
  const totalSize = stats.assets
    ?.filter((a: { name: string }) => a.name.endsWith(".js") || a.name.endsWith(".css"))
    .reduce((sum: number, a: { size: number }) => sum + a.size, 0) / 1024;

  results.push({
    metric: "Total Bundle Size (KB, gzip)",
    value: Math.round(totalSize * 100) / 100,
    budget: budget.bundle.maxTotalSize,
    pass: totalSize <= budget.bundle.maxTotalSize,
  });

  // 최대 청크 사이즈
  const maxChunk = Math.max(
    ...stats.assets
      ?.filter((a: { name: string }) => a.name.endsWith(".js"))
      .map((a: { size: number }) => a.size / 1024) ?? [0]
  );

  results.push({
    metric: "Max Chunk Size (KB, gzip)",
    value: Math.round(maxChunk * 100) / 100,
    budget: budget.bundle.maxChunkSize,
    pass: maxChunk <= budget.bundle.maxChunkSize,
  });

  return results;
}

function generateReport(results: AuditResult[]): string {
  const passed = results.filter((r) => r.pass).length;
  const total = results.length;
  const allPassed = passed === total;

  let report = `## Performance Budget Report\n\n`;
  report += allPassed
    ? `**PASS** (${passed}/${total})\n\n`
    : `**FAIL** (${passed}/${total})\n\n`;

  report += `| Metric | Value | Budget | Status |\n`;
  report += `|--------|-------|--------|--------|\n`;

  for (const r of results) {
    const status = r.pass ? "PASS" : "FAIL";
    report += `| ${r.metric} | ${r.value} | ${r.budget} | ${status} |\n`;
  }

  return report;
}

// 실행
async function main() {
  const previewUrl = process.env.PREVIEW_URL;
  const statsPath = process.env.STATS_PATH ?? "dist/stats.json";

  if (!previewUrl) {
    console.error("PREVIEW_URL 환경 변수가 필요합니다");
    process.exit(1);
  }

  console.log(`Auditing: ${previewUrl}`);

  const lhr = await runLighthouse(previewUrl);
  const metricResults = evaluateMetrics(lhr, budget);
  const bundleResults = evaluateBundle(statsPath, budget);
  const allResults = [...metricResults, ...bundleResults];

  const report = generateReport(allResults);
  console.log(report);

  writeFileSync(".perf-reports/report.md", report);

  const hasFailed = allResults.some((r) => !r.pass);
  if (hasFailed) {
    console.error("Performance budget exceeded!");
    process.exit(1);
  }
}

main();
```

```typescript
// scripts/build-benchmark.ts
import { execSync } from "node:child_process";
import { statSync, readdirSync } from "node:fs";
import { join } from "node:path";

interface BenchmarkResult {
  tool: string;
  coldBuild: number;    // ms
  hotBuild: number;     // ms
  outputSize: number;   // KB
  gzipSize: number;     // KB
  chunkCount: number;
  memoryPeak: number;   // MB
}

function measureBuild(
  command: string,
  outputDir: string,
  runs = 3
): Omit<BenchmarkResult, "tool"> {
  // Cold build (캐시 없음)
  execSync("rm -rf node_modules/.vite node_modules/.cache dist .next", {
    stdio: "pipe",
  });

  const coldTimes: number[] = [];
  const hotTimes: number[] = [];

  for (let i = 0; i < runs; i++) {
    if (i === 0) {
      // 첫 번째 실행: cold build
      const start = performance.now();
      execSync(command, { stdio: "pipe" });
      coldTimes.push(performance.now() - start);
    } else {
      // 이후 실행: hot build (캐시 활용)
      const start = performance.now();
      execSync(command, { stdio: "pipe" });
      hotTimes.push(performance.now() - start);
    }
  }

  // 출력물 분석
  const { totalSize, gzipSize, chunkCount } = analyzeOutput(outputDir);

  return {
    coldBuild: Math.round(median(coldTimes)),
    hotBuild: Math.round(median(hotTimes)),
    outputSize: Math.round(totalSize / 1024),
    gzipSize: Math.round(gzipSize / 1024),
    chunkCount,
    memoryPeak: getMemoryPeak(),
  };
}

function analyzeOutput(dir: string) {
  let totalSize = 0;
  let chunkCount = 0;

  const files = readdirSync(dir, { recursive: true }) as string[];
  for (const file of files) {
    const filePath = join(dir, file);
    try {
      const stat = statSync(filePath);
      if (stat.isFile() && (file.endsWith(".js") || file.endsWith(".css"))) {
        totalSize += stat.size;
        if (file.endsWith(".js")) chunkCount++;
      }
    } catch {
      // skip
    }
  }

  // gzip 사이즈 추정 (일반적으로 원본의 30-40%)
  const gzipSize = totalSize * 0.35;

  return { totalSize, gzipSize, chunkCount };
}

function median(arr: number[]): number {
  const sorted = [...arr].sort((a, b) => a - b);
  const mid = Math.floor(sorted.length / 2);
  return sorted.length % 2 ? sorted[mid] : (sorted[mid - 1] + sorted[mid]) / 2;
}

function getMemoryPeak(): number {
  try {
    const mem = execSync("ps aux --sort=-rss | head -2 | tail -1", {
      encoding: "utf-8",
    });
    const rss = mem.trim().split(/\s+/)[5];
    return Math.round(parseInt(rss, 10) / 1024);
  } catch {
    return 0;
  }
}
```

## 6. Web Worker RUM 수집기

Real User Monitoring (RUM) 데이터 수집을 Web Worker로 분리하여 메인 스레드 부하를 완전히 제거하는 패턴입니다.

### 6.1 Worker 설계

```typescript
// workers/rum-collector.worker.ts

interface RUMEvent {
  type: "cwv" | "navigation" | "resource" | "error" | "custom";
  timestamp: number;
  sessionId: string;
  pageUrl: string;
  data: Record<string, unknown>;
}

interface WorkerMessage {
  action: "collect" | "flush" | "configure";
  payload?: RUMEvent | RUMConfig;
}

interface RUMConfig {
  endpoint: string;
  batchSize: number;
  flushInterval: number; // ms
  sampleRate: number;    // 0-1
  environment: string;
  appVersion: string;
}

const defaultConfig: RUMConfig = {
  endpoint: "/api/rum",
  batchSize: 25,
  flushInterval: 10_000,
  sampleRate: 1.0,
  environment: "production",
  appVersion: "unknown",
};

let config = { ...defaultConfig };
let buffer: RUMEvent[] = [];
let flushTimer: ReturnType<typeof setInterval> | null = null;

// 메시지 핸들러
self.onmessage = (e: MessageEvent<WorkerMessage>) => {
  const { action, payload } = e.data;

  switch (action) {
    case "configure":
      config = { ...config, ...(payload as RUMConfig) };
      startFlushTimer();
      break;

    case "collect":
      // 샘플링 체크
      if (Math.random() > config.sampleRate) return;
      buffer.push(payload as RUMEvent);

      // 배치 사이즈 도달 시 즉시 전송
      if (buffer.length >= config.batchSize) {
        flush();
      }
      break;

    case "flush":
      flush();
      break;
  }
};

function startFlushTimer() {
  if (flushTimer) clearInterval(flushTimer);
  flushTimer = setInterval(flush, config.flushInterval);
}

async function flush() {
  if (buffer.length === 0) return;

  const events = [...buffer];
  buffer = [];

  try {
    const response = await fetch(config.endpoint, {
      method: "POST",
      headers: { "Content-Type": "application/json" },
      body: JSON.stringify({
        events,
        meta: {
          environment: config.environment,
          appVersion: config.appVersion,
          userAgent: navigator.userAgent,
          timestamp: Date.now(),
        },
      }),
      keepalive: true, // 페이지 이탈 시에도 전송 보장
    });

    if (!response.ok) {
      // 실패 시 버퍼에 다시 추가 (재시도)
      buffer.unshift(...events);
      self.postMessage({ type: "error", message: `Flush failed: ${response.status}` });
    } else {
      self.postMessage({ type: "flushed", count: events.length });
    }
  } catch (error) {
    // 네트워크 오류 시 버퍼에 복원
    buffer.unshift(...events);
    self.postMessage({ type: "error", message: String(error) });
  }
}
```

### 6.2 메인 스레드 SDK

```typescript
// rum-sdk.ts
type CWVMetricName = "LCP" | "FCP" | "CLS" | "INP" | "TTFB" | "FID";

interface RUMSDKConfig {
  endpoint?: string;
  sampleRate?: number;
  environment?: string;
  appVersion?: string;
  batchSize?: number;
  flushInterval?: number;
  collectCWV?: boolean;
  collectResources?: boolean;
  collectErrors?: boolean;
  collectNavigations?: boolean;
}

class RUMSDK {
  private worker: Worker | null = null;
  private sessionId: string;
  private initialized = false;

  constructor() {
    this.sessionId = this.generateSessionId();
  }

  init(config: RUMSDKConfig = {}) {
    if (this.initialized) return;
    this.initialized = true;

    // Worker 생성
    this.worker = new Worker(
      new URL("./workers/rum-collector.worker.ts", import.meta.url),
      { type: "module" }
    );

    // Worker 설정
    this.worker.postMessage({
      action: "configure",
      payload: {
        endpoint: config.endpoint ?? "/api/rum",
        batchSize: config.batchSize ?? 25,
        flushInterval: config.flushInterval ?? 10_000,
        sampleRate: config.sampleRate ?? 1.0,
        environment: config.environment ?? "production",
        appVersion: config.appVersion ?? "unknown",
      },
    });

    // Worker 메시지 수신 (디버깅)
    this.worker.onmessage = (e) => {
      if (e.data.type === "error") {
        console.warn("[RUM]", e.data.message);
      }
    };

    // 각 수집기 활성화
    if (config.collectCWV !== false) this.collectCoreWebVitals();
    if (config.collectResources !== false) this.collectResourceTimings();
    if (config.collectErrors !== false) this.collectErrors();
    if (config.collectNavigations !== false) this.collectNavigations();

    // 페이지 이탈 시 마지막 flush
    document.addEventListener("visibilitychange", () => {
      if (document.visibilityState === "hidden") {
        this.flush();
      }
    });
  }

  private collectCoreWebVitals() {
    // web-vitals 라이브러리 대신 직접 PerformanceObserver 사용
    // (Worker에서 의존성 줄이기)

    // LCP
    const lcpObserver = new PerformanceObserver((list) => {
      const entries = list.getEntries();
      const lastEntry = entries[entries.length - 1] as PerformanceEntry & {
        startTime: number;
        element?: Element;
      };
      this.send("cwv", {
        metric: "LCP" as CWVMetricName,
        value: lastEntry.startTime,
        element: lastEntry.element?.tagName,
      });
    });
    lcpObserver.observe({ type: "largest-contentful-paint", buffered: true });

    // CLS
    let clsValue = 0;
    const clsObserver = new PerformanceObserver((list) => {
      for (const entry of list.getEntries() as (PerformanceEntry & {
        hadRecentInput: boolean;
        value: number;
      })[]) {
        if (!entry.hadRecentInput) {
          clsValue += entry.value;
        }
      }
      this.send("cwv", { metric: "CLS" as CWVMetricName, value: clsValue });
    });
    clsObserver.observe({ type: "layout-shift", buffered: true });

    // INP
    const inpObserver = new PerformanceObserver((list) => {
      for (const entry of list.getEntries() as (PerformanceEntry & {
        duration: number;
        name: string;
      })[]) {
        this.send("cwv", {
          metric: "INP" as CWVMetricName,
          value: entry.duration,
          eventType: entry.name,
        });
      }
    });
    inpObserver.observe({ type: "event", buffered: true });

    // FCP
    const fcpObserver = new PerformanceObserver((list) => {
      for (const entry of list.getEntries()) {
        if (entry.name === "first-contentful-paint") {
          this.send("cwv", {
            metric: "FCP" as CWVMetricName,
            value: entry.startTime,
          });
        }
      }
    });
    fcpObserver.observe({ type: "paint", buffered: true });

    // TTFB
    const navObserver = new PerformanceObserver((list) => {
      const nav = list.getEntries()[0] as PerformanceNavigationTiming;
      this.send("cwv", {
        metric: "TTFB" as CWVMetricName,
        value: nav.responseStart - nav.requestStart,
      });
    });
    navObserver.observe({ type: "navigation", buffered: true });
  }

  private collectResourceTimings() {
    const observer = new PerformanceObserver((list) => {
      for (const entry of list.getEntries() as PerformanceResourceTiming[]) {
        // 자체 RUM 엔드포인트는 제외
        if (entry.name.includes("/api/rum")) continue;

        this.send("resource", {
          name: entry.name,
          type: entry.initiatorType,
          duration: entry.duration,
          transferSize: entry.transferSize,
          decodedBodySize: entry.decodedBodySize,
          startTime: entry.startTime,
        });
      }
    });
    observer.observe({ type: "resource", buffered: true });
  }

  private collectErrors() {
    window.addEventListener("error", (event) => {
      this.send("error", {
        message: event.message,
        filename: event.filename,
        lineno: event.lineno,
        colno: event.colno,
        stack: event.error?.stack,
      });
    });

    window.addEventListener("unhandledrejection", (event) => {
      this.send("error", {
        message: String(event.reason),
        type: "unhandledrejection",
        stack: event.reason?.stack,
      });
    });
  }

  private collectNavigations() {
    // Navigation API 활용
    if ("navigation" in window) {
      navigation.addEventListener("navigatesuccess", () => {
        this.send("navigation", {
          url: location.href,
          type: "spa",
        });
      });
    }
  }

  // 커스텀 이벤트 전송 (비즈니스 메트릭)
  track(eventName: string, data: Record<string, unknown> = {}) {
    this.send("custom", { eventName, ...data });
  }

  private send(type: string, data: Record<string, unknown>) {
    this.worker?.postMessage({
      action: "collect",
      payload: {
        type,
        timestamp: Date.now(),
        sessionId: this.sessionId,
        pageUrl: location.href,
        data,
      },
    });
  }

  flush() {
    this.worker?.postMessage({ action: "flush" });
  }

  destroy() {
    this.flush();
    this.worker?.terminate();
    this.worker = null;
    this.initialized = false;
  }

  private generateSessionId(): string {
    return crypto.randomUUID();
  }
}

export const rum = new RUMSDK();
```
