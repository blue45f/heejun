## 1. AI 기반 마이크로 FE 설계

> 마이크로 프론트엔드 아키텍처에서 **가장 어려운 것은 올바른 경계 설정**이다. AI를 활용하면 모놀리스 코드베이스의 의존성 그래프를 분석하여 최적의 분할 지점을 찾고, 팀 간 경계를 데이터 기반으로 설계할 수 있다. **전체 설계 시간의 30% 이상을 AI 기반 분석에 투자**한다.

### 2.3 MF 2.0 런타임 플러그인

```typescript
// mf-plugins/version-strategy.ts
import type { FederationRuntimePlugin } from '@module-federation/runtime';

const VersionStrategyPlugin: () => FederationRuntimePlugin = () => ({
  name: 'version-strategy-plugin',
  beforeInit(args) {
    // 환경 변수 또는 Feature Flag에 따라 리모트 URL 동적 결정
    const remoteOverrides = getRemoteOverrides();

    args.userOptions.remotes = args.userOptions.remotes?.map((remote) => {
      if (typeof remote === 'object' && remote.name && remoteOverrides[remote.name]) {
        return {
          ...remote,
          entry: remoteOverrides[remote.name],
        };
      }
      return remote;
    });

    return args;
  },
  errorLoadRemote(args) {
    // 리모트 로드 실패 시 폴백 버전 로드
    console.error(`Failed to load remote: ${args.id}`, args.error);
    return args;
  },
});

function getRemoteOverrides(): Record<string, string> {
  // Feature Flag 또는 URL 파라미터에서 오버라이드 정보 획득
  const params = new URLSearchParams(window.location.search);
  const overrides: Record<string, string> = {};

  params.forEach((value, key) => {
    if (key.startsWith('remote_')) {
      const remoteName = key.replace('remote_', '');
      overrides[remoteName] = value;
    }
  });

  return overrides;
}

export default VersionStrategyPlugin;
```

```typescript
// mf-plugins/preload-strategy.ts
import type { FederationRuntimePlugin } from '@module-federation/runtime';

const PreloadStrategyPlugin: () => FederationRuntimePlugin = () => ({
  name: 'preload-strategy-plugin',
  afterResolve(args) {
    // 현재 라우트 기반으로 다음에 필요할 리모트를 프리로드
    const predictedRemotes = predictNextRemotes(window.location.pathname);
    predictedRemotes.forEach((remoteName) => {
      const link = document.createElement('link');
      link.rel = 'prefetch';
      link.href = `${getRemoteBaseUrl(remoteName)}/remoteEntry.js`;
      document.head.appendChild(link);
    });
    return args;
  },
});

function predictNextRemotes(currentPath: string): string[] {
  const routePredictionMap: Record<string, string[]> = {
    '/': ['dashboard'],
```

```typescript
// shared/event-bus.ts
type EventHandler<T = unknown> = (payload: T) => void;

class MicroFrontendEventBus {
  private handlers = new Map<string, Set<EventHandler>>();
  private channel: BroadcastChannel;

  constructor(channelName = 'mfe-event-bus') {
    this.channel = new BroadcastChannel(channelName);
    this.channel.onmessage = (event) => {
      this.emit(event.data.type, event.data.payload, false);
    };
  }

  on<T>(eventType: string, handler: EventHandler<T>): () => void {
    if (!this.handlers.has(eventType)) {
      this.handlers.set(eventType, new Set());
    }
    this.handlers.get(eventType)!.add(handler as EventHandler);

    return () => {
      this.handlers.get(eventType)?.delete(handler as EventHandler);
    };
  }

  emit<T>(eventType: string, payload: T, broadcast = true): void {
    this.handlers.get(eventType)?.forEach((handler) => handler(payload));
    if (broadcast) {
      this.channel.postMessage({ type: eventType, payload });
    }
  }

  destroy(): void {
    this.channel.close();
    this.handlers.clear();
  }
}

export const eventBus = new MicroFrontendEventBus();
```

```typescript
// shared/auth-provider.ts
import { sharedStore } from './shared-store';
import { eventBus } from './event-bus';

interface AuthToken {
  accessToken: string;
  refreshToken: string;
  expiresAt: number;
}

class SharedAuthProvider {
  private tokenKey = 'mfe_auth_token';

  async initialize(): Promise<void> {
    const stored = localStorage.getItem(this.tokenKey);
    if (stored) {
      const token: AuthToken = JSON.parse(stored);
      if (token.expiresAt > Date.now()) {
        await this.fetchAndSetUser(token.accessToken);
      } else {
        await this.refreshToken(token.refreshToken);
      }
    }

    // 다른 탭의 로그인/로그아웃 동기화
    eventBus.on<{ action: 'login' | 'logout' }>('auth:changed', ({ action }) => {
      if (action === 'logout') {
        sharedStore.getState().setUser(null);
      } else {
        const stored = localStorage.getItem(this.tokenKey);
        if (stored) {
          const token: AuthToken = JSON.parse(stored);
          this.fetchAndSetUser(token.accessToken);
        }
      }
    });
  }

  async login(accessToken: string, refreshToken: string, expiresAt: number): Promise<void> {
    localStorage.setItem(this.tokenKey, JSON.stringify({ accessToken, refreshToken, expiresAt }));
    await this.fetchAndSetUser(accessToken);
    eventBus.emit('auth:changed', { action: 'login' });
  }

  logout(): void {
    localStorage.removeItem(this.tokenKey);
    sharedStore.getState().setUser(null);
    eventBus.emit('auth:changed', { action: 'logout' });
  }

  getAccessToken(): string | null {
    const stored = localStorage.getItem(this.tokenKey);
    if (!stored) return null;
    const token: AuthToken = JSON.parse(stored);
    return token.expiresAt > Date.now() ? token.accessToken : null;
  }

  private async fetchAndSetUser(accessToken: string): Promise<void> {
    const response = await fetch('/api/me', {
      headers: { Authorization: `Bearer ${accessToken}` },
    });
    if (response.ok) {
      const user = await response.json();
      sharedStore.getState().setUser(user);
    }
  }

  private async refreshToken(refreshToken: string): Promise<void> {
    const response = await fetch('/api/auth/refresh', {
      method: 'POST',
      headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify({ refreshToken }),
    });
    if (response.ok) {
      const { accessToken, refreshToken: newRefresh, expiresAt } = await response.json();
      await this.login(accessToken, newRefresh, expiresAt);
    }
  }
}

export const authProvider = new SharedAuthProvider();
```

```typescript
// host/src/ErrorBoundary.tsx
import React, { Component, type ErrorInfo, type ReactNode } from 'react';

interface ErrorBoundaryProps {
  children: ReactNode;
  fallback: ReactNode;
  onError?: (error: Error, errorInfo: ErrorInfo) => void;
  retryable?: boolean;
}

interface ErrorBoundaryState {
  hasError: boolean;
  error: Error | null;
  retryCount: number;
}

export class ErrorBoundary extends Component<ErrorBoundaryProps, ErrorBoundaryState> {
  private static MAX_RETRIES = 3;

  state: ErrorBoundaryState = { hasError: false, error: null, retryCount: 0 };

  static getDerivedStateFromError(error: Error): Partial<ErrorBoundaryState> {
    return { hasError: true, error };
  }

  componentDidCatch(error: Error, errorInfo: ErrorInfo): void {
    this.props.onError?.(error, errorInfo);
  }

  handleRetry = (): void => {
    if (this.state.retryCount < ErrorBoundary.MAX_RETRIES) {
      this.setState((prev) => ({
        hasError: false,
        error: null,
        retryCount: prev.retryCount + 1,
      }));
    }
  };

  render(): ReactNode {
    if (this.state.hasError) {
      if (this.props.retryable && this.state.retryCount < ErrorBoundary.MAX_RETRIES) {
        return (
          <div>
            {this.props.fallback}
            <button onClick={this.handleRetry}>
              재시도 ({this.state.retryCount}/{ErrorBoundary.MAX_RETRIES})
            </button>
          </div>
        );
      }
      return this.props.fallback;
    }
    return this.props.children;
  }
}
```

```typescript
import React from 'react';
import { loadRemote } from '@module-federation/runtime';

interface RemoteAppLoaderProps {
  remoteName: string;
  moduleName: string;
}

const remoteCache = new Map<string, React.ComponentType>();

export const RemoteAppLoader: React.FC<RemoteAppLoaderProps> = ({ remoteName, moduleName }) => {
  const cacheKey = `${remoteName}/${moduleName}`;

  if (!remoteCache.has(cacheKey)) {
    const RemoteComponent = React.lazy(async () => {
      const module = await loadRemote<{ default: React.ComponentType }>(
        `${remoteName}/${moduleName.replace('./', '')}`,
      );
      if (!module) throw new Error(`Failed to load remote: ${cacheKey}`);
      return module;
    });
    remoteCache.set(cacheKey, RemoteComponent);
  }

  const CachedComponent = remoteCache.get(cacheKey)!;
  return <CachedComponent />;
};
```

### 4.1 PR별 리모트 앱 오버라이드

> 핵심 아이디어: PR이 생성되면 해당 마이크로 앱만 별도 Preview URL로 배포하고, Host 앱은 **URL 파라미터 또는 쿠키**를 통해 특정 리모트를 오버라이드한다. 이를 통해 **다른 앱은 프로덕션 버전을 그대로 사용**하면서 변경된 앱만 Preview 버전으로 테스트할 수 있다.

```typescript
// shared/remote-override-manager.ts
interface RemoteOverrideConfig {
  remoteName: string;
  previewUrl: string;
  prNumber: number;
  branch: string;
  expiresAt: number;
}

class RemoteOverrideManager {
  private storageKey = 'mfe_remote_overrides';

  /**
   * URL 파라미터에서 오버라이드 정보를 읽고 저장한다.
   * 예: ?remote_dashboard=https://preview-pr-123.example.com/remoteEntry.js
   */
  initFromUrl(): void {
    const params = new URLSearchParams(window.location.search);
    params.forEach((value, key) => {
      if (key.startsWith('remote_')) {
        const remoteName = key.replace('remote_', '');
        this.setOverride(remoteName, value);
      }
    });
  }

  setOverride(remoteName: string, previewUrl: string, ttlHours = 4): void {
    const overrides = this.getOverrides();
    overrides[remoteName] = {
      remoteName,
      previewUrl,
      prNumber: this.extractPrNumber(previewUrl),
      branch: '',
      expiresAt: Date.now() + ttlHours * 3600 * 1000,
    };
    localStorage.setItem(this.storageKey, JSON.stringify(overrides));
  }

  getOverrides(): Record<string, RemoteOverrideConfig> {
    const stored = localStorage.getItem(this.storageKey);
    if (!stored) return {};
    const overrides: Record<string, RemoteOverrideConfig> = JSON.parse(stored);

    // 만료된 오버라이드 제거
    const now = Date.now();
    const valid = Object.fromEntries(
      Object.entries(overrides).filter(([, config]) => config.expiresAt > now),
    );
    localStorage.setItem(this.storageKey, JSON.stringify(valid));
    return valid;
  }

  getRemoteUrl(remoteName: string, defaultUrl: string): string {
    const overrides = this.getOverrides();
    return overrides[remoteName]?.previewUrl || defaultUrl;
  }

  clearAll(): void {
    localStorage.removeItem(this.storageKey);
  }

  private extractPrNumber(url: string): number {
    const match = url.match(/pr-(\d+)/);
    return match ? parseInt(match[1], 10) : 0;
  }
}

export const remoteOverrideManager = new RemoteOverrideManager();
```

```typescript
interface RemoteVersionMap {
  [remoteName: string]: {
    production: string;
    canary: string;
    preview?: string;
  };
}

const REMOTE_VERSIONS: RemoteVersionMap = {
  dashboard: {
    production: 'https://cdn.example.com/dashboard/v2.3.1/remoteEntry.js',
    canary: 'https://cdn.example.com/dashboard/canary/remoteEntry.js',
  },
  orders: {

  },
  settings: {
    production: 'https://cdn.example.com/settings/v3.0.0/remoteEntry.js',
    canary: 'https://cdn.example.com/settings/canary/remoteEntry.js',
  },
};

export function resolveRemoteUrl(remoteName: string): string {
  const flags = sharedStore.getState().featureFlags;
  const versions = REMOTE_VERSIONS[remoteName];

  if (!versions) throw new Error(`Unknown remote: ${remoteName}`);

  // Feature Flag 기반 버전 선택 우선순위
  // 1. PR Preview 오버라이드

  }
  // 2. Canary 배포
  if (flags[`canary_${remoteName}`]) {
    return versions.canary;
  }
  // 3. Production
  return versions.production;
```

```typescript
// deploy/version-manifest.ts
interface VersionManifest {
  timestamp: string;
  apps: Record<string, AppVersion>;
}

interface AppVersion {
  version: string;
  commitHash: string;
  buildTimestamp: string;
  remoteEntryUrl: string;
  integrity: string; // SRI hash
}

async function updateManifest(
  appName: string,
  version: string,
  commitHash: string,
  remoteEntryUrl: string,
): Promise<void> {
  const manifestUrl = 'https://cdn.example.com/manifest.json';
  const response = await fetch(manifestUrl);
  const manifest: VersionManifest = await response.json();

  manifest.timestamp = new Date().toISOString();
  manifest.apps[appName] = {
    version,
    commitHash,
    buildTimestamp: new Date().toISOString(),
    remoteEntryUrl,
    integrity: await computeSriHash(remoteEntryUrl),
  };

  // S3에 매니페스트 업로드
  console.log(`Updated manifest for ${appName}@${version}`);
}

async function computeSriHash(url: string): Promise<string> {
  const response = await fetch(url);
  const buffer = await response.arrayBuffer();
  const hash = await crypto.subtle.digest('SHA-384', buffer);
  return `sha384-${btoa(String.fromCharCode(...new Uint8Array(hash)))}`;
}

export { updateManifest };
```

### 6.2 프리로딩 전략

```typescript
// host/src/preload-manager.ts
interface PreloadConfig {
  remoteName: string;
  trigger: 'route-proximity' | 'viewport' | 'idle' | 'interaction';
  priority: 'high' | 'low';
}

const preloadConfigs: PreloadConfig[] = [
  { remoteName: 'dashboard', trigger: 'route-proximity', priority: 'high' },
  { remoteName: 'orders', trigger: 'idle', priority: 'low' },
  { remoteName: 'settings', trigger: 'interaction', priority: 'low' },
];

class PreloadManager {
  private preloaded = new Set<string>();

  /** 라우트 근접 기반 프리로드 */
  preloadByRoute(currentPath: string): void {
    const adjacencyMap: Record<string, string[]> = {
      '/dashboard': ['orders'],
      '/orders': ['dashboard'],
      '/settings': [],
    };

    const nextApps = adjacencyMap[currentPath] || [];
    nextApps.forEach((app) => this.preload(app));
  }

  /** Idle 시 백그라운드 프리로드 */
  preloadOnIdle(): void {
    if ('requestIdleCallback' in window) {
      requestIdleCallback(() => {
        preloadConfigs
          .filter((c) => c.trigger === 'idle' && !this.preloaded.has(c.remoteName))
          .forEach((c) => this.preload(c.remoteName));
      });
    }
  }

  /** Intersection Observer 기반 뷰포트 프리로드 */
  preloadOnViewport(element: HTMLElement, remoteName: string): () => void {
    const observer = new IntersectionObserver(
      (entries) => {
        entries.forEach((entry) => {
          if (entry.isIntersecting) {
            this.preload(remoteName);
            observer.disconnect();
          }
        });
      },
      { rootMargin: '200px' },
    );

    observer.observe(element);
    return () => observer.disconnect();
  }

  private preload(remoteName: string): void {
    if (this.preloaded.has(remoteName)) return;
    this.preloaded.add(remoteName);

    const link = document.createElement('link');
    link.rel = 'modulepreload';
    link.href = `${this.getRemoteEntryUrl(remoteName)}`;
    document.head.appendChild(link);
  }

  private getRemoteEntryUrl(remoteName: string): string {
    const urls: Record<string, string> = {
      dashboard: process.env.VITE_DASHBOARD_REMOTE_URL || '/dashboard/remoteEntry.js',
      orders: process.env.VITE_ORDERS_REMOTE_URL || '/orders/remoteEntry.js',
      settings: process.env.VITE_SETTINGS_REMOTE_URL || '/settings/remoteEntry.js',
    };
    return urls[remoteName] || '';
  }
}

export const preloadManager = new PreloadManager();
```

### 6.3 중복 의존성 탐지 스크립트

```typescript
// scripts/detect-duplicate-deps.ts
import * as fs from 'fs';
import * as path from 'path';

interface DuplicateReport {
  dependency: string;
  versions: { app: string; version: string }[];
  sizeImpact: string;
  recommendation: 'align' | 'share-singleton' | 'accept';
}

function detectDuplicates(appsDir: string): DuplicateReport[] {
  const apps = fs.readdirSync(appsDir).filter((d) =>
    fs.statSync(path.join(appsDir, d)).isDirectory(),
  );

  const depMap = new Map<string, { app: string; version: string }[]>();

  for (const app of apps) {
    const pkgPath = path.join(appsDir, app, 'package.json');
    if (!fs.existsSync(pkgPath)) continue;

    const pkg = JSON.parse(fs.readFileSync(pkgPath, 'utf-8'));
    const allDeps = { ...pkg.dependencies, ...pkg.devDependencies };

    for (const [dep, version] of Object.entries(allDeps)) {
      if (!depMap.has(dep)) depMap.set(dep, []);
      depMap.get(dep)!.push({ app, version: version as string });
    }
  }

  const duplicates: DuplicateReport[] = [];
  for (const [dep, versions] of depMap) {
    if (versions.length < 2) continue;
    const uniqueVersions = new Set(versions.map((v) => v.version));
    if (uniqueVersions.size > 1) {
      duplicates.push({
        dependency: dep,
        versions,
        sizeImpact: 'unknown',
        recommendation: dep.startsWith('react') ? 'share-singleton' : 'align',
      });
    }
  }

  return duplicates;
}

// 실행: npx tsx scripts/detect-duplicate-deps.ts
const report = detectDuplicates('./apps');
console.table(report);
```

## 7. 모노레포 vs 폴리레포 비교

| 기준 | 모노레포 (pnpm workspace) | 폴리레포 (각 앱 별도 레포) |
|---|---|---|
| 코드 공유 | 쉬움 (workspace: 프로토콜) | npm 패키지 발행 필요 |
| 의존성 관리 | 한 곳에서 통합 관리 | 앱별 독립 관리 |
| CI/CD 속도 | 영향 분석 필요 (Turborepo, Nx) | 자연스럽게 분리 |
| 팀 자율성 | 낮음 (공유 린트/포맷 규칙) | 높음 (팀별 규칙 가능) |
| 버전 관리 | 단일 히스토리, 원자적 커밋 | 앱별 독립 버전 관리 |
| 리팩토링 | 전체 코드 한 번에 변경 가능 | 패키지 단위 변경 후 릴리스 |
| 초기 셋업 | 복잡 (워크스페이스, 빌드 캐시 등) | 간단 (표준 프로젝트) |
| 권장 시나리오 | 5개 이하 앱, 같은 팀 | 5개 초과 앱, 독립 팀 |
