## 1. AI 활용 극대화 - 프롬프트 7선

> 모노레포 운영에서 AI를 활용하면 **패키지 구조 자동 설계**, **변경 영향 범위 지능적 분석**, **의존성 문제 자동 탐지 및 정리**가 가능하다. CI 시간 50% 단축, 의존성 문제 자동 해결 90%가 목표다.

### 프롬프트 1: 패키지 구조 설계

## 2. 멀티 베타 환경 모노레포 전략

> 모노레포에서 멀티 베타 환경을 운영하면 N개의 앱이 독립적으로 Preview 배포되고, affected 감지로 변경된 앱만 선택적으로 배포하며, 패키지별 스냅샷 릴리스로 의존 앱이 즉시 테스트할 수 있다.

### 2.1 앱별 독립 Preview 환경 (affected 감지 → 변경 앱만 배포)

```yaml
# .github/workflows/preview-deploy.yml
name: Preview Deploy (Affected Apps Only)

on:
  pull_request:
    types: [opened, synchronize, reopened]

concurrency:
  group: preview-${{ github.event.pull_request.number }}
  cancel-in-progress: true

jobs:
  detect-affected:
    runs-on: ubuntu-latest
    outputs:
      affected-apps: ${{ steps.affected.outputs.apps }}
      affected-packages: ${{ steps.affected.outputs.packages }}
    steps:
      - uses: actions/checkout@v4
        with:
          fetch-depth: 0

      - uses: pnpm/action-setup@v4
      - uses: actions/setup-node@v4
        with:
          node-version: 22
          cache: 'pnpm'

      - run: pnpm install --frozen-lockfile

      - id: affected
        run: |
          # Turborepo dry-run으로 affected 패키지 감지
          AFFECTED_APPS=$(pnpm turbo build --filter='./apps/*' --dry-run=json \
            | jq -r '[.tasks[].package] | unique | join(",")')
          AFFECTED_PKGS=$(pnpm turbo build --dry-run=json \
            | jq -r '[.tasks[].package] | unique | join(",")')

          echo "apps=$AFFECTED_APPS" >> "$GITHUB_OUTPUT"
          echo "packages=$AFFECTED_PKGS" >> "$GITHUB_OUTPUT"

  deploy-preview:
    needs: detect-affected
    if: needs.detect-affected.outputs.affected-apps != ''
    runs-on: ubuntu-latest
    strategy:
      matrix:
        app: ${{ fromJson(format('["{0}"]', join(fromJson(format('["{0}"]', needs.detect-affected.outputs.affected-apps)), '","'))) }}
      fail-fast: false
    steps:
      - uses: actions/checkout@v4

      - uses: pnpm/action-setup@v4
      - uses: actions/setup-node@v4
        with:
          node-version: 22
          cache: 'pnpm'

      - run: pnpm install --frozen-lockfile

      # Turborepo Remote Cache
      - run: pnpm turbo build --filter=${{ matrix.app }}
        env:
          TURBO_TOKEN: ${{ secrets.TURBO_TOKEN }}
          TURBO_TEAM: ${{ secrets.TURBO_TEAM }}

      # 앱별 독립 Preview URL 배포
      - uses: amondnet/vercel-action@v25
        with:
          vercel-token: ${{ secrets.VERCEL_TOKEN }}
          vercel-org-id: ${{ secrets.VERCEL_ORG_ID }}
          vercel-project-id: ${{ secrets.VERCEL_PROJECT_IDS[matrix.app] }}
          working-directory: apps/${{ matrix.app }}
          alias-domains: |
            preview-${{ github.event.pull_request.number }}-${{ matrix.app }}.example.com

      # PR에 배포 URL 코멘트
      - uses: actions/github-script@v7
        with:
          script: |
            github.rest.issues.createComment({
              owner: context.repo.owner,
              repo: context.repo.repo,
              issue_number: context.issue.number,
              body: `### Preview deployed: \`${{ matrix.app }}\`\nhttps://preview-${{ github.event.pull_request.number }}-${{ matrix.app }}.example.com`
            });
```

### 2.2 패키지별 버전 관리 + 스냅샷 릴리스

```typescript
// scripts/snapshot-release.ts
import { execSync } from 'node:child_process';

/**
 * PR 머지 전 스냅샷 버전 배포
 * 예: @repo/ui@0.0.0-snapshot-pr-123-20260401
 */
function createSnapshotRelease(prNumber: number): void {
  const date = new Date().toISOString().slice(0, 10).replace(/-/g, '');
  const tag = `snapshot-pr-${prNumber}-${date}`;

  // changeset에 스냅샷 버전 적용
  execSync(`pnpm changeset version --snapshot ${tag}`, { stdio: 'inherit' });

  // 스냅샷 버전으로 npm 배포 (dist-tag: snapshot)
  execSync(`pnpm changeset publish --tag snapshot`, { stdio: 'inherit' });

  console.log(`Snapshot released with tag: ${tag}`);
}

// CLI에서 호출: npx tsx scripts/snapshot-release.ts 123
const prNumber = parseInt(process.argv[2], 10);
if (!isNaN(prNumber)) {
  createSnapshotRelease(prNumber);
}
```

```yaml
# .github/workflows/snapshot-release.yml
name: Snapshot Release

on:
  pull_request:
    types: [labeled]

jobs:
  snapshot:
    if: contains(github.event.label.name, 'snapshot-release')
    runs-on: ubuntu-latest
    steps:
      - uses: actions/checkout@v4
      - uses: pnpm/action-setup@v4
      - uses: actions/setup-node@v4
        with:
          node-version: 22
          cache: 'pnpm'
          registry-url: 'https://registry.npmjs.org'

      - run: pnpm install --frozen-lockfile
      - run: pnpm turbo build --filter='./packages/*'
      - run: npx tsx scripts/snapshot-release.ts ${{ github.event.pull_request.number }}
        env:
          NODE_AUTH_TOKEN: ${{ secrets.NPM_TOKEN }}
```

### 2.3 N개 앱 동시 배포 매트릭스

```yaml
# .github/workflows/production-deploy.yml
name: Production Deploy Matrix

on:
  push:
    branches: [main]

jobs:
  detect-and-build:
    runs-on: ubuntu-latest
    outputs:
      deploy-matrix: ${{ steps.matrix.outputs.result }}
    steps:
      - uses: actions/checkout@v4
        with:
          fetch-depth: 2

      - uses: pnpm/action-setup@v4
      - uses: actions/setup-node@v4
        with:
          node-version: 22
          cache: 'pnpm'

      - run: pnpm install --frozen-lockfile

      - id: matrix
        uses: actions/github-script@v7
        with:
          script: |
            const { execSync } = require('child_process');
            const affected = execSync(
              'pnpm turbo build --filter="./apps/*" --dry-run=json'
            ).toString();
            const parsed = JSON.parse(affected);
            const apps = [...new Set(parsed.tasks.map(t => t.package))];

            // 배포 매트릭스 생성
            const matrix = apps.map(app => ({
              app,
              environment: 'production',
              url: `https://${app}.example.com`,
            }));

            return { include: matrix };

  deploy:
    needs: detect-and-build
    if: needs.detect-and-build.outputs.deploy-matrix != '{"include":[]}'
    runs-on: ubuntu-latest
    strategy:
      matrix: ${{ fromJson(needs.detect-and-build.outputs.deploy-matrix) }}
      max-parallel: 3
      fail-fast: false
    environment:
      name: ${{ matrix.environment }}
      url: ${{ matrix.url }}
    steps:
      - uses: actions/checkout@v4

      - uses: pnpm/action-setup@v4
      - uses: actions/setup-node@v4
        with:
          node-version: 22
          cache: 'pnpm'

      - run: pnpm install --frozen-lockfile

      - run: pnpm turbo build --filter=${{ matrix.app }}
        env:
          TURBO_TOKEN: ${{ secrets.TURBO_TOKEN }}
          TURBO_TEAM: ${{ secrets.TURBO_TEAM }}

      - name: Deploy ${{ matrix.app }}
        run: |
          echo "Deploying ${{ matrix.app }} to production..."
          # 실제 배포 명령 (Vercel, AWS, etc.)
```

### 2.4 Remote Caching (Turborepo + S3)

```typescript
// turbo.json
const turboConfig = {
  "$schema": "https://turbo.build/schema.json",
  "globalDependencies": ["**/.env.*local"],
```

```typescript
// scripts/setup-remote-cache.ts

/**
 * 자체 호스팅 Remote Cache 서버 설정 (S3 백엔드)
 * Turborepo의 자체 호스팅 캐시 서버 API를 구현
 */
interface RemoteCacheConfig {
  provider: 'vercel' | 's3' | 'gcs';
  bucket?: string;
  region?: string;
  prefix?: string;
  ttlDays: number;
}

function generateTurborepoEnv(config: RemoteCacheConfig): Record<string, string> {
  if (config.provider === 'vercel') {
    return {
      TURBO_TOKEN: '${VERCEL_TURBO_TOKEN}',
      TURBO_TEAM: '${VERCEL_TEAM_SLUG}',
    };
  }

  // 자체 호스팅 서버용
  return {
    TURBO_TOKEN: '${CUSTOM_TURBO_TOKEN}',
    TURBO_TEAM: 'my-team',
    TURBO_API: '${CUSTOM_TURBO_API_URL}', // 예: https://cache.example.com
  };
}

/** S3 캐시 정리 (TTL 초과 항목 삭제) 스크립트 생성 */
function generateCacheCleanupScript(config: RemoteCacheConfig): string {
  if (config.provider !== 's3') return '';

  return `
#!/bin/bash
# S3 Remote Cache 정리 (${config.ttlDays}일 초과 항목 삭제)
aws s3 ls s3://${config.bucket}/${config.prefix ?? 'turbo-cache'}/ --recursive \\
  | awk -v cutoff="$(date -d '-${config.ttlDays} days' +%Y-%m-%d)" '$1 < cutoff {print $4}' \\
  | xargs -I {} aws s3 rm s3://${config.bucket}/{}
`.trim();
}

export { generateTurborepoEnv, generateCacheCleanupScript };
export type { RemoteCacheConfig };
```

## 3. Turborepo vs Nx vs pnpm workspace 비교

| 항목 | Turborepo | Nx | pnpm workspace (단독) |
|------|-----------|----|-----------------------|
| **설정 복잡도** | 낮음 (turbo.json) | 중간 (nx.json + project.json) | 최소 (pnpm-workspace.yaml) |
| **빌드 오케스트레이션** | 태스크 기반 DAG | 태스크 + 프로젝트 그래프 | 없음 (수동 스크립트) |
| **Remote Caching** | 내장 (Vercel / 자체호스팅) | Nx Cloud (유료) / 자체호스팅 | 없음 |
| **affected 감지** | `--filter` (기본) | `nx affected` (고급) | 없음 (수동 구현) |
| **코드 생성기** | 없음 | 내장 (generators) | 없음 |
| **플러그인 생태계** | 최소 | 풍부 (React, Next.js, Vite 등) | N/A |
| **학습 곡선** | 낮음 | 중간~높음 | 최소 |
| **증분 빌드** | 해시 기반 캐시 | 해시 기반 + 의존성 분석 | 없음 |
| **모놀리스 → 모노레포** | 단계적 도입 용이 | 마이그레이션 도구 내장 | 수동 |
| **CI 통합** | GitHub Actions 친화적 | CI 전용 에이전트 지원 | 범용 |
| **추천 시나리오** | 중소규모, 빠른 도입 | 대규모, 엔터프라이즈 | 초소규모, 단순 공유 |

### 선택 기준 의사결정 트리

```text
모노레포 규모는?
```

### 4.4 순환 의존성 검사 자동화

```typescript
// scripts/check-circular-deps.ts
import { execSync } from 'node:child_process';

interface PackageDep {
  name: string;
  dependencies: string[];
}

function detectCircularDependencies(): string[][] {
  // pnpm list로 의존성 그래프 추출
  const output = execSync('pnpm list -r --json --depth 1').toString();
  const packages: Array<{ name: string; dependencies?: Record<string, unknown> }> =
    JSON.parse(output);

  const graph = new Map<string, string[]>();

  for (const pkg of packages) {
    const internalDeps = Object.keys(pkg.dependencies ?? {})
      .filter((dep) => dep.startsWith('@repo/'));
    graph.set(pkg.name, internalDeps);
  }

  // DFS로 순환 감지
  const cycles: string[][] = [];
  const visited = new Set<string>();
  const stack = new Set<string>();

  function dfs(node: string, path: string[]): void {
    if (stack.has(node)) {
      const cycleStart = path.indexOf(node);
      cycles.push(path.slice(cycleStart).concat(node));
      return;
    }
    if (visited.has(node)) return;

    visited.add(node);
    stack.add(node);

    for (const dep of graph.get(node) ?? []) {
      dfs(dep, [...path, node]);
    }

    stack.delete(node);
  }

  for (const pkg of graph.keys()) {
    dfs(pkg, []);
  }

  return cycles;
}

const cycles = detectCircularDependencies();
if (cycles.length > 0) {
  console.error('Circular dependencies detected:');
  cycles.forEach((cycle) => console.error(`  ${cycle.join(' → ')}`));
  process.exit(1);
} else {
  console.log('No circular dependencies found.');
}
```

### 5.2 자동 changeset 생성 (커밋 메시지 기반)

```typescript
// scripts/auto-changeset.ts
import { execSync } from 'node:child_process';
import { writeFileSync, mkdirSync, existsSync } from 'node:fs';
import { randomBytes } from 'node:crypto';

interface ChangesetEntry {
  packages: string[];
  type: 'major' | 'minor' | 'patch';
  summary: string;
}

/** 커밋 메시지에서 changeset 정보 추출 */
function parseCommitMessage(message: string): ChangesetEntry | null {
  // Conventional Commits 파싱
  const match = message.match(
    /^(feat|fix|docs|style|refactor|perf|test|chore)(?:\((.+)\))?(!)?:\s+(.+)/,
  );
  if (!match) return null;

  const [, type, scope, breaking, summary] = match;

  // 영향 받는 패키지 감지
  const affectedPackages = detectAffectedPackages();
  if (affectedPackages.length === 0) return null;

  // semver 타입 결정
  let semverType: 'major' | 'minor' | 'patch';
  if (breaking) {
    semverType = 'major';
  } else if (type === 'feat') {
    semverType = 'minor';
  } else {
    semverType = 'patch';
  }

  return {
    packages: affectedPackages,
    type: semverType,
    summary: `${scope ? `[${scope}] ` : ''}${summary}`,
  };
}

function detectAffectedPackages(): string[] {
  try {
    const output = execSync(
      'pnpm turbo build --filter="./packages/*" --dry-run=json',
    ).toString();
    const parsed = JSON.parse(output);
    return [...new Set<string>(parsed.tasks.map((t: { package: string }) => t.package))];
  } catch {
    return [];
  }
}

function generateChangeset(entry: ChangesetEntry): void {
  const id = randomBytes(4).toString('hex');
  const dir = '.changeset';
  if (!existsSync(dir)) mkdirSync(dir, { recursive: true });

  const content = [
    '---',
    ...entry.packages.map((pkg) => `"${pkg}": ${entry.type}`),
    '---',
    '',
    entry.summary,
    '',
  ].join('\n');

  writeFileSync(`${dir}/auto-${id}.md`, content);
  console.log(`Generated changeset: ${dir}/auto-${id}.md`);
}

// 실행
const lastCommit = execSync('git log -1 --pretty=%B').toString().trim();
const entry = parseCommitMessage(lastCommit);
if (entry) {
  generateChangeset(entry);
}
```

### 6.2 커스텀 affected 분석 스크립트

```typescript
// scripts/affected.ts
import { execSync } from 'node:child_process';

interface AffectedResult {
  directlyChanged: string[];
  transitivelyAffected: string[];
  actions: Record<string, ('build' | 'test' | 'deploy')[]>;
}

function analyzeAffected(baseBranch: string = 'main'): AffectedResult {
  // 변경된 파일 목록
  const changedFiles = execSync(`git diff --name-only ${baseBranch}...HEAD`)
    .toString()
    .trim()
    .split('\n')
    .filter(Boolean);

  // 파일 → 패키지 매핑
  const directlyChanged = new Set<string>();
  for (const file of changedFiles) {
    if (file.startsWith('apps/')) {
      const app = file.split('/')[1];
      directlyChanged.add(`@repo/${app}`);
    } else if (file.startsWith('packages/')) {
      const pkg = file.split('/')[1];
      directlyChanged.add(`@repo/${pkg}`);
    }
  }

  // Turborepo로 transitive 영향 분석
  const turboOutput = execSync(
    `pnpm turbo build --filter='...[${baseBranch}]' --dry-run=json`,
  ).toString();
  const parsed = JSON.parse(turboOutput);
  const allAffected = new Set<string>(
    parsed.tasks.map((t: { package: string }) => t.package),
  );

  // 직접 변경 vs 간접 영향 분리
  const transitivelyAffected = [...allAffected].filter(
    (pkg) => !directlyChanged.has(pkg),
  );

  // 작업 결정: 직접 변경 → 빌드+테스트+배포, 간접 영향 → 빌드+테스트
  const actions: Record<string, ('build' | 'test' | 'deploy')[]> = {};
  for (const pkg of directlyChanged) {
    actions[pkg] = pkg.startsWith('@repo/') && !pkg.includes('config-')
      ? ['build', 'test', 'deploy']
      : ['build', 'test'];
  }
  for (const pkg of transitivelyAffected) {
    actions[pkg] = ['build', 'test'];
  }

  return {
    directlyChanged: [...directlyChanged],
    transitivelyAffected,
    actions,
  };
}

// CLI 실행
const result = analyzeAffected(process.argv[2]);
console.log(JSON.stringify(result, null, 2));

export { analyzeAffected };
export type { AffectedResult };
```

## 7. CI 최적화 (캐시, 병렬, Remote Caching)

### 7.1 최적화 전략 비교

| 전략 | 효과 | 구현 난이도 | 적용 우선순위 |
|------|------|------------|-------------|
| **Remote Caching** | 빌드 시간 60~80% 단축 | 낮음 | 1순위 |
| **affected 기반 선별 실행** | 불필요 작업 50~70% 제거 | 낮음 | 1순위 |
| **작업 병렬화 (DAG)** | 전체 시간 30~50% 단축 | 자동 (Turborepo) | 1순위 |
| **Docker 레이어 캐시** | 빌드 시간 40~60% 단축 | 중간 | 2순위 |
| **pnpm store 캐시** | install 시간 70% 단축 | 낮음 | 2순위 |
| **테스트 분할 실행** | 테스트 시간 N분의1 | 중간 | 3순위 |

### 7.2 최적화된 CI 파이프라인

```yaml
# .github/workflows/ci.yml
name: CI

on:
  pull_request:
    branches: [main]
  push:
    branches: [main]

env:
  TURBO_TOKEN: ${{ secrets.TURBO_TOKEN }}
  TURBO_TEAM: ${{ secrets.TURBO_TEAM }}

jobs:
  setup:
    runs-on: ubuntu-latest
    outputs:
      affected-apps: ${{ steps.affected.outputs.apps }}
      has-changes: ${{ steps.affected.outputs.has-changes }}
    steps:
      - uses: actions/checkout@v4
        with:
          fetch-depth: 0

      - uses: pnpm/action-setup@v4
      - uses: actions/setup-node@v4
        with:
          node-version: 22
          cache: 'pnpm'

      - run: pnpm install --frozen-lockfile

      - id: affected
        run: |
          AFFECTED=$(pnpm turbo build --dry-run=json --filter='...[HEAD^]' \
            | jq -r '[.tasks[].package] | unique')
          echo "apps=$AFFECTED" >> "$GITHUB_OUTPUT"
          echo "has-changes=$([ "$AFFECTED" != "[]" ] && echo true || echo false)" >> "$GITHUB_OUTPUT"

  quality:
    needs: setup
    if: needs.setup.outputs.has-changes == 'true'
    runs-on: ubuntu-latest
    strategy:
      matrix:
        task: [lint, typecheck, test]
      fail-fast: false
    steps:
      - uses: actions/checkout@v4
      - uses: pnpm/action-setup@v4
      - uses: actions/setup-node@v4
        with:
          node-version: 22
          cache: 'pnpm'

      - run: pnpm install --frozen-lockfile

      # Remote Cache 활용으로 중복 작업 스킵
      - run: pnpm turbo ${{ matrix.task }} --filter='...[HEAD^]'

  build:
    needs: [setup, quality]
    if: needs.setup.outputs.has-changes == 'true'
    runs-on: ubuntu-latest
    steps:
      - uses: actions/checkout@v4
      - uses: pnpm/action-setup@v4
      - uses: actions/setup-node@v4
        with:
          node-version: 22
          cache: 'pnpm'

      - run: pnpm install --frozen-lockfile
      - run: pnpm turbo build --filter='...[HEAD^]'
```
